NovaOryn.Kernel.Serial.IsPciSerialController

public static Boolean IsPciSerialController(UInt32 classCode) => ((classCode >> 16) & 0xFFU) == 0x07U && ((classCode >> 8) & 0xFFU) == 0x00U
