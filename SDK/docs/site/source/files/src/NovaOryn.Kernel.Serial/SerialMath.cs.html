5/// <summary>Pure serial/UART identification and baud-rate helpers shared by runtime code and tests.</summary>
6public static class SerialMath
7{
8    /// <summary>Gets whether a PCI class code identifies a communications/serial controller.</summary>
9    public static Boolean IsPciSerialController(UInt32 classCode) => ((classCode >> 16) & 0xFFU) == 0x07U && ((classCode >> 8) & 0xFFU) == 0x00U;
10
11    /// <summary>Gets whether the PCI serial programming interface is compatible with the 8250-to-16950 register model.</summary>
12    public static Boolean Is16550CompatibleProgrammingInterface(UInt32 classCode)
13    {
14        if (!IsPciSerialController(classCode)) return false;
15        UInt32 programmingInterface = classCode & 0xFFU;
16        return programmingInterface <= 0x06U;
17    }
18
19    /// <summary>Calculates the integer UART divisor for the standard 1.8432 MHz input clock.</summary>
20    public static Boolean TryCalculateDivisor(UInt32 baudRate, out UInt16 divisor)
21    {
22        divisor = 0;
23        if (baudRate == 0U || baudRate > 115200U || (115200U % baudRate) != 0U) return false;
24        UInt32 value = 115200U / baudRate;
25        if (value == 0U || value > UInt16.MaxValue) return false;
26        divisor = (UInt16)value;
27        return true;
28    }
