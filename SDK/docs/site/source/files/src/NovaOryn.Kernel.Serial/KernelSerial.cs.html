37    /// <summary>Discovers PCI UARTs, attaches started VirtIO consoles, and mirrors KernelConsole output to available secondary transports.</summary>
38    public static Boolean Initialize()
39    {
40        if (_initialized) return true;
41        if (!_legacy16550 && !InitializeEarly16550()) return false;
42        if (!KernelPci.IsInitialized() || !KernelVirtio.IsInitialized()) return false;
43        _pciDiscovered = 0U; _pciOnline = 0U; _virtioConsoles = 0U; _hasPciMirror = false; _virtioMirror = default; _transmitFailures = 0UL;
44        DiscoverPciUarts();
45        DiscoverVirtioConsoles();
46        if ((_hasPciMirror || _virtioMirror.Value != 0U) && !KernelConsole.SetSecondarySerialWriter(&WriteSecondaryByte)) return false;
47        _initialized = true;
48        return true;
49    }

65    private static Boolean WriteSecondaryByte(Byte value)
66    {
67        Boolean attempted = false;
68        if (_hasPciMirror)
69        {
70            attempted = true;
71            if (!WritePciByte(value)) _transmitFailures++;
72        }
73        if (_virtioMirror.Value != 0U)
74        {
75            attempted = true;
76            Byte local = value;
77            if (!KernelVirtio.WriteConsole(_virtioMirror, &local, 1U)) _transmitFailures++;
78        }
79        _ = attempted;
80        return true;
81    }

83    private static Boolean DiscoverPciUarts()
84    {
85        UInt32 count = KernelPci.GetDeviceCount();
86        for (UInt32 index=0U; index<count; index++)
87        {
88            if (!KernelPci.TryGetDevice(index, out PciDeviceInfo device) || !SerialMath.IsPciSerialController(device.ClassCode)) continue;
89            _pciDiscovered++;
90            if (!SerialMath.Is16550CompatibleProgrammingInterface(device.ClassCode)) continue;
91            if (!TryInitializePciUart(device, out Byte barIndex, out PciBarType barType, out UInt16 ioBase)) continue;
92            _pciOnline++;
93            if (_hasPciMirror) continue;
94            _hasPciMirror = true; _pciMirrorLocation = device.Location; _pciMirrorBar = barIndex; _pciMirrorBarType = barType; _pciMirrorIoBase = ioBase;
95        }
96        return true;
97    }

99    private static Boolean TryInitializePciUart(PciDeviceInfo device, out Byte selectedBar, out PciBarType selectedType, out UInt16 ioBase)
100    {
101        selectedBar=0; selectedType=PciBarType.None; ioBase=0;
102        for (Byte barIndex=0; barIndex<6; barIndex++)
103        {
104            if (!KernelPci.TryGetBar(device.Location, barIndex, out PciBarInfo bar) || bar.Length < 8UL) continue;
105            if (bar.Type == PciBarType.Io)
106            {
107                if (bar.Address == 0UL || bar.Address > UInt16.MaxValue) continue;
108                if (!EnablePciDecode(device.Location, true, false)) return false;
109                UInt16 basePort=(UInt16)bar.Address;
110                if (!ConfigureUart(device.Location, barIndex, PciBarType.Io, basePort)) return false;
111                selectedBar=barIndex; selectedType=PciBarType.Io; ioBase=basePort; return true;
112            }
113            if (bar.Type == PciBarType.Memory32 || bar.Type == PciBarType.Memory64)
114            {
115                if (!EnablePciDecode(device.Location, false, true)) return false;
116                if (!ConfigureUart(device.Location, barIndex, bar.Type, 0)) return false;
117                selectedBar=barIndex; selectedType=bar.Type; return true;
118            }
119        }
120        return false;
121    }

123    private static Boolean ConfigureUart(PciLocation location, Byte barIndex, PciBarType barType, UInt16 ioBase)
124    {
125        if (!SerialMath.TryCalculateDivisor(DefaultBaudRate, out UInt16 divisor)) return false;
126        if (!WriteRegister(location,barIndex,barType,ioBase,1,0x00)) return false;
127        if (!WriteRegister(location,barIndex,barType,ioBase,3,0x80)) return false;
128        if (!WriteRegister(location,barIndex,barType,ioBase,0,(Byte)(divisor & 0xFF))) return false;
129        if (!WriteRegister(location,barIndex,barType,ioBase,1,(Byte)(divisor >> 8))) return false;
130        if (!WriteRegister(location,barIndex,barType,ioBase,3,0x03)) return false;
131        if (!WriteRegister(location,barIndex,barType,ioBase,2,0xC7)) return false;
132        return WriteRegister(location,barIndex,barType,ioBase,4,0x0B);
133    }

147    private static Boolean WritePciByte(Byte value)
148    {
149        for(UInt32 attempt=0U;attempt<TransmitPollLimit;attempt++)
150        {
151            if(!ReadRegister(_pciMirrorLocation,_pciMirrorBar,_pciMirrorBarType,_pciMirrorIoBase,5,out Byte status))return false;
152            if((status&0x20)!=0)return WriteRegister(_pciMirrorLocation,_pciMirrorBar,_pciMirrorBarType,_pciMirrorIoBase,0,value);
153        }
154        return false;
155    }

157    private static Boolean EnablePciDecode(PciLocation location, Boolean io, Boolean memory)
158    {
159        if(!KernelPci.TryRead16(location,0x04,out UInt16 command))return false;
160        UInt16 next=command;if(io)next=(UInt16)(next|0x0001);if(memory)next=(UInt16)(next|0x0002);
161        return next==command||KernelPci.TryWrite16(location,0x04,next);
162    }
