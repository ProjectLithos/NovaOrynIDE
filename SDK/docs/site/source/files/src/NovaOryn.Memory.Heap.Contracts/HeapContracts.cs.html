5/// <summary>Identifies a heap allocation policy supplied by NovaOryn or an SDK consumer.</summary>
6/// <nova.when>Use when selecting or reporting the raw-storage methodology behind a kernel heap.</nova.when>
7/// <nova.depends>NovaOryn.Core</nova.depends>
8[SupportedArchitecture(SupportedArchitecture.All)]
9[BootStage(BootStage.ManagedBootstrap)]
10public enum KernelHeapMethodology
11{
12    /// <summary>Uses monotonic bump allocation; individual releases are not supported.</summary>
13    Bump = 0,
14    /// <summary>Uses address-ordered first-fit allocation with adjacent-block coalescing.</summary>
15    FirstFit = 1,
16    /// <summary>Uses an SDK-consumer implementation of the heap contract.</summary>
17    Custom = 2
18}

147/// <summary>Defines the common raw-storage contract for kernel heap methodologies.</summary>
148/// <nova.when>Implement to provide a custom heap while retaining NovaOryn-compatible allocation semantics.</nova.when>
149/// <nova.depends>KernelHeapAllocationRequest, KernelHeapAllocation, KernelHeapStatistics</nova.depends>
150[SupportedArchitecture(SupportedArchitecture.All)]
151[BootStage(BootStage.ManagedBootstrap)]
152public interface IKernelHeap
153{
154    /// <summary>Gets the selected methodology.</summary>
155    /// <nova.when>Use for diagnostics or methodology-specific policy decisions.</nova.when>
156    /// <nova.depends>The concrete heap implementation.</nova.depends>
157    KernelHeapMethodology Methodology { get; }
158
159    /// <summary>Attempts to allocate one raw byte range.</summary>
160    /// <nova.when>Use whenever a kernel subsystem needs heap-owned raw storage.</nova.when>
161    /// <nova.depends>A valid request and sufficient implementation capacity.</nova.depends>
162    /// <returns><see langword="true"/> when a live allocation was created.</returns>
163    /// <example><code>bool ok = heap.TryAllocate(KernelHeapAllocationRequest.Default(128), out KernelHeapAllocation allocation);</code></example>
164    bool TryAllocate(KernelHeapAllocationRequest request, out KernelHeapAllocation allocation);
165
166    /// <summary>Attempts to release one exact live allocation.</summary>
167    /// <nova.when>Use when storage returned by this heap is no longer required.</nova.when>
168    /// <nova.depends>An allocation value previously returned by this heap.</nova.depends>
169    /// <returns><see langword="true"/> when the token and range identify a live allocation.</returns>
170    /// <example><code>bool released = heap.TryRelease(allocation);</code></example>
171    bool TryRelease(KernelHeapAllocation allocation);

173    /// <summary>Gets current allocation accounting.</summary>
174    /// <nova.when>Use when reporting heap capacity, consumption, or fragmentation.</nova.when>
