63    /// <summary>Creates a validated descriptor without permitting arithmetic overflow.</summary>
64    /// <nova.when>Use for firmware translation, explicit reservations, and split output ranges.</nova.when>
65    /// <nova.depends>4 KiB page alignment</nova.depends>
66    /// <returns><see langword="true"/> when the range is non-empty, aligned, and representable.</returns>
67    /// <example><code>bool valid = MemoryDescriptor.TryCreate(new PhysicalAddress(0x100000), 256, MemoryType.UsableConventional, MemoryCacheAttributes.WriteBack, MemoryRuntimeStatus.NotRuntime, MemoryAvailability.AvailableAfterExitBootServices, false, 0, out MemoryDescriptor descriptor);</code></example>
68    public static bool TryCreate(
69        PhysicalAddress physicalStart,
70        ulong pageCount,
71        MemoryType memoryType,
72        MemoryCacheAttributes cacheAttributes,
73        MemoryRuntimeStatus runtimeStatus,
74        MemoryAvailability availability,
75        bool hasNumaNode,
76        uint numaNode,
77        out MemoryDescriptor descriptor)
78    {
79        descriptor = default;
80        if (pageCount == 0 || (physicalStart.Value & 0xFFFUL) != 0) return false;
81        if (pageCount > ulong.MaxValue / 4096UL) return false;
82        ulong length = pageCount * 4096UL;
83        if (physicalStart.Value > ulong.MaxValue - length) return false;
84        descriptor = new MemoryDescriptor(physicalStart, pageCount, length, memoryType, cacheAttributes, runtimeStatus, availability, hasNumaNode, numaNode);
85        return true;
86    }

88    /// <summary>Creates a page-aligned split of an existing descriptor.</summary>
89    /// <nova.when>Use within a normaliser after selecting interval boundaries.</nova.when>
90    /// <nova.depends>TryCreate</nova.depends>
91    /// <returns><see langword="true"/> when the requested subrange lies inside this descriptor.</returns>
92    /// <example><code>bool split = descriptor.TrySlice(0x100000, 0x110000, out MemoryDescriptor part);</code></example>
93    public bool TrySlice(ulong start, ulong endExclusive, out MemoryDescriptor descriptor)
94    {
95        descriptor = default;
96        if (start < PhysicalStart.Value || endExclusive > EndExclusive || start >= endExclusive) return false;
97        if ((start & 0xFFFUL) != 0 || (endExclusive & 0xFFFUL) != 0) return false;
98        return TryCreate(new PhysicalAddress(start), (endExclusive - start) / 4096UL, MemoryType, CacheAttributes, RuntimeStatus, Availability, HasNumaNode, NumaNode, out descriptor);
99    }

101    /// <summary>Determines whether two adjacent ranges have identical metadata.</summary>
102    /// <nova.when>Use before merging compatible adjacent output ranges.</nova.when>
103    /// <nova.depends>Validated descriptors</nova.depends>
104    /// <returns><see langword="true"/> when the descriptors can be coalesced without losing information.</returns>
105    /// <example><code>bool compatible = first.IsMergeCompatible(second);</code></example>
106    public bool IsMergeCompatible(MemoryDescriptor other)
107    {
108        return EndExclusive == other.PhysicalStart.Value &&
109            MemoryType == other.MemoryType && CacheAttributes == other.CacheAttributes &&
110            RuntimeStatus == other.RuntimeStatus && Availability == other.Availability &&
111            HasNumaNode == other.HasNumaNode && (!HasNumaNode || NumaNode == other.NumaNode);
112    }

115/// <summary>Defines a NovaOryn-owned range that must override firmware availability.</summary>
116/// <nova.when>Use to reserve the kernel image, boot structures, framebuffer, MMIO, page tables, or early allocations.</nova.when>
117/// <nova.depends>MemoryDescriptor.TryCreate</nova.depends>
118public readonly struct MemoryReservation
119{
120    private MemoryReservation(MemoryDescriptor descriptor) => Descriptor = descriptor;
121
122    /// <summary>Gets the validated reservation descriptor.</summary>
123    /// <nova.when>Use when overlaying this reservation during normalisation.</nova.when>
124    public MemoryDescriptor Descriptor { get; }
125
126    /// <summary>Creates a validated reservation with permanently unavailable ownership.</summary>
127    /// <nova.when>Use before normalisation to overlay NovaOryn-owned physical ranges.</nova.when>
128    /// <nova.depends>4 KiB alignment</nova.depends>
129    /// <returns><see langword="true"/> when the reservation type and range are valid.</returns>
130    /// <example><code>bool reserved = MemoryReservation.TryCreate(new PhysicalAddress(0x200000), 32, MemoryType.PageTables, MemoryCacheAttributes.WriteBack, out MemoryReservation reservation);</code></example>
131    public static bool TryCreate(PhysicalAddress start, ulong pageCount, MemoryType type, MemoryCacheAttributes attributes, out MemoryReservation reservation)
132    {
133        reservation = default;
134        if (type is not (MemoryType.LoaderKernelImage or MemoryType.BootStructures or MemoryType.Framebuffer or MemoryType.MemoryMappedIo or MemoryType.PageTables or MemoryType.EarlyAllocatorAllocations or MemoryType.FirmwareReserved)) return false;
135        MemoryAvailability availability = type == MemoryType.MemoryMappedIo || type == MemoryType.Framebuffer || type == MemoryType.FirmwareReserved
136            ? MemoryAvailability.PermanentlyReserved
137            : MemoryAvailability.Unavailable;
138        if (!MemoryDescriptor.TryCreate(start, pageCount, type, attributes, MemoryRuntimeStatus.NotRuntime, availability, false, 0, out MemoryDescriptor descriptor)) return false;
139        reservation = new MemoryReservation(descriptor);
140        return true;
141    }
142}
