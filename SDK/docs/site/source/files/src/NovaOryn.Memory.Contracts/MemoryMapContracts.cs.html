3/// <summary>Provides immutable indexed and cursor-based diagnostics over a normalised memory map.</summary>
4/// <nova.when>Use after normalisation and before constructing a physical allocator.</nova.when>
5/// <nova.depends>MemoryDescriptor</nova.depends>
6public sealed class NormalisedMemoryMap
7{
8    private readonly MemoryDescriptor[] _descriptors;
9
10    internal NormalisedMemoryMap(MemoryDescriptor[] descriptors, int count, MemoryMapNormalisationMethod method)
11    {
12        _descriptors = new MemoryDescriptor[count];
13        Array.Copy(descriptors, _descriptors, count);
14        Method = method;
15        ulong total = 0;
16        ulong usable = 0;
17        for (int index = 0; index < count; index++)
18        {
19            total += _descriptors[index].Length;
20            if (_descriptors[index].Availability == MemoryAvailability.AvailableAfterExitBootServices) usable += _descriptors[index].Length;
21        }
22        TotalBytes = total;
23        UsableBytes = usable;
24    }
25
26    /// <summary>Gets the number of immutable descriptors.</summary>
27    /// <nova.when>Use to bound indexed diagnostics and allocator scans.</nova.when>
28    public int Count => _descriptors.Length;
29    /// <summary>Gets the normalisation implementation that produced the map.</summary>
30    /// <nova.when>Use when reporting or auditing the selected overlap policy.</nova.when>
31    public MemoryMapNormalisationMethod Method { get; }
32    /// <summary>Gets the total represented byte count.</summary>
33    /// <nova.when>Use for boot diagnostics and physical-memory accounting.</nova.when>
34    public ulong TotalBytes { get; }
35    /// <summary>Gets the total immediately allocatable byte count.</summary>
36    /// <nova.when>Use to estimate memory available to the first physical allocator.</nova.when>
37    public ulong UsableBytes { get; }
38
39    /// <summary>Attempts to read one descriptor by index.</summary>
40    /// <nova.when>Use for allocation scans and deterministic diagnostics.</nova.when>
41    /// <nova.depends>Count</nova.depends>
42    /// <returns><see langword="true"/> when the index exists.</returns>
43    /// <example><code>bool found = map.TryGetDescriptor(0, out MemoryDescriptor descriptor);</code></example>
44    public bool TryGetDescriptor(int index, out MemoryDescriptor descriptor)
45    {
46        descriptor = default;
47        if ((uint)index >= (uint)_descriptors.Length) return false;
48        descriptor = _descriptors[index];
49        return true;
50    }
51
52    /// <summary>Creates a forward-only diagnostic cursor without exposing the backing array.</summary>
53    /// <nova.when>Use when printing or auditing every descriptor.</nova.when>
54    /// <nova.depends>MemoryMapDiagnosticCursor</nova.depends>
55    /// <returns>A cursor positioned before the first descriptor.</returns>
56    /// <example><code>MemoryMapDiagnosticCursor cursor = map.CreateDiagnosticCursor();</code></example>
57    public MemoryMapDiagnosticCursor CreateDiagnosticCursor() => new(_descriptors);
58}

60/// <summary>Provides immutable forward enumeration without implementing mutable collection interfaces.</summary>
61/// <nova.when>Use for boot diagnostics where exposing an array would permit mutation.</nova.when>
62/// <nova.depends>NormalisedMemoryMap</nova.depends>
63public struct MemoryMapDiagnosticCursor
64{
65    private readonly MemoryDescriptor[] _descriptors;
66    private int _index;
67
68    internal MemoryMapDiagnosticCursor(MemoryDescriptor[] descriptors)
69    {
70        _descriptors = descriptors;
71        _index = -1;
72        Current = default;
73    }
74
75    /// <summary>Gets the current descriptor after a successful move.</summary>
76    /// <nova.when>Read only after MoveNext returns true.</nova.when>
77    public MemoryDescriptor Current { get; private set; }
78
79    /// <summary>Advances to the next diagnostic descriptor.</summary>
80    /// <nova.when>Call until it returns <see langword="false"/>.</nova.when>
81    /// <nova.depends>CreateDiagnosticCursor</nova.depends>
82    /// <returns><see langword="true"/> when another descriptor is available.</returns>
83    /// <example><code>while (cursor.MoveNext()) { MemoryDescriptor descriptor = cursor.Current; }</code></example>
84    public bool MoveNext()
85    {
86        int next = _index + 1;
87        if ((uint)next >= (uint)_descriptors.Length) return false;
88        _index = next;
89        Current = _descriptors[next];
90        return true;
91    }
92}

147/// <summary>Provides caller-owned scratch storage so boot normalisation has bounded memory use.</summary>
148/// <nova.when>Allocate before obtaining the final firmware map, then reuse for all implementations.</nova.when>
149/// <nova.depends>Maximum expected descriptor and reservation counts</nova.depends>
150public sealed class MemoryMapNormalisationWorkspace
151{
152    internal readonly MemoryDescriptor[] Inputs;
153    internal readonly ulong[] Boundaries;
154    internal readonly MemoryDescriptor[] Outputs;
155
156    /// <summary>Creates bounded scratch storage.</summary>
157    /// <nova.when>Create before final-map capture so normalisation requires no unbounded allocation.</nova.when>
158    /// <nova.depends>Validated capacity limits</nova.depends>
159    /// <returns>A reusable bounded normalisation workspace.</returns>
160    /// <example><code>MemoryMapNormalisationWorkspace workspace = new(256, 512);</code></example>
161    /// <param name="descriptorCapacity">Maximum source plus reservation descriptors.</param>
162    /// <param name="outputCapacity">Maximum split output descriptors.</param>
163    public MemoryMapNormalisationWorkspace(int descriptorCapacity, int outputCapacity)
164    {
165        if (descriptorCapacity < 1) throw new ArgumentOutOfRangeException(nameof(descriptorCapacity));
166        if (outputCapacity < 1) throw new ArgumentOutOfRangeException(nameof(outputCapacity));
167        Inputs = new MemoryDescriptor[descriptorCapacity];
168        Boundaries = new ulong[checked(descriptorCapacity * 2)];
169        Outputs = new MemoryDescriptor[outputCapacity];
170    }
171
172    /// <summary>Gets the maximum source plus reservation count.</summary>
173    /// <nova.when>Use to verify workspace headroom before normalisation.</nova.when>
174    public int DescriptorCapacity => Inputs.Length;
175    /// <summary>Gets the maximum normalised output count.</summary>
176    /// <nova.when>Use to verify split-range headroom before normalisation.</nova.when>
177    public int OutputCapacity => Outputs.Length;
178}
