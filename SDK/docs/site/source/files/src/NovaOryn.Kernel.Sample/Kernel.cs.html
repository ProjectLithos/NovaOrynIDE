16    [KernelEntry]
17    public static bool KMain(BootContext boot)
18    {
19        SerialConsole serial = new();
20        FramebufferConsole framebuffer = new();
21        if (!serial.Configure(SerialConfiguration.Com1())) return false;
22        if (!framebuffer.Configure(FramebufferConfiguration.Default(ConsoleFontSize))) return false;
23        if (!serial.Initialize(boot)) return false;
24        if (!framebuffer.Initialize(boot)) return false;
25        if (!WriteLine(serial, framebuffer, "NovaOryn KMain started.")) return false;
26        if (!NativeUefiMemoryMapSource.TryCreate(boot, out NativeUefiMemoryMapSource? finalMap)) return false;
27        if (finalMap is null || finalMap.Count < 1) return false;
28        if (!WriteLine(serial, framebuffer, "Final UEFI memory map retained; ExitBootServices succeeded.")) return false;
29        if (!WriteLine(serial, framebuffer, "Native UEFI memory-map adapter online.")) return false;
30        KernelDiagnosticSink diagnostics = new(serial, framebuffer);
31        if (!PlatformInitialization.Initialize(diagnostics)) return false;
32        if (!WriteLine(serial, framebuffer, "GDT, TSS, IDT, exceptions, and interrupt-controller contracts online.")) return false;
33        if (!AddressSpaceExamples.ValidateStandardLayout()) return false;
34        if (!VirtualMemoryExamples.ValidateContracts()) return false;
35        if (!WriteLine(serial, framebuffer, "Virtual-memory contracts and x64 page-table codec online.")) return false;
36        if (!HeapExamples.ValidateContracts()) return false;
37        if (!WriteLine(serial, framebuffer, "Early-allocation and first-fit heap methodologies online.")) return false;
38        if (!WriteLine(serial, framebuffer, "CPU halted.")) return false;
39        return CPU.Halt();
40    }

42    [UnmanagedCallersOnly(EntryPoint = "NovaOrynManagedEntry")]
43    public static unsafe byte NativeEntry(nint bootContextAddress)
44    {
45        if (bootContextAddress == 0) return 0;
46        NativeBootContext* native = (NativeBootContext*)bootContextAddress;
47        if (native->Signature != 0x4E59524F41564F4EUL) return 0;
48
49        Framebuffer framebuffer = new(
50            new PhysicalAddress(native->FramebufferAddress),
51            native->FramebufferSize,
52            native->Width,
53            native->Height,
54            native->PixelsPerScanLine,
55            (FramebufferPixelFormat)native->PixelFormat,
56            new PixelBitMask(native->RedMask, native->GreenMask, native->BlueMask, native->ReservedMask));
57        BootContext boot = new(
58            BootProtocol.Uefi,
59            framebuffer,
60            new PhysicalAddress(native->FinalMemoryMapAddress),
61            native->FinalMemoryMapLength,
62            native->FinalMemoryMapKey,
63            native->FinalMemoryDescriptorSize,
64            native->FinalMemoryDescriptorVersion,
65            native->FinalMemoryMapFlag == 1 && native->ExitBootServicesStatus == 0,
66            new PhysicalAddress(native->AcpiRootPointerAddress));
67        return KMain(boot) ? (byte)1 : (byte)0;
68    }
