11/// <summary>Initialises and demonstrates the public x64 descriptor, interrupt, and controller facilities.</summary>
12internal static unsafe class PlatformInitialization
13{
14    private const int EmergencyStackSize = 4096;
15
16    internal static bool Initialize(IExceptionDiagnosticSink diagnostics)
17    {
18        byte* gdtStorage = stackalloc byte[64];
19        byte* tssStorage = stackalloc byte[104];
20        byte* idtStorage = stackalloc byte[4096];
21        byte* ringZeroStack = stackalloc byte[EmergencyStackSize];
22        byte* doubleFaultStack = stackalloc byte[EmergencyStackSize];
23        byte* nmiStack = stackalloc byte[EmergencyStackSize];
24        byte* machineCheckStack = stackalloc byte[EmergencyStackSize];
25
26        ProcessorId processor = new(0);
27        SegmentSelector kernelCode = SegmentSelector.Create(1, DescriptorPrivilegeLevel.Kernel);
28        SegmentSelector kernelData = SegmentSelector.Create(2, DescriptorPrivilegeLevel.Kernel);
29        SegmentSelector userData = SegmentSelector.Create(3, DescriptorPrivilegeLevel.User);
30        SegmentSelector userCode = SegmentSelector.Create(4, DescriptorPrivilegeLevel.User);
31        SegmentSelector taskState = SegmentSelector.Create(5, DescriptorPrivilegeLevel.Kernel);
32
33        X64TaskStateSegment tss = new();
34        TaskStateSegmentConfiguration tssConfiguration = new(
35            processor,
36            ToAddress(tssStorage),
37            104,
38            ToAddress(ringZeroStack + EmergencyStackSize),
39            ToAddress(doubleFaultStack + EmergencyStackSize),
40            ToAddress(nmiStack + EmergencyStackSize),
41            ToAddress(machineCheckStack + EmergencyStackSize),
42            IoPermissionBitmapPolicy.Disabled,
43            default,
44            0);
45        if (!tss.Configure(tssConfiguration)) return false;
46
47        X64GlobalDescriptorTable gdt = new();
48        GlobalDescriptorTableConfiguration gdtConfiguration = new(
49            processor,
50            ToAddress(gdtStorage),
51            64,
52            kernelCode,
53            kernelData,
54            userCode,
55            userData,
56            taskState);
57        if (!gdt.Configure(gdtConfiguration, tss)) return false;
58        if (!gdt.Install()) return false;
59        if (!tss.Install(taskState)) return false;
60
61        X64InterruptDescriptorTable idt = new();
62        InterruptDescriptorTableConfiguration idtConfiguration = new(
63            processor,
64            ToAddress(idtStorage),
65            4096,
66            kernelCode,
67            1,
68            2,
69            3);
70        if (!idt.Configure(idtConfiguration)) return false;
71        EssentialExceptionHandlers exceptions = new(idt, diagnostics);
72        if (!exceptions.RegisterAll()) return false;
73        if (!idt.Install()) return false;
74
75        return DemonstrateInterruptControllerContracts();
76    }
77
78    private static bool DemonstrateInterruptControllerContracts()
79    {
80        X64InterruptVectorAllocator vectors = new();
81        X64InterruptController controller = new(vectors);
82        InterruptControllerCapabilities capabilities = controller.GetCapabilities();
83        if (!capabilities.Msi || !capabilities.MsiX) return false;
84
85        byte vector = controller.AllocateVector();
86        if (vector == 0) return false;
87        InterruptRouteConfiguration configuration = new(
88            new InterruptSource(0),
89            vector,
90            InterruptPolarity.Conforms,
91            InterruptTriggerMode.Edge,
92            new InterruptAffinity(new ProcessorId(0)),
93            8,
94            true,
95            InterruptDeliveryMechanism.Msi);
96        InterruptRouteResult route = controller.Route(configuration);
97        if (!route.Succeeded) return false;
98        if (controller.CreateMessage(route.Handle).Vector != vector) return false;
99        if (!controller.SetAffinity(route.Handle, new InterruptAffinity(new ProcessorId(0)))) return false;
100        if (!controller.SetPriority(route.Handle, 8)) return false;
101        if (!controller.Unmask(route.Handle)) return false;
102        if (!controller.Mask(route.Handle)) return false;
103        if (!controller.RemoveRoute(route.Handle)) return false;
104        return controller.ReleaseVector(vector);
105    }
106
107    private static Address ToAddress(byte* pointer) => new((ulong)(nuint)pointer);
108}
