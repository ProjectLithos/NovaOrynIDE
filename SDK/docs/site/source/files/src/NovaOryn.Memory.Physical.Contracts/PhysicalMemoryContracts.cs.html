78/// <summary>Describes one exact page-aligned physical-frame range.</summary>
79/// <nova.when>Use for reservations and for reporting allocator-owned physical ranges.</nova.when>
80/// <nova.depends>NovaOryn.Primitives.PhysicalAddress</nova.depends>
81[SupportedArchitecture(SupportedArchitecture.All)]
82[BootStage(BootStage.ManagedBootstrap)]
83public readonly struct PhysicalFrameRange
84{
85    private PhysicalFrameRange(PhysicalAddress start, ulong pageCount)
86    {
87        Start = start;
88        PageCount = pageCount;
89    }
90
91    /// <summary>Gets the first physical byte of the range.</summary>
92    /// <nova.when>Use as the physical address of the first frame.</nova.when>
93    public PhysicalAddress Start { get; }
94    /// <summary>Gets the number of 4 KiB frames in the range.</summary>
95    /// <nova.when>Use for frame accounting and page-table mapping.</nova.when>
96    public ulong PageCount { get; }
97    /// <summary>Gets the range length in bytes.</summary>
98    /// <nova.when>Use for diagnostics or byte-address calculations.</nova.when>
99    public ulong Length => PageCount * 4096UL;
100    /// <summary>Gets the exclusive physical end address.</summary>
101    /// <nova.when>Use for non-overlap and upper-bound checks.</nova.when>
102    public PhysicalAddress EndExclusive
103    {
104        get { return new PhysicalAddress(Start.Value + Length); }
105    }
106
107    /// <summary>Attempts to create a validated 4 KiB page-aligned physical range.</summary>
108    /// <nova.when>Use before reserving a fixed physical range.</nova.when>
109    /// <nova.depends>4 KiB physical frame size</nova.depends>
110    /// <returns><see langword="true"/> when the range is non-empty, aligned, and cannot overflow.</returns>
111    /// <example><code>bool valid = PhysicalFrameRange.TryCreate(new PhysicalAddress(0x100000), 16, out PhysicalFrameRange range);</code></example>
112    public static bool TryCreate(PhysicalAddress start, ulong pageCount, out PhysicalFrameRange range)
113    {
114        range = default;
115        if (pageCount == 0 || (start.Value & 0xFFFUL) != 0) return false;
116        if (pageCount > ulong.MaxValue / 4096UL) return false;
117        ulong length = pageCount * 4096UL;
118        if (start.Value > ulong.MaxValue - length) return false;
119        range = new PhysicalFrameRange(start, pageCount);
120        return true;
121    }
122}

124/// <summary>Describes a contiguous allocation request and its physical-address constraints.</summary>
125/// <nova.when>Use for normal, aligned, below-limit, or DMA-oriented frame allocation.</nova.when>
126/// <nova.depends>PhysicalMemoryPurpose</nova.depends>
127[SupportedArchitecture(SupportedArchitecture.All)]
128[BootStage(BootStage.ManagedBootstrap)]
129public readonly struct PhysicalAllocationRequest
130{
131    private PhysicalAllocationRequest(ulong pageCount, ulong alignmentPages, PhysicalAddress minimumAddress, PhysicalAddress maximumAddressExclusive, PhysicalMemoryPurpose purpose)
132    {
133        PageCount = pageCount;
134        AlignmentPages = alignmentPages;
135        MinimumAddress = minimumAddress;
136        MaximumAddressExclusive = maximumAddressExclusive;
137        Purpose = purpose;
138    }
139
140    /// <summary>Gets the requested number of contiguous 4 KiB pages.</summary>
141    /// <nova.when>Use to compare requested and allocator-reserved page counts.</nova.when>
142    public ulong PageCount { get; }
143    /// <summary>Gets the required power-of-two alignment in pages.</summary>
144    /// <nova.when>Use for page-table, huge-page, and device-alignment constraints.</nova.when>
145    public ulong AlignmentPages { get; }
146    /// <summary>Gets the inclusive minimum physical address.</summary>
147    /// <nova.when>Use to constrain allocations away from low physical memory.</nova.when>
148    public PhysicalAddress MinimumAddress { get; }
149    /// <summary>Gets the exclusive maximum physical address, or zero for no upper limit.</summary>
150    /// <nova.when>Use for DMA32-style or device-specific physical-address ceilings.</nova.when>
151    public PhysicalAddress MaximumAddressExclusive { get; }
152    /// <summary>Gets the intended allocation ownership.</summary>
153    /// <nova.when>Use for diagnostics and later mapping policy.</nova.when>
154    public PhysicalMemoryPurpose Purpose { get; }
155
156    /// <summary>Attempts to create a validated contiguous frame-allocation request.</summary>
157    /// <nova.when>Use before calling <see cref="IPhysicalMemoryManager.TryAllocate"/>.</nova.when>
158    /// <nova.depends>4 KiB physical frame size and power-of-two alignment</nova.depends>
159    /// <returns><see langword="true"/> when the page count, alignment, and address interval are valid.</returns>
160    /// <example><code>bool valid = PhysicalAllocationRequest.TryCreate(8, 8, default, new PhysicalAddress(0x100000000), PhysicalMemoryPurpose.Dma, out PhysicalAllocationRequest request);</code></example>
161    public static bool TryCreate(ulong pageCount, ulong alignmentPages, PhysicalAddress minimumAddress, PhysicalAddress maximumAddressExclusive, PhysicalMemoryPurpose purpose, out PhysicalAllocationRequest request)
162    {
163        request = default;
164        if (pageCount == 0 || pageCount > ulong.MaxValue / 4096UL) return false;
165        if (alignmentPages == 0 || alignmentPages > ulong.MaxValue / 4096UL || (alignmentPages & (alignmentPages - 1)) != 0) return false;
166        if ((minimumAddress.Value & 0xFFFUL) != 0 || (maximumAddressExclusive.Value & 0xFFFUL) != 0) return false;
167        if (maximumAddressExclusive.Value != 0 && minimumAddress.Value >= maximumAddressExclusive.Value) return false;
168        request = new PhysicalAllocationRequest(pageCount, alignmentPages, minimumAddress, maximumAddressExclusive, purpose);
169        return true;
170    }
171}

310/// <summary>Defines the replaceable contract for physical-frame allocation.</summary>
311/// <nova.when>Implement for a custom allocator or consume through a selected NovaOryn allocator methodology.</nova.when>

313[SupportedArchitecture(SupportedArchitecture.All)]
314[BootStage(BootStage.ManagedBootstrap)]
315public interface IPhysicalMemoryManager
316{

318    /// <nova.when>Use for diagnostics and configuration reporting.</nova.when>
319    PhysicalAllocatorMethod Method { get; }
320    /// <summary>Gets whether the manager has accepted a normalised map and workspace.</summary>
321    /// <nova.when>Check before allocation when manager ownership is indirect.</nova.when>
322    bool IsInitialized { get; }
323    /// <summary>Gets the physical frame size in bytes.</summary>
324    /// <nova.when>Use when converting frame counts to byte lengths.</nova.when>
325    ulong PageSize { get; }
326
327    /// <summary>Initialises the manager from immediately allocatable ranges in a normalised boot map.</summary>
328    /// <nova.when>Call once after boot memory-map normalisation and before virtual-memory construction.</nova.when>

346    bool TryRelease(PhysicalAllocation allocation, out PhysicalMemoryStatus status);
347
348    /// <summary>Removes an exact currently-free range from ordinary allocation.</summary>

350    /// <nova.depends>PhysicalFrameRange</nova.depends>
351    /// <returns><see langword="true"/> when every requested frame was free and the reservation was recorded.</returns>

353    bool TryReserve(PhysicalFrameRange range, PhysicalMemoryPurpose purpose, out PhysicalReservation reservation, out PhysicalMemoryStatus status);
354
355    /// <summary>Releases a live reservation back to the same manager.</summary>
356    /// <nova.when>Use only when the reserving subsystem has made the exact range safe for general reuse.</nova.when>
357    /// <nova.depends>PhysicalReservation token returned by TryReserve</nova.depends>
358    /// <returns><see langword="true"/> when the live reservation was found and released exactly once.</returns>
359    /// <example><code>bool released = manager.TryReleaseReservation(reservation, out PhysicalMemoryStatus status);</code></example>
360    bool TryReleaseReservation(PhysicalReservation reservation, out PhysicalMemoryStatus status);
361
362    /// <summary>Returns a current allocator accounting snapshot.</summary>

365    /// <returns>Current total, free, allocated, reserved, and largest-run counters.</returns>

367    PhysicalMemoryStatistics GetStatistics();
