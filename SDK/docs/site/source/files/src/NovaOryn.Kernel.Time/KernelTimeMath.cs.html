9    public static UInt64 HpetTicksToNanoseconds(UInt64 ticks, UInt64 periodFemtoseconds)
10    {
11        if (periodFemtoseconds == 0UL) return 0UL;
12        UInt64 whole = ticks / 1000000UL;
13        UInt64 remainder = ticks % 1000000UL;
14        if (whole != 0UL && periodFemtoseconds > 0xFFFFFFFFFFFFFFFFUL / whole) return 0xFFFFFFFFFFFFFFFFUL;
15        UInt64 high = whole * periodFemtoseconds;
16        UInt64 low = (remainder * periodFemtoseconds) / 1000000UL;
17        return 0xFFFFFFFFFFFFFFFFUL - high < low ? 0xFFFFFFFFFFFFFFFFUL : high + low;
18    }

44    }
