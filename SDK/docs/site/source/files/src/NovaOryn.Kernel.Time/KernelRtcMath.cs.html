5/// <summary>Provides deterministic conversion and validation helpers for RTC/CMOS calendar values.</summary>
6public static class KernelRtcMath

8    /// <summary>Converts one packed BCD byte to binary.</summary>
9    public static Byte BcdToBinary(Byte value) => (Byte)(((value >> 4) * 10U) + (value & 0x0FU));

11    /// <summary>Converts a CMOS hour value to a 24-hour binary value.</summary>
12    public static Byte DecodeHour(Byte value, Boolean binaryMode, Boolean twentyFourHourMode)
13    {
14        Boolean pm = (value & 0x80U) != 0U;
15        Byte hour = (Byte)(value & 0x7FU);
16        if (!binaryMode) hour = BcdToBinary(hour);
17        if (twentyFourHourMode) return hour;
18        hour = (Byte)(hour % 12U);
19        return pm ? (Byte)(hour + 12U) : hour;
20    }

22    /// <summary>Combines CMOS year and optional century values into a Gregorian year.</summary>
23    public static UInt16 DecodeYear(Byte year, Byte century, Boolean hasCentury, Boolean binaryMode)
24    {
25        UInt16 y = binaryMode ? year : BcdToBinary(year);
26        if (hasCentury)
27        {
28            UInt16 c = binaryMode ? century : BcdToBinary(century);
29            return (UInt16)(c * 100U + y);
30        }
31        return (UInt16)(y >= 70U ? 1900U + y : 2000U + y);
32    }

34    /// <summary>Gets whether a Gregorian calendar date and time has valid field ranges.</summary>
35    public static Boolean IsValid(UInt16 year, Byte month, Byte day, Byte hour, Byte minute, Byte second)
36    {
37        if (year < 1601U || year > 9999U || month < 1U || month > 12U || day < 1U || hour > 23U || minute > 59U || second > 59U) return false;
38        return day <= DaysInMonth(year, month);
39    }

41    /// <summary>Gets the number of days in a Gregorian calendar month.</summary>
42    public static Byte DaysInMonth(UInt16 year, Byte month)
43    {
44        if (month == 2U) return IsLeapYear(year) ? (Byte)29U : (Byte)28U;
45        return month == 4U || month == 6U || month == 9U || month == 11U ? (Byte)30U : (Byte)31U;
46    }

48    /// <summary>Gets whether a Gregorian year is a leap year.</summary>
49    public static Boolean IsLeapYear(UInt16 year) => (year % 4U) == 0U && ((year % 100U) != 0U || (year % 400U) == 0U);
50}
