45/// <summary>Provides stable, format-aware reads from the PC-compatible RTC/CMOS calendar.</summary>
46public static class KernelRtcCmos
47{
48    private const UInt16 IndexPort = 0x70;
49    private const UInt16 DataPort = 0x71;
50    private const Byte StatusA = 0x0A;
51    private const Byte StatusB = 0x0B;
52    private const Byte UpdateInProgress = 0x80;
53    private const UInt32 WaitLimit = 1000000U;
54    private static Boolean _initialized, _binaryMode, _twentyFourHourMode;
55    private static Byte _centuryRegister;
56
57    /// <summary>Initializes and probes the RTC/CMOS calendar facility.</summary>
58    public static Boolean Initialize()
59    {
60        _initialized = false;
61        _centuryRegister = 0;
62        if (KernelAcpiFadt.IsInitialized()) _centuryRegister = KernelAcpiFadt.GetInfo().CenturyRegister;
63        if (!WaitForStableWindow() || !TryReadRegister(StatusB, out Byte status)) return false;
64        _binaryMode = (status & 0x04) != 0;
65        _twentyFourHourMode = (status & 0x02) != 0;
66        _initialized = true;
67        if (!TryRead(out _)) { _initialized = false; return false; }
68        return true;
69    }
70
71    /// <summary>Gets whether RTC/CMOS initialization succeeded.</summary>
72    public static Boolean IsInitialized() => _initialized;
73
74    /// <summary>Gets an RTC/CMOS capability snapshot.</summary>
75    public static KernelRtcCapabilities GetCapabilities() => new(_initialized, _centuryRegister != 0, _binaryMode, _twentyFourHourMode);
76
77    /// <summary>Reads a stable Gregorian calendar sample from RTC/CMOS.</summary>
78    public static Boolean TryRead(out KernelRtcDateTime value)
79    {
80        value = default;
81        if (!_initialized) return false;
82        for (Byte attempt = 0; attempt < 8; attempt++)
83        {
84            if (!WaitForStableWindow() || !TryReadRaw(out RtcRaw first)) return false;
85            if (!WaitForStableWindow() || !TryReadRaw(out RtcRaw second)) return false;
86            if (!first.Equals(second)) continue;
87            if (!TryReadRegister(StatusB, out Byte status)) return false;
88            Boolean binary = (status & 0x04) != 0;
89            Boolean hour24 = (status & 0x02) != 0;
90            Byte secondValue = binary ? second.Second : KernelRtcMath.BcdToBinary(second.Second);
91            Byte minuteValue = binary ? second.Minute : KernelRtcMath.BcdToBinary(second.Minute);
92            Byte hourValue = KernelRtcMath.DecodeHour(second.Hour, binary, hour24);
93            Byte dayValue = binary ? second.Day : KernelRtcMath.BcdToBinary(second.Day);
94            Byte monthValue = binary ? second.Month : KernelRtcMath.BcdToBinary(second.Month);
95            UInt16 yearValue = KernelRtcMath.DecodeYear(second.Year, second.Century, _centuryRegister != 0, binary);
96            Byte weekdayValue = binary ? second.DayOfWeek : KernelRtcMath.BcdToBinary(second.DayOfWeek);
97            if (!KernelRtcMath.IsValid(yearValue, monthValue, dayValue, hourValue, minuteValue, secondValue)) return false;
98            value = new KernelRtcDateTime(yearValue, monthValue, dayValue, hourValue, minuteValue, secondValue, weekdayValue);
99            return true;
100        }
101        return false;
102    }
103
104    private static Boolean TryReadRaw(out RtcRaw value)
105    {
106        value = default;
107        if (!TryReadRegister(0x00, out Byte second) || !TryReadRegister(0x02, out Byte minute) || !TryReadRegister(0x04, out Byte hour) ||
108            !TryReadRegister(0x06, out Byte dayOfWeek) || !TryReadRegister(0x07, out Byte day) || !TryReadRegister(0x08, out Byte month) || !TryReadRegister(0x09, out Byte year)) return false;
109        Byte century = 0;
110        if (_centuryRegister != 0 && !TryReadRegister(_centuryRegister, out century)) return false;
111        value = new RtcRaw(second, minute, hour, dayOfWeek, day, month, year, century);
112        return true;
113    }
114
115    private static Boolean WaitForStableWindow()
116    {
117        for (UInt32 i = 0U; i < WaitLimit; i++)
118        {
119            if (!TryReadRegister(StatusA, out Byte status)) return false;
120            if ((status & UpdateInProgress) == 0) return true;
121            Native.Pause();
122        }
123        return false;
124    }
125
126    private static Boolean TryReadRegister(Byte register, out Byte value)
127    {
128        value = 0;
129        return Native.WritePort8(IndexPort, (Byte)(register & 0x7F)) && Native.ReadPort8(DataPort, out value);
130    }
131
132    private readonly struct RtcRaw
133    {
134        internal RtcRaw(Byte second, Byte minute, Byte hour, Byte dayOfWeek, Byte day, Byte month, Byte year, Byte century)
135        { Second = second; Minute = minute; Hour = hour; DayOfWeek = dayOfWeek; Day = day; Month = month; Year = year; Century = century; }
136        internal Byte Second { get; }
137        internal Byte Minute { get; }
138        internal Byte Hour { get; }
139        internal Byte DayOfWeek { get; }
140        internal Byte Day { get; }
141        internal Byte Month { get; }
142        internal Byte Year { get; }
143        internal Byte Century { get; }
144        internal Boolean Equals(RtcRaw other) => Second == other.Second && Minute == other.Minute && Hour == other.Hour && DayOfWeek == other.DayOfWeek && Day == other.Day && Month == other.Month && Year == other.Year && Century == other.Century;
145    }
