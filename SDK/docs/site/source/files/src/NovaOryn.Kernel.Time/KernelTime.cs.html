18    /// <summary>Initializes HPET-backed monotonic time, invariant-TSC calibration, and Local APIC timer calibration.</summary>
19    public static Boolean Initialize()
20    {
21        _initialized = false; _source = KernelClockSource.None; _hpetAvailable = false; _tscAvailable = Native.SupportsTsc(); _invariantTsc = _tscAvailable && Native.SupportsInvariantTsc(); _localApicTimer = false;
22        if (!KernelAcpi.IsInitialized() || !KernelAcpi.TryGetHpet(out AcpiHpetInfo hpet) || hpet.AddressSpace != (Byte)0U) return false;
23        _hpetBase = hpet.BaseAddress;
24        UInt64 capabilities = Native.ReadMmio64(_hpetBase);
25        _hpetPeriodFs = capabilities >> 32;
26        _hpet64 = (capabilities & (1UL << 13)) != 0UL;
27        if (_hpetPeriodFs == 0UL || _hpetPeriodFs > 100000000UL) return false;
28        UInt64 config = Native.ReadMmio64(_hpetBase + HpetConfigOffset);
29        if (!Native.WriteMmio64(_hpetBase + HpetConfigOffset, config | 1UL)) return false;
30        _hpetStart = ReadHpetCounter();
31        _hpetAvailable = true;
32        _source = KernelClockSource.Hpet;
33        if (_invariantTsc)
34        {
35            UInt64 startHpet = ReadHpetCounter(); UInt64 startTsc = Native.ReadTimestampCounter();
36            if (WaitHpetNanoseconds(CalibrationNanoseconds))
37            {
38                UInt64 elapsedNs = HpetElapsedNanoseconds(startHpet, ReadHpetCounter());
39                UInt64 elapsedTsc = Native.ReadTimestampCounter() - startTsc;
40                _tscFrequency = KernelTimeMath.CalibrateFrequency(elapsedTsc, elapsedNs);
41                if (_tscFrequency >= 1000000UL) { _tscStart = Native.ReadTimestampCounter(); _source = KernelClockSource.InvariantTsc; }
42            }
43        }
44        if (KernelAcpi.TryGetLocalApicAddress(out _localApicBase)) _localApicTimer = CalibrateLocalApicTimer();
45        KernelRtcCmos.Initialize();
46        _initialized = true;
47        return true;
48    }

107    private static Boolean ProgramLocalApicTimer(Byte vector, UInt64 nanoseconds, KernelTimerMode mode)
108    {
109        if (!_localApicTimer || vector < (Byte)32U || nanoseconds == 0UL) return false;
110        UInt64 ticks = KernelTimeMath.NanosecondsToTicksCeiling(nanoseconds, _localApicFrequency);
111        if (ticks == 0UL || ticks > 0xFFFFFFFFU) return false;
112        UInt32 lvt = vector; if (mode == KernelTimerMode.Periodic) lvt |= LocalApicPeriodic;
113        return Native.WriteMmio32(_localApicBase + LocalApicDivide, DivideBy16) && Native.WriteMmio32(_localApicBase + LocalApicLvtTimer, lvt) && Native.WriteMmio32(_localApicBase + LocalApicInitialCount, (UInt32)ticks);
114    }

116    private static Boolean CalibrateLocalApicTimer()
117    {
118        if (_localApicBase == 0UL) return false;
119        UInt32 spurious = Native.ReadMmio32(_localApicBase + LocalApicSpurious);
120        if ((spurious & 0xFFU) < 0x10U) spurious = (spurious & 0xFFFFFF00U) | 0xFFU;
121        if (!Native.WriteMmio32(_localApicBase + LocalApicSpurious, spurious | 0x100U)) return false;
122        if (!Native.WriteMmio32(_localApicBase + LocalApicDivide, DivideBy16)) return false;
123        if (!Native.WriteMmio32(_localApicBase + LocalApicLvtTimer, LocalApicMasked)) return false;
124        if (!Native.WriteMmio32(_localApicBase + LocalApicInitialCount, 0xFFFFFFFFU)) return false;
125        UInt64 startHpet = ReadHpetCounter();
126        if (!WaitHpetNanoseconds(CalibrationNanoseconds)) return false;
127        UInt32 current = Native.ReadMmio32(_localApicBase + LocalApicCurrentCount);
128        if (!Native.WriteMmio32(_localApicBase + LocalApicInitialCount, 0U)) return false;
129        UInt64 elapsedNs = HpetElapsedNanoseconds(startHpet, ReadHpetCounter());
130        UInt64 elapsedTicks = (UInt64)(0xFFFFFFFFU - current);
131        _localApicFrequency = KernelTimeMath.CalibrateFrequency(elapsedTicks, elapsedNs);
132        return _localApicFrequency != 0UL;
133    }

135    private static Boolean WaitHpetNanoseconds(UInt64 nanoseconds)
136    {
137        UInt64 requiredTicks = HpetTicksForNanoseconds(nanoseconds); if (requiredTicks == 0UL) return false;
138        UInt64 start = ReadHpetCounter(); UInt64 maximumIterations = 100000000UL;
139        for (UInt64 iteration = 0UL; iteration < maximumIterations; iteration++) if (HpetDelta(start, ReadHpetCounter()) >= requiredTicks) return true;
140        return false;
141    }

143    private static UInt64 ReadHpetCounter()
144    {
145        UInt64 value = Native.ReadMmio64(_hpetBase + HpetCounterOffset);
146        return _hpet64 ? value : value & 0xFFFFFFFFUL;
147    }
148    private static UInt64 HpetDelta(UInt64 start, UInt64 end) => _hpet64 ? end - start : (UInt32)end - (UInt32)start;
149    private static UInt64 HpetElapsedNanoseconds(UInt64 start, UInt64 end) => KernelTimeMath.HpetTicksToNanoseconds(HpetDelta(start, end), _hpetPeriodFs);
150    private static UInt64 HpetFrequencyHz() => _hpetPeriodFs == 0UL ? 0UL : 1000000000000000UL / _hpetPeriodFs;
151    private static UInt64 HpetTicksForNanoseconds(UInt64 nanoseconds)
152    {
153        if (_hpetPeriodFs == 0UL) return 0UL;
154        UInt64 whole = nanoseconds / 1000000UL; UInt64 remainder = nanoseconds % 1000000UL;
155        if (whole > 0xFFFFFFFFFFFFFFFFUL / 1000000000000UL) return 0xFFFFFFFFFFFFFFFFUL;
156        UInt64 femtoseconds = whole * 1000000000000UL + remainder * 1000000UL;
157        UInt64 ticks = femtoseconds / _hpetPeriodFs; if ((femtoseconds % _hpetPeriodFs) != 0UL) ticks++;
158        return ticks;
159    }
