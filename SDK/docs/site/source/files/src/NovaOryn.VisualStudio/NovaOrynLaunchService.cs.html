16    public bool TryGetActiveProject(out string projectFile)
17    {
18        ThreadHelper.ThrowIfNotOnUIThread(); projectFile = string.Empty;
19        if (!(ServiceProvider.GlobalProvider.GetService(typeof(DTE)) is DTE dte) || dte.Solution == null || !dte.Solution.IsOpen) return false;
20        Project project = ResolveStartupProject(dte) ?? (dte.ActiveSolutionProjects as Array)?.Cast<object>().OfType<Project>().FirstOrDefault();
21        string candidate = project?.FullName;
22        if (!NovaOrynProjectRecognizer.IsNovaOrynProject(candidate)) return false;
23        projectFile = Path.GetFullPath(candidate); return true;
24    }
25    public bool QueueLaunch(bool buildOnly)
26    {
27        ThreadHelper.ThrowIfNotOnUIThread();
28        if (Interlocked.CompareExchange(ref _launchActive, 1, 0) != 0)
29        {
30            _package.Output.WriteLine("[INFO] A NovaOryn build/run is already active; duplicate launch ignored.");
31            return false;
32        }
33        _package.JoinableTaskFactory.RunAsync(async delegate
34        {
35            try { await LaunchAsync(buildOnly, CancellationToken.None); }
36            finally { Interlocked.Exchange(ref _launchActive, 0); }
37        }).FileAndForget("NovaOryn/Launch");
38        return true;
39    }
40    private async Task LaunchAsync(bool buildOnly, CancellationToken token)
41    {
42        await _package.JoinableTaskFactory.SwitchToMainThreadAsync(token); _package.Output.Activate();
43        if (!TryGetActiveProject(out string projectFile)) { _package.Output.WriteLine("[FAIL] The startup project is not a NovaOryn kernel project."); return; }
44        if (!SdkInstallationDetector.IsInstalled(out string sdkRoot)) { _package.Output.WriteLine($"[FAIL] NovaOryn SDK was not found at {sdkRoot}."); return; }
45        DTE dte = await _package.GetServiceAsync(typeof(DTE)) as DTE;
46        if (dte != null) dte.ExecuteCommand("File.SaveAll");
47        string manifest = Path.Combine(Path.GetDirectoryName(projectFile) ?? string.Empty, "NovaOrynProject.json");
48        string script = Path.Combine(sdkRoot, "Build-NovaOryn.ps1");
49        string configuration = ResolveConfiguration(dte);
50        string arguments = $"-NoProfile -ExecutionPolicy Bypass -File {Quote(script)} -Project {Quote(manifest)} -Configuration {configuration}" + (buildOnly ? " -NoRun" : " -Run");
51        _package.Output.WriteLine($"[INFO] NovaOryn {(buildOnly ? "Build" : "Run")}: {Path.GetFileName(projectFile)} ({configuration}, fast kernel path; SDK validation skipped).");
52        using var process = new System.Diagnostics.Process { StartInfo = new System.Diagnostics.ProcessStartInfo("powershell.exe", arguments) { WorkingDirectory = sdkRoot, UseShellExecute = false, CreateNoWindow = true, RedirectStandardOutput = true, RedirectStandardError = true }, EnableRaisingEvents = true };
53        process.OutputDataReceived += (_, e) => WriteLine(e.Data); process.ErrorDataReceived += (_, e) => WriteLine(e.Data);
54        if (!process.Start()) { _package.Output.WriteLine("[FAIL] NovaOryn build process did not start."); return; }
55        process.BeginOutputReadLine(); process.BeginErrorReadLine(); await Task.Run(() => process.WaitForExit(), token);
56        WriteLine(process.ExitCode == 0 ? "[ OK ] NovaOryn operation completed." : $"[FAIL] NovaOryn operation failed with exit code {process.ExitCode}.");
57    }
58    private void WriteLine(string line)
59    {
60        if (string.IsNullOrWhiteSpace(line)) return;
61        _package.JoinableTaskFactory.RunAsync(async delegate { await _package.JoinableTaskFactory.SwitchToMainThreadAsync(); _package.Output.WriteLine(line); }).FileAndForget("NovaOryn/Output");
62    }
63    private static string ResolveConfiguration(DTE dte)
64    {
65        ThreadHelper.ThrowIfNotOnUIThread(); try { return dte?.Solution?.SolutionBuild?.ActiveConfiguration?.Name ?? "Debug"; } catch { return "Debug"; }
66    }
67    private static Project ResolveStartupProject(DTE dte)
68    {
69        ThreadHelper.ThrowIfNotOnUIThread();
70        string unique = (dte.Solution.SolutionBuild.StartupProjects as Array)?.Cast<object>().Select(x => x?.ToString()).FirstOrDefault(x => !string.IsNullOrWhiteSpace(x));
71        if (string.IsNullOrWhiteSpace(unique)) return null;
72        foreach (Project p in dte.Solution.Projects) if (string.Equals(p.UniqueName, unique, StringComparison.OrdinalIgnoreCase)) return p;
73        return null;
74    }
75    private static string Quote(string value) => "\"" + value.Replace("\"", "\\\"") + "\"";
76}
