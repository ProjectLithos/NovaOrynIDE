1using System;
2using System.Runtime.InteropServices;
3using System.Threading;
4using System.Threading.Tasks;

6using Microsoft.VisualStudio;

9namespace NovaOryn.VisualStudio;
10/// <summary>
11/// Provides the NovaOryn Visual Studio package and registers its build and run command handlers.
12/// </summary>
13[PackageRegistration(UseManagedResourcesOnly = true, AllowsBackgroundLoading = true)]
14[InstalledProductRegistration("NovaOryn OS SDK", "Visual Studio integration for NovaOryn kernels", "0.29.0")]
15[ProvideMenuResource("Menus.ctmenu", 1)]
16[ProvideAutoLoad(UIContextGuids80.SolutionExists, PackageAutoLoadFlags.BackgroundLoad)]
17[Guid(PackageIds.PackageGuidString)]
18public sealed class NovaOrynPackage : AsyncPackage
19{
20    private IVsRegisterPriorityCommandTarget _registration;
21    private NovaOrynPriorityLaunchCommandTarget _target;
22    private uint _cookie;
23    private DTE _dte;
24    private SolutionEvents _solutionEvents;
25    internal NovaOrynOutputPane Output { get; private set; }
26    /// <summary>
27    /// Initialises the NovaOryn Visual Studio services and command targets.
28    /// </summary>
29    /// <param name="token">A cancellation token supplied by Visual Studio.</param>
30    /// <param name="progress">Visual Studio package-load progress reporting.</param>
31    /// <returns>A task representing package initialisation.</returns>
32    protected override async Task InitializeAsync(CancellationToken token, IProgress<ServiceProgressData> progress)
33    {
34        await JoinableTaskFactory.SwitchToMainThreadAsync(token);
35        Output = await NovaOrynOutputPane.CreateAsync(this, token);
36        _dte = await GetServiceAsync(typeof(SDTE)) as DTE;
37        if (_dte != null)
38        {
39            NovaOrynSolutionSynchronizer.EnsureSdkProjectsLoaded(_dte, Output);
40            _solutionEvents = _dte.Events.SolutionEvents;
41            _solutionEvents.ProjectAdded += OnProjectAdded;
42        }
43        await NovaOrynCommands.InitializeAsync(this, token);
44        _registration = await GetServiceAsync(typeof(SVsRegisterPriorityCommandTarget)) as IVsRegisterPriorityCommandTarget
45            ?? throw new InvalidOperationException("Visual Studio priority command service is unavailable.");
46        _target = new NovaOrynPriorityLaunchCommandTarget(new NovaOrynLaunchService(this));
47        ErrorHandler.ThrowOnFailure(_registration.RegisterPriorityCommandTarget(0, _target, out _cookie));
48        Output.WriteLine("[ OK ] NovaOryn Visual Studio extension 0.29.0 loaded.");
49    }
50    private void OnProjectAdded(Project project)
51    {
52        ThreadHelper.ThrowIfNotOnUIThread();
53        if (_dte != null) NovaOrynSolutionSynchronizer.EnsureSdkProjectsLoaded(_dte, Output);
54    }
55
56    /// <summary>
57    /// Releases the registered Visual Studio command target and package resources.
58    /// </summary>
59    /// <param name="disposing">True when managed resources should be released.</param>
60    protected override void Dispose(bool disposing)
61    {
62        ThreadHelper.ThrowIfNotOnUIThread();
63        if (disposing && _solutionEvents != null) _solutionEvents.ProjectAdded -= OnProjectAdded;
64        if (disposing && _registration != null && _cookie != 0) _registration.UnregisterPriorityCommandTarget(_cookie);
65        base.Dispose(disposing);
66    }
67}
