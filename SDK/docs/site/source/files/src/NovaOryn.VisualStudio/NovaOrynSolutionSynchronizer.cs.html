14    internal static bool EnsureSdkProjectsLoaded(DTE dte, NovaOrynOutputPane output)
15    {
16        ThreadHelper.ThrowIfNotOnUIThread();
17        if (_running || dte == null || dte.Solution == null || !dte.Solution.IsOpen) return false;
18        _running = true;
19        try
20        {
21            HashSet<string> loaded = new(StringComparer.OrdinalIgnoreCase);
22            List<string> novaProjects = new();
23            foreach (Project project in dte.Solution.Projects)
24            {
25                CollectProject(project, loaded, novaProjects);
26            }
27
28            bool changed = false;
29            foreach (string projectFile in novaProjects)
30            {
31                string directory = Path.GetDirectoryName(projectFile) ?? string.Empty;
32                string sdkDirectory = Path.Combine(directory, "Sdk");
33                if (!Directory.Exists(sdkDirectory)) continue;
34
35                if (RefreshSdkTree(sdkDirectory, output)) changed = true;
36                if (RepairRequiredProjectReferences(projectFile, sdkDirectory, output)) changed = true;
37
38                foreach (string sdkProject in Directory.EnumerateFiles(sdkDirectory, "*.csproj", SearchOption.AllDirectories))
39                {
40                    string full = Path.GetFullPath(sdkProject);
41                    if (loaded.Contains(full)) continue;
42                    Project added = dte.Solution.AddFromFile(full, false);
43                    if (added == null) continue;
44                    loaded.Add(full);
45                    changed = true;
46                    output.WriteLine($"[ OK ] Loaded NovaOryn SDK project into solution: {Path.GetFileNameWithoutExtension(full)}");
47                }
48            }
49            return changed || novaProjects.Count != 0;
50        }
51        finally
52        {
53            _running = false;
54        }
55    }

57    private static bool RefreshSdkTree(string sdkDirectory, NovaOrynOutputPane output)
58    {
59        ThreadHelper.ThrowIfNotOnUIThread();
60        string installedRoot = SdkInstallationDetector.ResolveSdkRoot();
61        string sourceSdkDirectory = Path.Combine(installedRoot, "templates", "NovaOrynKernel", "Sdk");
62        if (!Directory.Exists(sourceSdkDirectory))
63        {
64            output.WriteLine($"[FAIL] Installed NovaOryn SDK template tree was not found: {sourceSdkDirectory}");
65            return false;
66        }
67
68        Directory.CreateDirectory(sdkDirectory);
69        HashSet<string> expected = new(StringComparer.OrdinalIgnoreCase);
70        bool changed = false;
71        foreach (string source in Directory.EnumerateFiles(sourceSdkDirectory, "*", SearchOption.AllDirectories))
72        {
73            string relative = GetRelativeChildPath(sourceSdkDirectory, source);
74            expected.Add(relative);
75            string destination = Path.Combine(sdkDirectory, relative);
76            Directory.CreateDirectory(Path.GetDirectoryName(destination) ?? sdkDirectory);
77            if (File.Exists(destination) && FilesEqual(source, destination)) continue;
78            File.Copy(source, destination, true);
79            changed = true;
80        }
81
82        foreach (string destination in Directory.EnumerateFiles(sdkDirectory, "*", SearchOption.AllDirectories))
83        {
84            string relative = GetRelativeChildPath(sdkDirectory, destination);
85            if (expected.Contains(relative)) continue;
86            File.Delete(destination);
87            changed = true;
88        }
89
90        RemoveEmptyDirectories(sdkDirectory);
91        if (changed) output.WriteLine("[ OK ] Refreshed SDK-owned project assemblies from the installed NovaOryn template; user Kernel\\Kernel.cs was preserved.");
92        return changed;
93    }

95    private static string GetRelativeChildPath(string root, string path)
96    {
97        string normalizedRoot = Path.GetFullPath(root).TrimEnd(Path.DirectorySeparatorChar, Path.AltDirectorySeparatorChar);
98        string normalizedPath = Path.GetFullPath(path);
99        if (normalizedPath.Length <= normalizedRoot.Length) return Path.GetFileName(normalizedPath);
100        return normalizedPath.Substring(normalizedRoot.Length + 1);
101    }

103    private static bool FilesEqual(string left, string right)
104    {
105        FileInfo leftInfo = new FileInfo(left);
106        FileInfo rightInfo = new FileInfo(right);
107        if (leftInfo.Length != rightInfo.Length) return false;
108        byte[] leftBytes = File.ReadAllBytes(left);
109        byte[] rightBytes = File.ReadAllBytes(right);
110        if (leftBytes.Length != rightBytes.Length) return false;
111        for (int index = 0; index < leftBytes.Length; index++)
112        {
113            if (leftBytes[index] != rightBytes[index]) return false;
114        }
115        return true;
116    }

118    private static bool RemoveEmptyDirectories(string root)
119    {
120        string[] directories = Directory.EnumerateDirectories(root, "*", SearchOption.AllDirectories).ToArray();
121        Array.Sort(directories, (left, right) => right.Length.CompareTo(left.Length));
122        bool removed = false;
123        foreach (string directory in directories)
124        {
125            if (Directory.EnumerateFileSystemEntries(directory).Any()) continue;
126            Directory.Delete(directory);
127            removed = true;
128        }
129        return removed;
130    }

132    private static bool RepairRequiredProjectReferences(string projectFile, string sdkDirectory, NovaOrynOutputPane output)
133    {
134        ThreadHelper.ThrowIfNotOnUIThread();
135        if (!ValidateRequiredSdkFiles(sdkDirectory, output)) return false;
136
137        string text = File.ReadAllText(projectFile);
138        string original = text;
139        string installedRoot = SdkInstallationDetector.ResolveSdkRoot();
140        string[] requiredProjects = GetRequiredProjectNames(installedRoot, output);
141        if (requiredProjects.Length == 0) return false;
142
143        const string itemGroupEnd = "  </ItemGroup>";
144        int insertIndex = text.IndexOf(itemGroupEnd, StringComparison.Ordinal);
145        if (insertIndex < 0)
146        {
147            output.WriteLine("[FAIL] NovaOryn root project is missing the expected SDK ProjectReference ItemGroup and could not be repaired automatically.");
148            return false;
149        }
150
151        foreach (string projectName in requiredProjects)
152        {
153            string relative = $"Sdk\\{projectName}\\{projectName}.csproj";
154            if (text.IndexOf(relative, StringComparison.OrdinalIgnoreCase) >= 0) continue;
155            string reference = $"    <ProjectReference Include=\"{relative}\" />" + Environment.NewLine;
156            text = text.Insert(insertIndex, reference);
157            insertIndex += reference.Length;
158            output.WriteLine($"[ OK ] Repaired {projectName} project reference for Visual Studio design-time evaluation.");
159        }
160
161        if (string.Equals(original, text, StringComparison.Ordinal)) return false;
162        File.WriteAllText(projectFile, text);
163        File.SetLastWriteTimeUtc(projectFile, DateTime.UtcNow);
164        output.WriteLine("[ OK ] NovaOryn root project reference graph repaired; Visual Studio project evaluation will refresh from the updated project file.");
165        return true;
166    }

169    private static string[] GetRequiredProjectNames(string installedRoot, NovaOrynOutputPane output)
170    {
171        ThreadHelper.ThrowIfNotOnUIThread();
172        string templateProject = Path.Combine(installedRoot, "templates", "NovaOrynKernel", "NovaOrynKernel.csproj");
173        if (!File.Exists(templateProject))
174        {
175            output.WriteLine($"[FAIL] Installed NovaOryn root project template was not found: {templateProject}");
176            return Array.Empty<string>();
177        }
178
179        List<string> names = new();
180        foreach (string line in File.ReadAllLines(templateProject))
181        {
182            const string marker = "<ProjectReference Include=\"Sdk\\";
183            int markerIndex = line.IndexOf(marker, StringComparison.OrdinalIgnoreCase);
184            if (markerIndex < 0) continue;
185            int nameStart = markerIndex + marker.Length;
186            int nameEnd = line.IndexOf('\\', nameStart);
187            if (nameEnd <= nameStart) continue;
188            string name = line.Substring(nameStart, nameEnd - nameStart);
189            if (names.Contains(name, StringComparer.OrdinalIgnoreCase)) continue;
190            names.Add(name);
191        }
192
193        if (names.Count != 0) return names.ToArray();
194        output.WriteLine("[FAIL] Installed NovaOryn root project template contains no SDK ProjectReference entries and cannot be used to repair Visual Studio projects.");
195        return Array.Empty<string>();
196    }

238    private static bool CollectProject(Project project, HashSet<string> loaded, List<string> novaProjects)
239    {
240        ThreadHelper.ThrowIfNotOnUIThread();
241        if (project == null) return false;
242        try
243        {
244            string fullName = project.FullName;
245            if (!string.IsNullOrWhiteSpace(fullName) && File.Exists(fullName))
246            {
247                string full = Path.GetFullPath(fullName);
248                loaded.Add(full);
249                if (NovaOrynProjectRecognizer.IsNovaOrynProject(full)) novaProjects.Add(full);
250            }
251        }
252        catch (Exception)
253        {
254            return false;
255        }
256        return true;
257    }
