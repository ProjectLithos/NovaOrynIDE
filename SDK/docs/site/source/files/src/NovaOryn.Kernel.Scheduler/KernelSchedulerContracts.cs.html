5/// <summary>Identifies the lifecycle state of a kernel thread.</summary>
6public enum KernelThreadState : Byte
7{
8    /// <summary>The slot is unused.</summary>
9    Unused = 0,
10    /// <summary>The thread can be selected by a scheduler.</summary>
11    Ready = 1,
12    /// <summary>The thread is executing on a processor.</summary>
13    Running = 2,
14    /// <summary>The thread is blocked until explicitly awakened.</summary>
15    Blocked = 3,
16    /// <summary>The thread has completed and will not run again.</summary>
17    Terminated = 4
18}
