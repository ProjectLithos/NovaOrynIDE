35    /// <summary>Initializes scheduler tables after SMP, heap and clock initialization.</summary>
36    public static Boolean Initialize()
37    {
38        if (_initialized) return true;
39        if (!KernelSmp.IsInitialized() || !KernelTime.IsInitialized) return false;
40        _processorCount = KernelSmp.GetProcessorCount();
41        if (_processorCount == 0U) return false;
42        if (!KernelSchedulerMath.TryGetTableBytes(MaximumThreads, (UInt32)sizeof(ThreadRecord), out UInt64 threadBytes)) return false;
43        if (!KernelSchedulerMath.TryGetTableBytes(_processorCount, (UInt32)sizeof(CpuScheduleState), out UInt64 cpuBytes)) return false;
44        if (!KernelHeap.TryAllocate(threadBytes, 64UL, true, out KernelHeapAllocation threadAlloc)) return false;
45        if (!KernelHeap.TryAllocate(cpuBytes, 64UL, true, out KernelHeapAllocation cpuAlloc)) return false;
46        _threads = (ThreadRecord*)threadAlloc.Address; _cpus = (CpuScheduleState*)cpuAlloc.Address;
47        for (UInt32 cpu=0U; cpu<_processorCount; cpu++)
48        {
49            CpuScheduleState* state=_cpus+cpu; state->CurrentSlot=0xFFFFFFFFU; state->IdleSlot=0xFFFFFFFFU;
50            if (!KernelSmp.TrySetSchedulerContext(cpu, (UInt64)state)) return false;
51        }
52        if (!CreateBootstrapThread()) return false;
53        _timerPreemption = KernelTime.GetCapabilities().HasLocalApicTimer;
54        _initialized=true; return true;
55    }

72    public static Boolean TryCreateThread(UInt64 entryPoint, UInt64 argument, KernelThreadPriority priority, UInt64 affinityMask, UInt64 stackBytes, out UInt64 threadId)
73    {
74        threadId=0UL; if (!_initialized || entryPoint==0UL || affinityMask==0UL) return false;
75        if (stackBytes==0UL) stackBytes=DefaultStackBytes;
76        if (!KernelSchedulerMath.IsValidStackSize(stackBytes)) return false;
77        if (!FindUnusedSlot(out UInt32 slot)) return false;
78        if (!KernelHeap.TryAllocate(stackBytes, 4096UL, true, out KernelHeapAllocation stack)) return false;
79        if (!KernelHeap.TryAllocate(ContextBytes, 16UL, true, out KernelHeapAllocation context)) { KernelHeap.TryRelease(stack); return false; }
80        if (!Native.InitializeThreadContext(context.Address, stack.Address+stack.ByteCount, entryPoint, argument)) { KernelHeap.TryRelease(context); KernelHeap.TryRelease(stack); return false; }
81        ThreadRecord* r=_threads+slot; r->Id=_nextThreadId++; r->StackBase=stack.Address; r->StackTop=stack.Address+stack.ByteCount; r->EntryPoint=entryPoint; r->Argument=argument; r->AffinityMask=affinityMask; r->ContextAddress=context.Address; r->State=(UInt32)KernelThreadState.Ready; r->Priority=(UInt32)priority; r->ProcessorIndex=NoProcessor; r->NextReady=0xFFFFFFFFU;
82        _activeThreads++; threadId=r->Id; return true;
83    }

105    public static Boolean TrySelectNext(UInt32 processorIndex, out UInt64 threadId)
106    {
107        threadId=0UL; if (!_initialized || processorIndex>=_processorCount) return false; UInt32 best=0xFFFFFFFFU; UInt32 bestPriority=0U;
108        for (UInt32 slot=0U;slot<MaximumThreads;slot++) { ThreadRecord* r=_threads+slot; if (r->State!=(UInt32)KernelThreadState.Ready || !KernelSchedulerMath.AllowsProcessor(r->AffinityMask,processorIndex)) continue; if (best==0xFFFFFFFFU || r->Priority>bestPriority) { best=slot; bestPriority=r->Priority; } }
109        if (best==0xFFFFFFFFU) return false; threadId=(_threads+best)->Id; return true;
110    }

117    private static Boolean Schedule(UInt32 cpu, Boolean preempt, out UInt64 nextThreadId)
118    {
119        nextThreadId=0UL; if (!_initialized || cpu>=_processorCount) return false; CpuScheduleState* cs=_cpus+cpu;
120        if (cs->CurrentSlot!=0xFFFFFFFFU) { ThreadRecord* current=_threads+cs->CurrentSlot; if (current->State==(UInt32)KernelThreadState.Running) { current->State=(UInt32)KernelThreadState.Ready; current->ProcessorIndex=NoProcessor; } }
121        if (!TrySelectNext(cpu,out nextThreadId)) return false; if (!FindThread(nextThreadId,out UInt32 slot)) return false; ThreadRecord* next=_threads+slot; UInt32 previousSlot=cs->CurrentSlot; ThreadRecord* previous=previousSlot==0xFFFFFFFFU ? null : _threads+previousSlot; next->State=(UInt32)KernelThreadState.Running; next->ProcessorIndex=cpu; cs->CurrentSlot=slot; cs->SwitchCount++; if(preempt) cs->PreemptionCount++; cs->LastDispatchNanoseconds=KernelTime.GetMonotonicNanoseconds();
122        if (previous!=null && previous!=next && previous->ContextAddress!=0UL && next->ContextAddress!=0UL) return Native.SwitchThreadContext(previous->ContextAddress,next->ContextAddress);
123        return true;
124    }
125    private static Boolean CreateBootstrapThread()
126    {
127        if (!KernelSmp.TryGetCurrentProcessor(out KernelProcessorState cpu)) return false; if (!KernelHeap.TryAllocate(ContextBytes,16UL,true,out KernelHeapAllocation context)) return false; ThreadRecord* r=_threads; r->Id=_nextThreadId++; r->State=(UInt32)KernelThreadState.Running; r->Priority=(UInt32)KernelThreadPriority.Critical; r->ProcessorIndex=cpu.Index; r->AffinityMask=cpu.Index<64U ? 1UL<<(Int32)cpu.Index : 0xFFFFFFFFFFFFFFFFUL; r->ContextAddress=context.Address; _cpus[cpu.Index].CurrentSlot=0U; _activeThreads=1U; return true;
128    }
129    private static Boolean FindUnusedSlot(out UInt32 slot)
130    { slot=0U; for(UInt32 i=1U;i<MaximumThreads;i++) if((_threads+i)->State==(UInt32)KernelThreadState.Unused){slot=i;return true;} return false; }
131    private static Boolean FindThread(UInt64 id,out UInt32 slot)
