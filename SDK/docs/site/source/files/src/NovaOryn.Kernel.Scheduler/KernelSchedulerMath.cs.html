10    /// <summary>Returns whether a logical processor is admitted by a 64-bit affinity mask.</summary>
11    public static Boolean AllowsProcessor(UInt64 mask, UInt32 processorIndex) => processorIndex < 64U && (mask & (1UL << (Int32)processorIndex)) != 0UL;
12    /// <summary>Clamps a requested quantum to NovaOryn's supported 100 microsecond through 1 second interval.</summary>
13    public static UInt64 ClampQuantum(UInt64 nanoseconds)
14    {
15        if (nanoseconds < 100000UL) return 100000UL;
16        if (nanoseconds > 1000000000UL) return 1000000000UL;
17        return nanoseconds;
18    }
19    /// <summary>Computes the byte count for a fixed-size record table without overflow.</summary>
20    public static Boolean TryGetTableBytes(UInt32 count, UInt32 recordBytes, out UInt64 bytes)
21    {
22        bytes = 0UL;
23        if (count == 0U || recordBytes == 0U) return false;
24        UInt64 result = (UInt64)count * (UInt64)recordBytes;
25        if (result / (UInt64)recordBytes != (UInt64)count) return false;
26        bytes = result; return true;
27    }
