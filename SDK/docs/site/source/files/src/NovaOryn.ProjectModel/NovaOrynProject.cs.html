17        error = string.Empty;
18        if (string.IsNullOrWhiteSpace(path)) throw new ArgumentException("Project path is required.", nameof(path));
19        string manifestPath = Path.GetFullPath(path);

21        try
22        {
23            project = JsonSerializer.Deserialize<NovaOrynProject>(File.ReadAllText(manifestPath));
24            if (project is null) { error = "Project manifest was empty."; return false; }
25            string root = Path.GetDirectoryName(manifestPath) ?? Environment.CurrentDirectory;
26            project = project with
27            {
28                ProjectFile = Resolve(root, project.ProjectFile),
29                OutputDirectory = Resolve(root, project.OutputDirectory)
30            };
31            return project.Validate(out error);
32        }

40    public bool Validate(out string error)
41    {
42        if (string.IsNullOrWhiteSpace(Name)) { error = "Name is required."; return false; }
43        if (!File.Exists(ProjectFile)) { error = $"Kernel project was not found: {ProjectFile}"; return false; }
44        if (!string.Equals(TargetArchitecture, "x64", StringComparison.OrdinalIgnoreCase)) { error = "NovaOryn 0.0.30 supports x64 only."; return false; }
45        if (!string.Equals(BootProtocol, "Uefi", StringComparison.OrdinalIgnoreCase)) { error = "NovaOryn 0.0.30 supports UEFI only."; return false; }
46        if (!string.Equals(KernelEntry, "KMain", StringComparison.Ordinal)) { error = "KernelEntry must be KMain."; return false; }
47        error = string.Empty;
48        return true;
