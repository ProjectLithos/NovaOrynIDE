458    // Decoded PS/2 events are consumed here; KernelConsole never rereads i8042 hardware.
459    private static Boolean HandleKeyboardEvent(Ps2KeyboardEvent input)
460    {
461        if (!input.Pressed) return true;
462        if (input.Key == Ps2Key.Up) return KernelConsole.ScrollUp();
463        if (input.Key == Ps2Key.Down) return KernelConsole.ScrollDown();
464        if (input.Control && input.Key == Ps2Key.D1) return KernelConsole.SetFramebufferBufferCount(1U);
465        if (input.Control && input.Key == Ps2Key.D2) return KernelConsole.SetFramebufferBufferCount(2U);
466        if (input.Control && input.Key == Ps2Key.D3) return KernelConsole.SetFramebufferBufferCount(3U);
467        if (input.Key == Ps2Key.D1) return KernelConsole.SetFontPreset(1U);
468        if (input.Key == Ps2Key.D2) return KernelConsole.SetFontPreset(2U);
469        if (input.Key == Ps2Key.D3) return KernelConsole.SetFontPreset(3U);
470        Char character = input.Character;
471        if (character == '\0' || character == '\b') return true;
472        if (character == '\n') return KernelConsole.WriteLine("");
473        if ((UInt32)character > 0x7FU) return KernelConsole.Write("?");
474        return KernelConsole.Write((Byte)character);
475    }
476    private static Boolean HandleUsbKeyboardEvent(UsbHidKeyboardEvent input)
477    {
478        if (!input.Pressed) return true;
479        if (input.Usage == 82U) return KernelConsole.ScrollUp();
480        if (input.Usage == 81U) return KernelConsole.ScrollDown();
481        Boolean control=(input.Modifiers&0x11U)!=0;
482        if (control && input.Usage == 30U) return KernelConsole.SetFramebufferBufferCount(1U);
483        if (control && input.Usage == 31U) return KernelConsole.SetFramebufferBufferCount(2U);
484        if (control && input.Usage == 32U) return KernelConsole.SetFramebufferBufferCount(3U);
485        if (input.Usage == 30U) return KernelConsole.SetFontPreset(1U);
486        if (input.Usage == 31U) return KernelConsole.SetFontPreset(2U);
487        if (input.Usage == 32U) return KernelConsole.SetFontPreset(3U);
488        Char character=input.Character;
489        if (character == '\0' || character == '\b') return true;
490        if (character == '\n') return KernelConsole.WriteLine("");
491        if ((UInt32)character > 0x7FU) return KernelConsole.Write("?");
492        return KernelConsole.Write((Byte)character);
