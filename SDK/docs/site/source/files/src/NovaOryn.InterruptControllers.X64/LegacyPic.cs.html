1namespace NovaOryn.InterruptControllers.X64;
2/// <summary>Controls the pair of legacy 8259 PICs during APIC takeover.</summary>
3public sealed class LegacyPic : ILegacyPic
4{
5    private const ushort MasterCommand=0x20, MasterData=0x21, SlaveCommand=0xA0, SlaveData=0xA1;
6    /// <inheritdoc />
7    public bool MaskAll() => NativeMethods.WritePort8(MasterData,0xFF) && NativeMethods.WritePort8(SlaveData,0xFF);
8    /// <inheritdoc />
9    public bool Disable() => MaskAll();
10    /// <inheritdoc />
11    public bool EndOfInterrupt(byte irq)
12    {
13        if (irq >= 16) return false;
14        bool slave = irq >= 8 ? NativeMethods.WritePort8(SlaveCommand,0x20) : true;
15        return slave && NativeMethods.WritePort8(MasterCommand,0x20);
16    }
17}
