6/// <summary>Coordinates PIC, Local APIC, I/O APIC, MSI, MSI-X, and x2APIC delivery behind one driver contract.</summary>
7public sealed class X64InterruptController : IInterruptController
8{
9    private const uint ApicBaseMsr=0x1B, X2ApicEoiMsr=0x80B, X2ApicIcrMsr=0x830;
10    private readonly IInterruptVectorAllocator _vectors;
11    private readonly RouteState[] _routes = new RouteState[256];
12    private readonly LegacyPic _pic = new();
13    private ulong _localApicBase;
14    private ulong _ioApicBase;
15    private bool _x2Apic;
16    private ulong _generation;
17
18    /// <summary>Creates an x64 interrupt-controller stack using the shared IDT vector allocator.</summary>
19    public X64InterruptController(IInterruptVectorAllocator vectors) => _vectors=vectors ?? throw new ArgumentNullException(nameof(vectors));
20    /// <summary>Configures discovered APIC MMIO addresses and mode.</summary>
21    public bool Configure(PhysicalAddress localApicBase, PhysicalAddress ioApicBase, bool x2Apic)
22    {
23        if (!x2Apic && localApicBase.Value==0) return false;
24        _localApicBase=localApicBase.Value; _ioApicBase=ioApicBase.Value; _x2Apic=x2Apic;
25        return _pic.Disable() && EnableLocalApic();
26    }
27    /// <inheritdoc />
28    public InterruptControllerCapabilities GetCapabilities() => new(true,true,_ioApicBase!=0,true,true,_x2Apic,224,0,15);
29    /// <inheritdoc />
30    public byte AllocateVector() => _vectors.Allocate();
31    /// <inheritdoc />
32    public bool ReleaseVector(byte vector) => _vectors.Release(vector);
33    /// <inheritdoc />
34    public InterruptRouteResult Route(InterruptRouteConfiguration c)
35    {
36        if (c.Vector<32 || c.Priority>15) return new(false,default,"Invalid vector or priority.");
37        ulong id=++_generation; var handle=new InterruptRouteHandle(id);
38        _routes[c.Vector]=new(true,handle,c);
39        bool ok=c.PreferredMechanism switch
40        {
41            InterruptDeliveryMechanism.IoApic => ProgramIoApic(c),
42            InterruptDeliveryMechanism.Msi or InterruptDeliveryMechanism.MsiX => true,
43            InterruptDeliveryMechanism.LegacyPic => false,
44            _ => true
45        };
46        return ok ? new(true,handle,string.Empty) : new(false,default,"Requested delivery mechanism is unavailable.");
47    }
48    /// <inheritdoc />
49    public bool RemoveRoute(InterruptRouteHandle handle) => Update(handle, static _=>default);
50    /// <inheritdoc />
51    public bool Mask(InterruptRouteHandle handle) => Mutate(handle,true,null,null);
52    /// <inheritdoc />
53    public bool Unmask(InterruptRouteHandle handle) => Mutate(handle,false,null,null);
54    /// <inheritdoc />
55    public bool SetAffinity(InterruptRouteHandle handle, InterruptAffinity affinity) => Mutate(handle,null,affinity,null);
56    /// <inheritdoc />
57    public bool SetPriority(InterruptRouteHandle handle, byte priority) => priority<=15 && Mutate(handle,null,null,priority);
58    /// <inheritdoc />
59    public bool EndOfInterrupt(byte vector)
60    {
61        if (_x2Apic) return NativeMethods.WriteMsr(X2ApicEoiMsr,0);
62        return _localApicBase!=0 && NativeMethods.WriteMmio32(_localApicBase+0xB0,0);
63    }
64    /// <inheritdoc />
65    public bool SendInterprocessorInterrupt(InterprocessorInterrupt interrupt)
66    {
67        ulong value=((ulong)interrupt.Target.ProcessorId.Value<<32) | ((ulong)interrupt.DeliveryMode<<8) | interrupt.Vector;
68        if (interrupt.AssertLevel) value|=1UL<<14;
69        if (_x2Apic) return NativeMethods.WriteMsr(X2ApicIcrMsr,value);
70        return _localApicBase!=0 && NativeMethods.WriteMmio32(_localApicBase+0x310,(uint)(value>>32)) && NativeMethods.WriteMmio32(_localApicBase+0x300,(uint)value);
71    }
72    /// <inheritdoc />
73    public MessageSignalledInterrupt CreateMessage(InterruptRouteHandle handle)
74    {
75        RouteState s=Find(handle); if (!s.Active) return default;
76        ulong address=0xFEE00000UL | ((ulong)s.Configuration.Affinity.ProcessorId.Value<<12);
77        uint data=s.Configuration.Vector | ((uint)InterruptDeliveryMode.Fixed<<8);
78        if (s.Configuration.TriggerMode==InterruptTriggerMode.Level) data|=1U<<15;
79        return new(address,data,s.Configuration.Vector);
80    }
81    private bool EnableLocalApic()
82    {
83        ulong value=NativeMethods.ReadMsr(ApicBaseMsr) | (1UL<<11);
84        if (_x2Apic) value|=1UL<<10;
85        return NativeMethods.WriteMsr(ApicBaseMsr,value);
86    }
87    private bool ProgramIoApic(InterruptRouteConfiguration c)
88    {
89        if (_ioApicBase==0 || c.Source.Value>=120) return false;
90        uint low=c.Vector | ((uint)c.Priority<<4);
91        if (c.Polarity==InterruptPolarity.ActiveLow) low|=1U<<13;
92        if (c.TriggerMode==InterruptTriggerMode.Level) low|=1U<<15;
93        if (c.InitiallyMasked) low|=1U<<16;
94        uint high=c.Affinity.ProcessorId.Value<<24;
95        uint index=0x10+(c.Source.Value*2);
96        return WriteIoApic(index,low) && WriteIoApic(index+1,high);
97    }
98    private bool WriteIoApic(uint register,uint value) => NativeMethods.WriteMmio32(_ioApicBase,register) && NativeMethods.WriteMmio32(_ioApicBase+0x10,value);
99    private bool Mutate(InterruptRouteHandle h,bool? masked,InterruptAffinity? affinity,byte? priority) => Update(h,s=>new(true,s.Handle,s.Configuration with { InitiallyMasked=masked??s.Configuration.InitiallyMasked, Affinity=affinity??s.Configuration.Affinity, Priority=priority??s.Configuration.Priority }));
100    private bool Update(InterruptRouteHandle h,Func<RouteState,RouteState> transform)
101    {
102        for(int i=0;i<_routes.Length;i++) if(_routes[i].Active && _routes[i].Handle==h){ _routes[i]=transform(_routes[i]); return !_routes[i].Active || _routes[i].Configuration.PreferredMechanism!=InterruptDeliveryMechanism.IoApic || ProgramIoApic(_routes[i].Configuration); }
103        return false;
104    }
105    private RouteState Find(InterruptRouteHandle h){ foreach(var s in _routes) if(s.Active&&s.Handle==h)return s; return default; }
106    private readonly record struct RouteState(bool Active,InterruptRouteHandle Handle,InterruptRouteConfiguration Configuration);
107}
