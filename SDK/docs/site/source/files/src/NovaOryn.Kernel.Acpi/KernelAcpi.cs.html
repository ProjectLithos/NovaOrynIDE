147    /// <summary>Validates the RSDP and root table supplied by firmware.</summary>
148    /// <returns><see langword="true"/> when a checksummed RSDT or XSDT is ready for table discovery.</returns>
149    public static Boolean Initialize(BootContext boot)
150    {
151        if (_initialized) return true;
152        _rsdpAddress = boot.GetAcpiRootPointerAddress();
153        if (_rsdpAddress == 0UL) { _status = KernelAcpiStatus.RootPointerUnavailable; return false; }
154        Byte* rsdp = (Byte*)_rsdpAddress;
155        if (!HasRsdpSignature(rsdp) || !ChecksumIsZero(rsdp, 20U)) { _status = KernelAcpiStatus.InvalidRootPointer; return false; }
156        _revision = rsdp[15];
157        UInt32 rsdt = Read32(rsdp + 16);
158        UInt64 xsdt = 0UL;
159        if (_revision >= 2)
160        {
161            UInt32 length = Read32(rsdp + 20);
162            if (length < 36U || length > 4096U || !ChecksumIsZero(rsdp, length)) { _status = KernelAcpiStatus.InvalidRootPointer; return false; }
163            xsdt = Read64(rsdp + 24);
164        }
165        _usesXsdt = xsdt != 0UL;
166        _rootAddress = _usesXsdt ? xsdt : rsdt;
167        if (_rootAddress == 0UL) { _status = KernelAcpiStatus.RootTableUnavailable; return false; }
168        Byte* root = (Byte*)_rootAddress;
169        UInt32 expected = _usesXsdt ? XsdtSignature : RsdtSignature;
170        if (Read32(root) != expected || !TryValidateTable(root, out UInt32 rootLength)) { _status = KernelAcpiStatus.InvalidRootTable; return false; }
171        UInt32 entrySize = _usesXsdt ? 8U : 4U;
172        UInt32 payload = rootLength - 36U;
173        if (payload % entrySize != 0U) { _status = KernelAcpiStatus.InvalidRootTable; return false; }
174        _rootEntryCount = payload / entrySize;
175        if (_rootEntryCount > MaximumRootEntries) { _status = KernelAcpiStatus.RootTableTooLarge; return false; }
176        _initialized = true;
177        _status = KernelAcpiStatus.Success;
178        return true;
179    }

181    /// <summary>Finds the first validated ACPI table with the requested four-byte signature.</summary>
182    /// <returns><see langword="true"/> when a checksummed table was found.</returns>
183    public static Boolean TryGetTable(UInt32 signature, out UInt64 address, out UInt32 length)
184    {
185        address = 0UL; length = 0U;
186        if (!_initialized) return false;
187        Byte* root = (Byte*)_rootAddress;
188        UInt32 entrySize = _usesXsdt ? 8U : 4U;
189        for (UInt32 index = 0U; index < _rootEntryCount; index++)
190        {
191            Byte* entry = root + 36U + index * entrySize;
192            UInt64 candidateAddress = _usesXsdt ? Read64(entry) : Read32(entry);
193            if (candidateAddress == 0UL) continue;
194            Byte* candidate = (Byte*)candidateAddress;
195            if (Read32(candidate) != signature) continue;
196            if (!TryValidateTable(candidate, out UInt32 candidateLength)) continue;
197            address = candidateAddress; length = candidateLength; return true;
198        }
199        return false;
200    }

209    /// <summary>Gets one enabled processor record by logical discovery index.</summary>
210    public static Boolean TryGetProcessor(UInt32 requestedIndex, out AcpiProcessorInfo processor)
211    {
212        processor = default;
213        if (!TryGetTable(ApicSignature, out UInt64 address, out UInt32 length) || length < 44U) return false;
214        Byte* table = (Byte*)address; UInt32 offset = 44U; UInt32 found = 0U;
215        while (offset + 2U <= length)
216        {
217            Byte type = table[offset]; Byte entryLength = table[offset + 1U];
218            if (entryLength < 2U || offset + entryLength > length) return false;
219            if (type == 0U && entryLength >= 8U)
220            {
221                UInt32 flags = Read32(table + offset + 4U); Boolean enabled = (flags & 3U) != 0U;
222                if (enabled && found++ == requestedIndex) { processor = new AcpiProcessorInfo(table[offset + 3U], table[offset + 2U], false, true); return true; }
223            }
224            else if (type == 9U && entryLength >= 16U)
225            {
226                UInt32 flags = Read32(table + offset + 8U); Boolean enabled = (flags & 3U) != 0U;
227                if (enabled && found++ == requestedIndex) { processor = new AcpiProcessorInfo(Read32(table + offset + 4U), Read32(table + offset + 12U), true, true); return true; }
228            }
229            offset += entryLength;
230        }
231        return false;
232    }

313    private static UInt32 CountMadtEntries(Byte firstType, Byte secondType, Boolean onlyEnabledProcessors)
314    {
315        if (!TryGetTable(ApicSignature, out UInt64 address, out UInt32 length) || length < 44U) return 0U;
316        Byte* table = (Byte*)address; UInt32 offset = 44U; UInt32 count = 0U;
317        while (offset + 2U <= length)
318        {
319            Byte type = table[offset]; Byte entryLength = table[offset + 1U];
320            if (entryLength < 2U || offset + entryLength > length) return count;
321            if (type == firstType || type == secondType)
322            {
323                if (!onlyEnabledProcessors) count++;
324                else if (type == 0U && entryLength >= 8U && (Read32(table + offset + 4U) & 3U) != 0U) count++;
325                else if (type == 9U && entryLength >= 16U && (Read32(table + offset + 8U) & 3U) != 0U) count++;
326            }
327            offset += entryLength;
328        }
329        return count;
330    }

332    private static Boolean TryValidateTable(Byte* table, out UInt32 length)
333    {
334        length = 0U; if (table == null) return false;
335        UInt32 candidateLength = Read32(table + 4U);
336        if (candidateLength < 36U || candidateLength > MaximumTableLength) return false;
337        if (!ChecksumIsZero(table, candidateLength)) return false;
338        length = candidateLength; return true;
339    }

341    private static Boolean HasRsdpSignature(Byte* value)
342    {
343        return value != null && value[0] == 0x52U && value[1] == 0x53U && value[2] == 0x44U && value[3] == 0x20U &&
344            value[4] == 0x50U && value[5] == 0x54U && value[6] == 0x52U && value[7] == 0x20U;
345    }

347    private static Boolean ChecksumIsZero(Byte* value, UInt32 length)
348    {
349        Byte sum = (Byte)0U; for (UInt32 index = 0U; index < length; index++) sum = (Byte)(sum + value[index]); return sum == (Byte)0U;
350    }
351    private static UInt16 Read16(Byte* value) => (UInt16)(value[0] | ((UInt16)value[1] << 8));
352    private static UInt32 Read32(Byte* value) => (UInt32)(value[0] | ((UInt32)value[1] << 8) | ((UInt32)value[2] << 16) | ((UInt32)value[3] << 24));
353    private static UInt64 Read64(Byte* value) => (UInt64)Read32(value) | ((UInt64)Read32(value + 4) << 32);
