5/// <summary>Contains deterministic SMP layout and startup-vector calculations.</summary>
6public static class KernelSmpMath
7{
8    /// <summary>Gets whether a physical address can be encoded as an x86 SIPI startup vector.</summary>
9    public static Boolean IsValidStartupTrampoline(UInt64 address) => address >= 0x1000UL && address < 0x100000UL && (address & 0xFFFUL) == 0UL;
10
11    /// <summary>Converts one validated low-memory trampoline address to its SIPI vector.</summary>
12    public static Boolean TryGetStartupVector(UInt64 address, out Byte vector)
13    {
14        vector = (Byte)0U;
15        if (!IsValidStartupTrampoline(address)) return false;
16        UInt64 value = address >> 12;
17        if (value == 0UL || value > 0xFFUL) return false;
18        vector = (Byte)value;
19        return true;
20    }
21
22    /// <summary>Gets whether an APIC identifier can be targeted through the xAPIC destination field.</summary>
23    public static Boolean IsXApicDestination(UInt32 apicId) => apicId <= 0xFFU;
24
25    /// <summary>Calculates a checked per-CPU table byte requirement.</summary>
26    public static Boolean TryGetStateTableBytes(UInt32 processorCount, UInt32 recordBytes, out UInt64 requiredBytes)
27    {
28        requiredBytes = 0UL;
29        if (processorCount == 0U || recordBytes == 0U) return false;
30        UInt64 count = processorCount;
31        if (count > 0xFFFFFFFFFFFFFFFFUL / recordBytes) return false;
32        requiredBytes = count * recordBytes;
33        return requiredBytes != 0UL;
34    }
