51    /// <summary>Initializes per-CPU records and starts each xAPIC application processor through INIT/SIPI.</summary>
52    /// <returns><see langword="true"/> when per-CPU state is usable; individual AP failures are reported through status and processor snapshots.</returns>
53    public static Boolean Initialize(BootContext boot)
54    {
55        if (_initialized) return true;
56        if (!KernelAcpi.IsInitialized()) return false;
57        _processorCount = KernelAcpi.GetProcessorCount();
58        if (_processorCount == 0U) { _status = KernelSmpStatus.NoProcessors; return false; }
59        if (_processorCount > MaximumProcessors) _processorCount = MaximumProcessors;
60        if (!KernelSmpMath.TryGetStateTableBytes(_processorCount, (UInt32)sizeof(PerCpuRecord), out UInt64 tableBytes)) { _status = KernelSmpStatus.StateAllocationFailed; return false; }
61        if (!KernelHeap.TryAllocate(tableBytes, 64UL, true, out KernelHeapAllocation stateAllocation)) { _status = KernelSmpStatus.StateAllocationFailed; return false; }
62        _records = (PerCpuRecord*)stateAllocation.Address;
63        if (!PopulateProcessorRecords()) { _status = KernelSmpStatus.NoProcessors; return false; }
64        UInt32 currentApicId = Native.GetCurrentApicId();
65        if (!FindProcessor(currentApicId, out _bootstrapIndex)) { _status = KernelSmpStatus.BootstrapProcessorNotFound; return false; }
66        PerCpuRecord* bsp = _records + _bootstrapIndex;
67        bsp->Flags |= 2U; bsp->StartupState = (UInt32)KernelProcessorStartupState.BootstrapProcessor;
68        _onlineCount = 1U;
69        _trampolineAddress = boot.GetApplicationProcessorTrampolineAddress();
70        UInt64 apicBaseMsr = Native.ReadModelSpecificRegister(ApicBaseMsr);
71        if ((apicBaseMsr & ApicEnabled) != 0UL && (apicBaseMsr & X2ApicEnabled) == 0UL)
72        {
73            _localApicBase = apicBaseMsr & 0x0000000FFFFFF000UL;
74            if (_localApicBase == 0UL) KernelAcpi.TryGetLocalApicAddress(out _localApicBase);
75            _xApicStartup = _localApicBase != 0UL;
76        }
77        _initialized = true;
78        if (_processorCount == 1U) { _status = KernelSmpStatus.Success; return true; }
79        if (!KernelSmpMath.IsValidStartupTrampoline(_trampolineAddress)) { MarkRemainingUnsupported(); _status = KernelSmpStatus.TrampolineUnavailable; return true; }
80        if (!_xApicStartup) { MarkRemainingUnsupported(); _status = KernelSmpStatus.LocalApicUnavailable; return true; }
81        StartApplicationProcessors();
82        _status = _onlineCount == _processorCount ? KernelSmpStatus.Success : KernelSmpStatus.Partial;
83        return true;
84    }

137    private static Boolean PopulateProcessorRecords()
138    {
139        if (_records == null) return false;
140        for (UInt32 index = 0U; index < _processorCount; index++)
141        {
142            if (!KernelAcpi.TryGetProcessor(index, out AcpiProcessorInfo acpi)) return false;
143            PerCpuRecord* record = _records + index;
144            record->Index = index; record->ApicId = acpi.ApicId; record->AcpiUid = acpi.AcpiUid;
145            record->Flags = acpi.IsX2Apic ? 1U : 0U; record->StartupState = (UInt32)KernelProcessorStartupState.Offline;
146            record->Reserved = 0U; record->KernelStackBase = 0UL; record->KernelStackTop = 0UL; record->SchedulerContext = 0UL;
147        }
148        return true;
149    }

151    private static Boolean FindProcessor(UInt32 apicId, out UInt32 index)
152    {
153        index = 0U;
154        if (_records == null) return false;
155        for (UInt32 candidate = 0U; candidate < _processorCount; candidate++) if ((_records + candidate)->ApicId == apicId) { index = candidate; return true; }
156        return false;
157    }

159    private static Boolean StartApplicationProcessors()
160    {
161        if (!KernelSmpMath.TryGetStartupVector(_trampolineAddress, out Byte vector)) return false;
162        UInt64 pageTableRoot = Native.ReadPageTableRoot();
163        if (pageTableRoot == 0UL || pageTableRoot > 0xFFFFFFFFUL) { MarkRemainingUnsupported(); return false; }
164        for (UInt32 index = 0U; index < _processorCount; index++)
165        {
166            if (index == _bootstrapIndex) continue;
167            PerCpuRecord* record = _records + index;
168            if (!KernelSmpMath.IsXApicDestination(record->ApicId)) { record->StartupState = (UInt32)KernelProcessorStartupState.Unsupported; continue; }
169            if (!KernelHeap.TryAllocate(ApplicationProcessorStackBytes, 16UL, true, out KernelHeapAllocation stackAllocation)) { record->StartupState = (UInt32)KernelProcessorStartupState.Failed; continue; }
170            record->KernelStackBase = stackAllocation.Address; record->KernelStackTop = stackAllocation.Address + ApplicationProcessorStackBytes; record->StartupState = (UInt32)KernelProcessorStartupState.Starting;
171            if (!Native.PrepareApplicationProcessorTrampoline(_trampolineAddress, pageTableRoot, record->KernelStackTop)) { record->StartupState = (UInt32)KernelProcessorStartupState.Failed; continue; }
172            if (!SendInitSequence(record->ApicId)) { record->StartupState = (UInt32)KernelProcessorStartupState.Failed; continue; }
173            if (!SendIpi(record->ApicId, StartupIpi | vector)) { record->StartupState = (UInt32)KernelProcessorStartupState.Failed; continue; }
174            if (!KernelTime.DelayNanoseconds(StartupDelayNanoseconds)) { record->StartupState = (UInt32)KernelProcessorStartupState.Failed; continue; }
175            if (Native.GetApplicationProcessorStartupStatus(_trampolineAddress) == 0U)
176            {
177                if (!SendIpi(record->ApicId, StartupIpi | vector)) { record->StartupState = (UInt32)KernelProcessorStartupState.Failed; continue; }
178            }
179            if (!WaitForStartup(record->ApicId)) { record->StartupState = (UInt32)KernelProcessorStartupState.Failed; continue; }
180            record->StartupState = (UInt32)KernelProcessorStartupState.OnlineParked; _onlineCount++;
181        }
182        return true;
183    }

185    private static Boolean SendInitSequence(UInt32 apicId)
186    {
187        if (!SendIpi(apicId, InitAssert)) return false;
188        if (!KernelTime.DelayNanoseconds(InitDelayNanoseconds)) return false;
189        if (!SendIpi(apicId, InitDeassert)) return false;
190        return KernelTime.DelayNanoseconds(StartupDelayNanoseconds);
191    }

193    private static Boolean SendIpi(UInt32 apicId, UInt32 command)
194    {
195        if (!WaitForIcrIdle()) return false;
196        if (!Native.WriteMmio32(_localApicBase + LocalApicIcrHigh, apicId << 24)) return false;
197        if (!Native.WriteMmio32(_localApicBase + LocalApicIcrLow, command)) return false;
198        return WaitForIcrIdle();
199    }
200
201    private static Boolean WaitForIcrIdle()
202    {
203        if (!KernelTime.TryCreateDeadline(StartupTimeoutNanoseconds, out UInt64 deadline)) return false;
204        while ((Native.ReadMmio32(_localApicBase + LocalApicIcrLow) & DeliveryPending) != 0U) if (KernelTime.HasReached(deadline)) return false;
205        return true;
206    }
207
208    private static Boolean WaitForStartup(UInt32 expectedApicId)
209    {
210        if (!KernelTime.TryCreateDeadline(StartupTimeoutNanoseconds, out UInt64 deadline)) return false;
211        while (Native.GetApplicationProcessorStartupStatus(_trampolineAddress) == 0U) if (KernelTime.HasReached(deadline)) return false;
212        return Native.GetApplicationProcessorObservedApicId(_trampolineAddress) == expectedApicId;
213    }

215    private static Boolean MarkRemainingUnsupported()
216    {
217        if (_records == null) return false;
218        for (UInt32 index = 0U; index < _processorCount; index++) if (index != _bootstrapIndex && (_records + index)->StartupState == (UInt32)KernelProcessorStartupState.Offline) (_records + index)->StartupState = (UInt32)KernelProcessorStartupState.Unsupported;
219        return true;
220    }
