5/// <summary>Provides stable encoding and validation rules for NovaOryn syscall ABI namespaces.</summary>
6public static class KernelSystemCallMath
7{
8    public const UInt64 NamespaceMask = 0xFFFF000000000000UL;
9    public const UInt64 GetSetEventNamespace = 0x4E4F000000000000UL;
10    public const UInt64 LinuxNamespace = 0x4C58000000000000UL;
11    public const UInt64 NtNamespace = 0x4E54000000000000UL;
12    public const UInt64 ServiceMask = 0x00000000FFFFFFFFUL;
13    public const UInt32 MaximumRegisteredService = 63U;
14
15    /// <summary>Encodes one NovaOryn Get/Set/Event operation for the x64 SYSCALL entry.</summary>
16    public static UInt64 EncodeGetSetEvent(KernelSystemCallOperation operation, UInt32 service)
17    {
18        return GetSetEventNamespace | ((UInt64)operation << 32) | service;
19    }
20
21    /// <summary>Encodes one Linux-style syscall number without changing the original numeric ID.</summary>
22    public static UInt64 EncodeLinux(UInt32 syscallNumber) => LinuxNamespace | syscallNumber;
23
24    /// <summary>Encodes one NT-style service number without assuming a Windows-version-specific table.</summary>
25    public static UInt64 EncodeNt(UInt32 serviceNumber) => NtNamespace | serviceNumber;
26
27    /// <summary>Attempts to decode the ABI namespace carried by an encoded x64 syscall number.</summary>
28    public static Boolean TryDecodeAbi(UInt64 encoded, out KernelSystemCallAbi abi)
29    {
30        UInt64 ns = encoded & NamespaceMask;
31        if (ns == GetSetEventNamespace) { abi = KernelSystemCallAbi.GetSetEvent; return true; }
32        if (ns == LinuxNamespace) { abi = KernelSystemCallAbi.Linux; return true; }
33        if (ns == NtNamespace) { abi = KernelSystemCallAbi.Nt; return true; }
34        abi = KernelSystemCallAbi.Unknown;
35        return false;
36    }
37
38    /// <summary>Gets the 32-bit service number carried by any supported syscall namespace.</summary>
39    public static UInt32 GetServiceNumber(UInt64 encoded) => (UInt32)(encoded & ServiceMask);
40
41    /// <summary>Gets the Get/Set/Event operation class encoded above the service ID.</summary>
42    public static KernelSystemCallOperation GetOperation(UInt64 encoded) => (KernelSystemCallOperation)((encoded >> 32) & 0xFFUL);
43
44    /// <summary>Determines whether a custom service ID fits the bounded freestanding registry.</summary>
45    public static Boolean IsRegistrableService(UInt32 service) => service <= MaximumRegisteredService;
46}
