35    /// <summary>Initializes the x64 SYSCALL/SYSRET boundary for the currently executing bootstrap processor.</summary>
36    public static Boolean Initialize()
37    {
38        if (_initialized) return true;
39        if (!KernelProtection.IsInitialized() || !KernelHeap.IsInitialized() || !KernelSmp.IsInitialized()) return false;
40        if (!KernelHeap.TryAllocate(SyscallStackBytes, 16UL, true, out KernelHeapAllocation stack)) return false;
41        if (!KernelHeap.TryAllocate(128UL, 16UL, true, out KernelHeapAllocation state)) return false;
42        _stackBase = stack.Address;
43        _stackTop = stack.Address + stack.ByteCount;
44        _stateAddress = state.Address;
45        if (!Native.ConfigureSystemCalls(_stateAddress, _stackTop)) return false;
46        _configuredProcessors = 1U;
47        KernelProtectionCapabilities protection = KernelProtection.GetCapabilities();
48        if (protection.SmapSupported)
49        {
50            if (!Native.EnableSmap()) return false;
51            _smapEnabled = Native.IsSmapEnabled();
52            if (!_smapEnabled) return false;
53        }
54        _initialized = true;
55        return true;
56    }

74    /// <summary>Copies bytes from a validated readable user range while honoring SMAP when enabled.</summary>
75    public static Boolean TryCopyFromUser(UInt64 userSource, UInt64 kernelDestination, UInt64 byteCount)
76    {
77        if (!_initialized || kernelDestination == 0UL || !ValidateUserRange(userSource, byteCount, false)) return false;
78        if (!Native.BeginUserMemoryAccess()) return false;
79        Byte* src=(Byte*)userSource; Byte* dst=(Byte*)kernelDestination;
80        for (UInt64 i=0UL;i<byteCount;i++) dst[i]=src[i];
81        return Native.EndUserMemoryAccess();
82    }

95    public static Int64 Dispatch(UInt64 encoded, UInt64 a0, UInt64 a1, UInt64 a2, UInt64 a3, UInt64 a4, UInt64 a5)
96    {
97        if (!_initialized || !KernelSystemCallMath.TryDecodeAbi(encoded, out KernelSystemCallAbi abi)) return (Int64)KernelSystemCallError.NotImplemented;
98        UInt32 service=KernelSystemCallMath.GetServiceNumber(encoded);
99        KernelSystemCallOperation operation=abi==KernelSystemCallAbi.GetSetEvent ? KernelSystemCallMath.GetOperation(encoded) : KernelSystemCallOperation.Get;
100        KernelSystemCallFrame frame=new(abi,operation,service,a0,a1,a2,a3,a4,a5);
101        if (abi==KernelSystemCallAbi.GetSetEvent) return DispatchNative(&frame);
102        if (abi==KernelSystemCallAbi.Linux) return DispatchLinux(&frame);
103        return DispatchNt(&frame);
104    }

109    private static Int64 DispatchNative(KernelSystemCallFrame* frame)
110    {
111        if (frame->Operation==KernelSystemCallOperation.Get)
112        {
113            if (frame->ServiceNumber==0U) return 1L;
114            if (frame->ServiceNumber==1U) return unchecked((Int64)KernelTime.GetMonotonicNanoseconds());
115            if (frame->ServiceNumber==2U) return KernelSmp.GetOnlineProcessorCount();
116        }
117        else if (frame->Operation==KernelSystemCallOperation.Set)
118        {
119            if (frame->ServiceNumber==0U) return KernelScheduler.SetQuantumNanoseconds(frame->Argument0) ? 0L : (Int64)KernelSystemCallError.InvalidArgument;
120        }
121        else if (frame->Operation==KernelSystemCallOperation.Event)
122        {
123            if (frame->ServiceNumber==0U && KernelSmp.TryGetCurrentProcessor(out KernelProcessorState cpu))
124                return KernelScheduler.Yield(cpu.Index,out UInt64 _) ? 0L : (Int64)KernelSystemCallError.Busy;
125        }
126        return DispatchRegistered(frame->Operation,frame);
127    }

129    private static Int64 DispatchLinux(KernelSystemCallFrame* frame)
130    {
131        if (frame->ServiceNumber==24U && KernelSmp.TryGetCurrentProcessor(out KernelProcessorState cpu))
132            return KernelScheduler.Yield(cpu.Index,out UInt64 _) ? 0L : (Int64)KernelSystemCallError.Busy;
133        Int64 custom=DispatchRegisteredAbi(KernelSystemCallAbi.Linux,frame);

143    private static Boolean Register(KernelSystemCallOperation operation, UInt32 service, delegate*<KernelSystemCallFrame*, Int64> handler)
144    {
145        if (handler==null || !KernelSystemCallMath.IsRegistrableService(service)) return false;
146        fixed (UInt64* get=_registry.Get, set=_registry.Set, evt=_registry.Event)
147        {
148            UInt64 value=(UInt64)(void*)handler;
149            if (operation==KernelSystemCallOperation.Get) get[service]=value;
150            else if (operation==KernelSystemCallOperation.Set) set[service]=value;
151            else if (operation==KernelSystemCallOperation.Event) evt[service]=value;
152            else return false;
153        }
154        return true;

170    private static Int64 DispatchRegistered(KernelSystemCallOperation operation, KernelSystemCallFrame* frame)
171    {
172        if (!KernelSystemCallMath.IsRegistrableService(frame->ServiceNumber)) return (Int64)KernelSystemCallError.NotImplemented;
173        UInt64 address=0UL;
174        fixed (UInt64* get=_registry.Get, set=_registry.Set, evt=_registry.Event)
175        {
176            if (operation==KernelSystemCallOperation.Get) address=get[frame->ServiceNumber];
177            else if (operation==KernelSystemCallOperation.Set) address=set[frame->ServiceNumber];
178            else if (operation==KernelSystemCallOperation.Event) address=evt[frame->ServiceNumber];
179        }
180        return Invoke(address,frame,(Int64)KernelSystemCallError.NotImplemented);

201    private static Boolean ValidateUserRange(UInt64 address, UInt64 byteCount, Boolean write)
202    {
203        if (!KernelProtectionMath.IsUserRange(address,byteCount)) return false;
204        UInt64 current=address, remaining=byteCount;
205        while (remaining!=0UL)
206        {
207            if (!KernelVirtualMemory.TryTranslate(current,out KernelVirtualTranslation translation)) return false;
208            KernelVirtualMemoryProtection p=translation.Protection;
209            if ((p & KernelVirtualMemoryProtection.User)==0 || (p & KernelVirtualMemoryProtection.Read)==0) return false;
210            if (write && (p & KernelVirtualMemoryProtection.Write)==0) return false;
211            UInt64 pageSize=(UInt64)translation.PageSize;
212            UInt64 pageRemaining=pageSize-(current & (pageSize-1UL));
213            UInt64 step=remaining<pageRemaining ? remaining : pageRemaining;
214            current+=step; remaining-=step;
215        }
216        return true;
217    }
