5/// <summary>Complete printable Set-1 US and UK keyboard layout translation.</summary>
6public static class KeyboardLayouts

8    /// <summary>Returns the stable display name for one layout.</summary>

11    /// <summary>Gets whether the supplied layout is installed.</summary>

14    /// <summary>Translates one logical key using the selected full layout and modifier state.</summary>
15    public static Char Translate(KeyboardLayout layout, Ps2Key key, Boolean shift, Boolean capsLock, Boolean altGr)
16    {
17        if (!IsInstalled(layout)) return '\0';
18        Boolean upper=shift^capsLock;
19        Char letter=Letter(key);
20        if(letter!='\0') return upper ? (Char)(letter-32) : letter;
21        if (layout==KeyboardLayout.English_UK && altGr && key==Ps2Key.D4) return '€';
22        return layout==KeyboardLayout.English_UK ? TranslateUk(key,shift) : TranslateUs(key,shift);
23    }

29    private static Char TranslateUs(Ps2Key k,Boolean s)
30    {
31        switch(k){

33            case Ps2Key.D1:return s?'!':'1';case Ps2Key.D2:return s?'@':'2';case Ps2Key.D3:return s?'#':'3';case Ps2Key.D4:return s?'$':'4';case Ps2Key.D5:return s?'%':'5';case Ps2Key.D6:return s?'^':'6';case Ps2Key.D7:return s?'&':'7';case Ps2Key.D8:return s?'*':'8';case Ps2Key.D9:return s?'(':'9';case Ps2Key.D0:return s?')':'0';
34            case Ps2Key.Minus:return s?'_':'-';case Ps2Key.Equals:return s?'+':'=';case Ps2Key.LeftBracket:return s?'{':'[';case Ps2Key.RightBracket:return s?'}':']';case Ps2Key.Backslash:return s?'|':'\\';case Ps2Key.Semicolon:return s?':':';';case Ps2Key.Apostrophe:return s?'"':'\'';case Ps2Key.Grave:return s?'~':'`';case Ps2Key.Comma:return s?'<':',';case Ps2Key.Period:return s?'>':'.';case Ps2Key.Slash:return s?'?':'/';
35            case Ps2Key.KeypadMultiply:return '*';case Ps2Key.KeypadMinus:return '-';case Ps2Key.KeypadPlus:return '+';case Ps2Key.KeypadDivide:return '/';case Ps2Key.KeypadEnter:return '\n';case Ps2Key.KeypadDecimal:return '.';case Ps2Key.Keypad0:return '0';case Ps2Key.Keypad1:return '1';case Ps2Key.Keypad2:return '2';case Ps2Key.Keypad3:return '3';case Ps2Key.Keypad4:return '4';case Ps2Key.Keypad5:return '5';case Ps2Key.Keypad6:return '6';case Ps2Key.Keypad7:return '7';case Ps2Key.Keypad8:return '8';case Ps2Key.Keypad9:return '9';default:return '\0';}
