8/// <summary>Establishes and validates the x64 ring-0/ring-3 protection boundary.</summary>
9public static class KernelProtection
10{
11    private static Boolean _initialized, _writeProtect, _executeDisable, _smepSupported, _smepEnabled, _smapSupported;
12
13    /// <summary>Initializes supervisor write protection and available safe execution protections.</summary>
14    public static Boolean Initialize()
15    {
16        if (_initialized) return true;
17        if (!KernelVirtualMemory.IsInitialized() || !KernelScheduler.IsInitialized()) return false;
18        if (!Native.EnableKernelWriteProtect()) return false;
19        _writeProtect = Native.IsKernelWriteProtectEnabled();
20        _executeDisable = Native.IsExecuteDisableEnabled();
21        if (!_writeProtect || !_executeDisable) return false;
22        _smepSupported = Native.SupportsSmep();
23        _smapSupported = Native.SupportsSmap();
24        _smepEnabled = !_smepSupported || Native.EnableSmep();
25        if (!_smepEnabled) return false;
26        _initialized = true;
27        return true;
28    }
29
30    public static Boolean IsInitialized() => _initialized;
31
32    /// <summary>Gets the current privilege-separation capabilities.</summary>
33    public static KernelProtectionCapabilities GetCapabilities() => new(_writeProtect, _executeDisable, _smepSupported, _smepSupported && _smepEnabled, _smapSupported);
34
35    /// <summary>Validates and maps one user-accessible page while refusing kernel-half addresses.</summary>
36    public static Boolean TryMapUserPage(UInt64 virtualAddress, UInt64 physicalAddress, KernelVirtualMemoryProtection protection)
37    {
38        if (!_initialized || !KernelProtectionMath.IsUserRange(virtualAddress, 4096UL)) return false;
39        KernelVirtualMemoryProtection userProtection = protection | KernelVirtualMemoryProtection.User | KernelVirtualMemoryProtection.Read;
40        return KernelVirtualMemory.TryMap(virtualAddress, physicalAddress, KernelVirtualPageSize.Page4KiB, userProtection);
41    }
42
43    /// <summary>Changes one existing user mapping without allowing supervisor-only protection.</summary>
44    public static Boolean TryProtectUserPage(UInt64 virtualAddress, KernelVirtualMemoryProtection protection)
45    {
46        if (!_initialized || !KernelProtectionMath.IsUserRange(virtualAddress, 4096UL)) return false;
47        KernelVirtualMemoryProtection userProtection = protection | KernelVirtualMemoryProtection.User | KernelVirtualMemoryProtection.Read;
48        return KernelVirtualMemory.TryProtect(virtualAddress, userProtection);
49    }
50
51    /// <summary>Creates a validated future ring-3 transition context for the process/syscall stages.</summary>
52    public static Boolean TryCreateUserModeContext(UInt64 entryPoint, UInt64 stackTop, UInt64 argument, out UserModeContext context)
53    {
54        context=default;
55        if (!_initialized || !KernelProtectionMath.IsValidUserEntry(entryPoint) || !KernelProtectionMath.IsValidUserStack(stackTop)) return false;
56        context = new UserModeContext(entryPoint, stackTop, argument);
57        return true;
58    }
59}
