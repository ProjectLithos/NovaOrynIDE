5/// <summary>Provides architecture-policy validation for the x64 user/kernel boundary.</summary>
6public static class KernelProtectionMath
7{
8    public const UInt64 MinimumUserAddress = 0x0000000000010000UL;
9    public const UInt64 MaximumUserAddress = 0x00007FFFFFFFFFFFUL;
10    public const UInt16 UserDataSelector = 0x001B;
11    public const UInt16 UserCodeSelector = 0x0023;

13    /// <summary>Determines whether an address belongs to the canonical lower-half user range.</summary>
14    public static Boolean IsUserAddress(UInt64 address) => address >= MinimumUserAddress && address <= MaximumUserAddress;
15
16    /// <summary>Determines whether a non-empty byte range is wholly contained in user space.</summary>
17    public static Boolean IsUserRange(UInt64 start, UInt64 byteCount)
18    {
19        if (byteCount == 0UL || !IsUserAddress(start)) return false;
20        UInt64 lastOffset = byteCount - 1UL;
21        if (start > UInt64.MaxValue - lastOffset) return false;
22        return IsUserAddress(start + lastOffset);
23    }
24
25    /// <summary>Determines whether an initial ring-3 stack pointer is canonical and Win64 ABI aligned.</summary>
26    public static Boolean IsValidUserStack(UInt64 stackTop) => IsUserAddress(stackTop) && (stackTop & 0xFUL) == 0UL;
27
28    /// <summary>Determines whether a ring-3 instruction pointer is canonical and outside the null guard.</summary>
29    public static Boolean IsValidUserEntry(UInt64 entryPoint) => IsUserAddress(entryPoint);
30}
