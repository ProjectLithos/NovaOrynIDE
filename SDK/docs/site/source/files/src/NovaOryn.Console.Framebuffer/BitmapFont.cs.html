52    internal static bool TryGetGlyphRow(char value, uint row, out byte bits)
53    {
54        if (row >= GlyphHeight)
55        {
56            bits = 0;
57            return false;
58        }
59
60        bits = GetGlyphRow(value, row);
61        return true;
62    }

64    internal static byte GetGlyphRow(char value, uint row)
65    {
66        if (row >= GlyphHeight) return 0;
67        if (value < (char)0x20 || value > (char)0x7E) value = (char)0x3F;
68
69        ulong packed = row < 8U ? GetTopHalf((byte)value) : GetBottomHalf((byte)value);
70        uint rowInHalf = row < 8U ? row : row - 8U;
71        uint shift = (7U - rowInHalf) * 8U;
72        return (byte)((packed >> (int)shift) & 0xFFUL);
73    }
