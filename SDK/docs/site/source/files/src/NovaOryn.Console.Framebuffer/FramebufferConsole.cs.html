92/// <summary>Writes text directly to a UEFI-provided linear framebuffer.</summary>
93public sealed unsafe class FramebufferConsole : IConsole
94{
95    private FramebufferConfiguration _configuration;
96    private BootFramebuffer _framebuffer;
97    private uint _cursorX;
98    private uint _cursorY;
99    private uint _glyphWidth;
100    private uint _characterAdvance;
101    private uint _lineHeight;
102    private uint _foreground;
103    private uint _background;
104    private bool _initialized;
105
106    /// <summary>Gets the exact glyph height currently configured for rendering, in framebuffer pixels.</summary>
107    /// <nova.when>Use after configuration to report or verify the renderer's active font size.</nova.when>
108    /// <nova.depends>FramebufferConfiguration.FontSize.</nova.depends>
109    public uint FontSize => _configuration.FontSize;
110
111    /// <summary>Applies console colours, font size, and margin.</summary>
112    /// <param name="configuration">The configuration to apply.</param>
113    /// <returns><see langword="true"/> when the configuration is valid.</returns>
114    public bool Configure(FramebufferConfiguration configuration)
115    {
116        if (configuration.FontSize < BitmapFont.MinimumFontSize ||
117            configuration.FontSize > BitmapFont.MaximumFontSize)
118        {
119            throw new ArgumentOutOfRangeException(nameof(configuration));
120        }
121
122        _configuration = configuration;
123        return true;
124    }
125
126    /// <summary>Initializes the console from a validated boot context.</summary>
127    /// <param name="boot">The boot context containing framebuffer information.</param>
128    /// <returns><see langword="true"/> when initialization succeeds.</returns>
129    public bool Initialize(BootContext boot)
130    {
131        if (!boot.TryGetFramebuffer(out BootFramebuffer framebuffer)) return false;
132        if (_configuration.FontSize == 0 && !Configure(FramebufferConfiguration.Default())) return false;
133
134        if (BitmapFont.GetFontContractVersion() != 2U) return false;
135
136        uint fontSize = _configuration.FontSize;
137        uint glyphWidth = BitmapFont.GetRenderedGlyphWidth(fontSize);
138        uint characterAdvance = BitmapFont.GetRenderedCharacterAdvance(fontSize);
139        uint lineHeight = BitmapFont.GetRenderedLineHeight(fontSize);
140        if (glyphWidth == 0 || characterAdvance < glyphWidth || lineHeight < fontSize) return false;
141        if (_configuration.Margin >= framebuffer.Width || _configuration.Margin >= framebuffer.Height) return false;
142        if (glyphWidth > framebuffer.Width - _configuration.Margin) return false;
143        if (fontSize > framebuffer.Height - _configuration.Margin) return false;
144
145        _framebuffer = framebuffer;
146        _cursorX = _configuration.Margin;
147        _cursorY = _configuration.Margin;
148        _glyphWidth = glyphWidth;
149        _characterAdvance = characterAdvance;
150        _lineHeight = lineHeight;
151        _foreground = PackColor(_configuration.ForegroundRed, _configuration.ForegroundGreen, _configuration.ForegroundBlue);
152        _background = PackColor(_configuration.BackgroundRed, _configuration.BackgroundGreen, _configuration.BackgroundBlue);
153        _initialized = true;
154        return Clear();
155    }

157    /// <summary>Clears the visible framebuffer and resets the cursor.</summary>
158    /// <returns><see langword="true"/> when the framebuffer is cleared.</returns>
159    public bool Clear()
160    {
161        if (!_initialized) return false;
162        ulong pixelCount = checked((ulong)_framebuffer.PixelsPerScanLine * _framebuffer.Height);
163        if (pixelCount > _framebuffer.SizeInBytes / 4UL) return false;
164        uint* pixel = (uint*)_framebuffer.Address.Value;
165        while (pixelCount != 0)
166        {
167            *pixel++ = _background;
168            pixelCount--;
169        }
170        _cursorX = _configuration.Margin;
171        _cursorY = _configuration.Margin;
172        return true;
173    }
174
175    /// <summary>Writes text at the current framebuffer cursor.</summary>

180        if (!_initialized) return false;
181        foreach (char character in text)
182        {
183            if (!WriteCharacter(character)) return false;

197    private bool WriteCharacter(char value)
198    {
199        if (value == '\r') return true;
200        if (value == '\n') return MoveToNextLine();
201        if (_cursorX > _framebuffer.Width - _glyphWidth && !MoveToNextLine()) return false;
202        if (_cursorY > _framebuffer.Height - _configuration.FontSize) return false;
203        if (!DrawGlyph(value, _cursorX, _cursorY)) return false;
204        _cursorX += _characterAdvance;
205        return true;
206    }
207
208    private bool MoveToNextLine()
209    {
210        _cursorX = _configuration.Margin;
211        uint lastLineY = _framebuffer.Height - _configuration.FontSize;
212        if (_cursorY <= lastLineY && _lineHeight <= lastLineY - _cursorY)
213        {
214            _cursorY += _lineHeight;
215            return true;
216        }
217        return ScrollUpOneLine();
218    }
219
220    private bool ScrollUpOneLine()
221    {
222        if (_lineHeight == 0 || _configuration.Margin >= _framebuffer.Height ||
223            _lineHeight >= _framebuffer.Height - _configuration.Margin) return false;
224
225        uint sourceY = _configuration.Margin + _lineHeight;
226        uint destinationY = _configuration.Margin;
227        uint rowsToMove = _framebuffer.Height - sourceY;
228        ulong framebufferPixels = _framebuffer.SizeInBytes / 4UL;
229        uint* pixels = (uint*)_framebuffer.Address.Value;
230
231        for (uint row = 0; row < rowsToMove; row++)
232        {
233            ulong sourceIndex = checked((ulong)(sourceY + row) * _framebuffer.PixelsPerScanLine);
234            ulong destinationIndex = checked((ulong)(destinationY + row) * _framebuffer.PixelsPerScanLine);
235            if (sourceIndex + _framebuffer.PixelsPerScanLine > framebufferPixels ||
236                destinationIndex + _framebuffer.PixelsPerScanLine > framebufferPixels) return false;
237
238            for (uint column = 0; column < _framebuffer.PixelsPerScanLine; column++)
239                pixels[destinationIndex + column] = pixels[sourceIndex + column];
240        }
241
242        uint clearStartY = _framebuffer.Height - _lineHeight;
243        for (uint row = clearStartY; row < _framebuffer.Height; row++)
244        {
245            ulong destinationIndex = checked((ulong)row * _framebuffer.PixelsPerScanLine);
246            if (destinationIndex + _framebuffer.PixelsPerScanLine > framebufferPixels) return false;
247            for (uint column = 0; column < _framebuffer.PixelsPerScanLine; column++)
248                pixels[destinationIndex + column] = _background;
249        }
250
251        return _cursorY <= _framebuffer.Height - _configuration.FontSize;
252    }
253
254    private bool DrawGlyph(char value, uint originX, uint originY)
255    {
256        uint fontSize = _configuration.FontSize;
257        for (uint renderedRow = 0; renderedRow < fontSize; renderedRow++)
258        {
259            uint sourceRow = BitmapFont.GetSourceRow(renderedRow, fontSize);
260            if (!BitmapFont.TryGetGlyphRow(value, sourceRow, out byte sourceBits)) return false;
261            uint bits = sourceBits;
262            for (uint renderedColumn = 0; renderedColumn < _glyphWidth; renderedColumn++)
263            {
264                uint sourceColumn = BitmapFont.GetSourceColumn(renderedColumn, _glyphWidth);
265                uint mask = 1U << (int)((BitmapFont.GlyphWidth - 1U) - sourceColumn);
266                if ((bits & mask) == 0) continue;
267                if (!DrawPixel(originX + renderedColumn, originY + renderedRow)) return false;
268            }
269        }
270        return true;
271    }
272
273    private bool DrawPixel(uint pixelX, uint pixelY)
274    {
275        if (pixelX >= _framebuffer.Width || pixelY >= _framebuffer.Height) return false;
276        ulong index = checked(((ulong)pixelY * _framebuffer.PixelsPerScanLine) + pixelX);
277        if (index >= _framebuffer.SizeInBytes / 4UL) return false;
278        *((uint*)_framebuffer.Address.Value + index) = _foreground;
279        return true;
280    }

282    private uint PackColor(byte red, byte green, byte blue)
283    {
284        return _framebuffer.PixelFormat switch
285        {
286            FramebufferPixelFormat.RedGreenBlueReserved8BitPerColor => red | ((uint)green << 8) | ((uint)blue << 16),
287            FramebufferPixelFormat.BlueGreenRedReserved8BitPerColor => blue | ((uint)green << 8) | ((uint)red << 16),
288            FramebufferPixelFormat.BitMask => EncodeMask(red, _framebuffer.PixelMask.Red) |
289                                              EncodeMask(green, _framebuffer.PixelMask.Green) |
290                                              EncodeMask(blue, _framebuffer.PixelMask.Blue),
291            _ => 0U
292        };
293    }
294
295    private static uint EncodeMask(byte component, uint mask)
296    {
297        int shift = System.Numerics.BitOperations.TrailingZeroCount(mask);
298        int bits = System.Numerics.BitOperations.PopCount(mask);
299        ulong maximum = bits == 32 ? uint.MaxValue : ((1UL << bits) - 1UL);
300        return ((uint)(((ulong)component * maximum) / byte.MaxValue) << shift) & mask;
301    }
