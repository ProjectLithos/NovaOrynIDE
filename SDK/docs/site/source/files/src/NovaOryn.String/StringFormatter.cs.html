11    /// <summary>Formats an unsigned 64-bit integer into a caller-owned character buffer.</summary>
12    public static unsafe Boolean TryFormat(UInt64 value, Char* destination, UInt32 capacity, out UInt32 written)
13    {
14        written = 0U;
15        if (destination == null || capacity == 0U) return false;
16        Char* reverse = stackalloc Char[20];
17        UInt32 count = 0U;
18        do
19        {
20            reverse[count++] = (Char)('0' + (Char)(value % 10UL));
21            value /= 10UL;
22        }
23        while (value != 0UL);
24        if (count > capacity) return false;
25        UInt32 index = 0U;
26        while (index < count)
27        {
28            destination[index] = reverse[count - index - 1U];
29            index++;
30        }
31        written = count;
32        return true;
33    }

35    /// <summary>Formats a signed 64-bit integer into a caller-owned character buffer.</summary>
36    public static unsafe Boolean TryFormat(Int64 value, Char* destination, UInt32 capacity, out UInt32 written)
37    {
38        written = 0U;
39        if (destination == null || capacity == 0U) return false;
40        Boolean negative = value < 0L;
41        UInt64 magnitude = negative ? (UInt64)(-(value + 1L)) + 1UL : (UInt64)value;
42        UInt32 offset = negative ? 1U : 0U;
43        if (offset >= capacity) return false;
44        UInt32 digits;
45        if (!TryFormat(magnitude, destination + offset, capacity - offset, out digits)) return false;
46        if (negative) destination[0] = '-';
47        written = digits + offset;
48        return true;
49    }
