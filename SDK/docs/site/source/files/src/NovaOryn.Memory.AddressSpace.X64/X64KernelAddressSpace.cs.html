6/// <summary>Defines and validates the standard NovaOryn four-level x64 kernel address-space policy.</summary>
7/// <nova.when>Use as the default x64 layout or as a reference when creating a custom layout.</nova.when>
8/// <nova.depends>Four-level x64 canonical addressing and 4 KiB pages.</nova.depends>
9[SupportedArchitecture(SupportedArchitecture.X64)]
10[BootStage(BootStage.ManagedBootstrap)]
11public static class X64KernelAddressSpace
12{
13    /// <summary>Gets the first user virtual byte; the first 64 KiB are intentionally left unmapped as a null/low-address guard.</summary>
14    public const ulong UserBase = 0x0000000000010000UL;
15    /// <summary>Gets the exclusive end of the low canonical user half.</summary>
16    public const ulong UserEndExclusive = 0x0000800000000000UL;
17    /// <summary>Gets the target base of the higher-half kernel image window.</summary>
18    public const ulong KernelImageBase = 0xFFFF800000000000UL;
19    /// <summary>Gets the reserved kernel-image window length (4 GiB).</summary>
20    public const ulong KernelImageLength = 0x0000000100000000UL;
21    /// <summary>Gets the base of the future kernel heap reservation.</summary>
22    public const ulong KernelHeapBase = 0xFFFF810000000000UL;
23    /// <summary>Gets the kernel heap reservation length (1 TiB).</summary>
24    public const ulong KernelHeapLength = 0x0000010000000000UL;
25    /// <summary>Gets the base of the kernel stack/guard-page arena.</summary>
26    public const ulong KernelStacksBase = 0xFFFF820000000000UL;
27    /// <summary>Gets the kernel stack arena length (1 TiB).</summary>
28    public const ulong KernelStacksLength = 0x0000010000000000UL;
29    /// <summary>Gets the base of the direct physical-memory map.</summary>
30    public const ulong DirectMapBase = 0xFFFF900000000000UL;
31    /// <summary>Gets the direct-map capacity (64 TiB of physical address space).</summary>
32    public const ulong DirectMapLength = 0x0000400000000000UL;
33    /// <summary>Gets the base of the kernel MMIO window.</summary>
34    public const ulong MmioBase = 0xFFFFD00000000000UL;
35    /// <summary>Gets the MMIO window length (16 TiB).</summary>
36    public const ulong MmioLength = 0x0000100000000000UL;
37    /// <summary>Gets the base of the dedicated page-table access window.</summary>
38    public const ulong PageTableWindowBase = 0xFFFFFF0000000000UL;
39    /// <summary>Gets the page-table access window length (512 GiB).</summary>
40    public const ulong PageTableWindowLength = 0x0000008000000000UL;

42    /// <summary>Creates the standard NovaOryn x64 kernel layout.</summary>
43    /// <nova.when>Use during x64 kernel bootstrap before heap, stack, MMIO, or direct-map consumers are enabled.</nova.when>
44    /// <nova.depends>Canonical four-level x64 address ranges.</nova.depends>
45    /// <returns><see langword="true"/> when all standard regions were created and validated.</returns>
46    /// <example><code>bool ok = X64KernelAddressSpace.TryCreateStandard(out KernelAddressSpaceLayout layout);</code></example>
47    public static bool TryCreateStandard(out KernelAddressSpaceLayout layout)
48    {
49        layout = default;
50        if (!KernelAddressSpaceRegion.TryCreate(KernelAddressSpaceRegionKind.User, UserBase, UserEndExclusive - UserBase, out var user)) return false;
51        if (!KernelAddressSpaceRegion.TryCreate(KernelAddressSpaceRegionKind.KernelImage, KernelImageBase, KernelImageLength, out var image)) return false;
52        if (!KernelAddressSpaceRegion.TryCreate(KernelAddressSpaceRegionKind.KernelHeap, KernelHeapBase, KernelHeapLength, out var heap)) return false;
53        if (!KernelAddressSpaceRegion.TryCreate(KernelAddressSpaceRegionKind.KernelStacks, KernelStacksBase, KernelStacksLength, out var stacks)) return false;
54        if (!KernelAddressSpaceRegion.TryCreate(KernelAddressSpaceRegionKind.DirectPhysicalMap, DirectMapBase, DirectMapLength, out var direct)) return false;
55        if (!KernelAddressSpaceRegion.TryCreate(KernelAddressSpaceRegionKind.Mmio, MmioBase, MmioLength, out var mmio)) return false;
56        if (!KernelAddressSpaceRegion.TryCreate(KernelAddressSpaceRegionKind.PageTableWindow, PageTableWindowBase, PageTableWindowLength, out var tables)) return false;
57        if (!KernelAddressSpaceLayout.TryCreate(user, image, heap, stacks, direct, mmio, tables, out layout)) return false;
58        return Validate(layout);
59    }
60
61    /// <summary>Validates canonicality, high/low-half placement, separation, and page alignment for an x64 kernel layout.</summary>
62    /// <nova.when>Use before accepting a custom x64 address-space policy.</nova.when>
63    /// <nova.depends>A structurally valid <see cref="KernelAddressSpaceLayout"/>.</nova.depends>
64    /// <returns><see langword="true"/> when every mapped endpoint is canonical and the user/kernel split is respected.</returns>
65    /// <example><code>bool valid = X64KernelAddressSpace.Validate(layout);</code></example>
66    public static bool Validate(KernelAddressSpaceLayout layout)
67    {
68        if (layout.User.BaseAddress < UserBase || layout.User.EndExclusive > UserEndExclusive) return false;
69        foreach (var region in new[] { layout.KernelImage, layout.KernelHeap, layout.KernelStacks, layout.DirectPhysicalMap, layout.Mmio, layout.PageTableWindow })
70        {
71            if (region.BaseAddress < 0xFFFF800000000000UL || !X64VirtualAddress.IsCanonical(region.BaseAddress) || !X64VirtualAddress.IsCanonical(region.EndExclusive - 1UL)) return false;
72        }
73        return true;
74    }

76    /// <summary>Attempts to translate a direct-map virtual address back to its physical byte address.</summary>
77    /// <nova.when>Use for diagnostics or physical-page ownership calculations within the direct-map window.</nova.when>
78    /// <nova.depends>The standard direct-map base and capacity.</nova.depends>
79    /// <returns><see langword="true"/> when the virtual address belongs to the direct-map window.</returns>
80    /// <example><code>bool ok = X64KernelAddressSpace.TryDirectMapToPhysical(X64KernelAddressSpace.DirectMapBase + 0x2000, out ulong physical);</code></example>
81    public static bool TryDirectMapToPhysical(ulong virtualAddress, out ulong physicalAddress)
82    { physicalAddress = 0; if (virtualAddress < DirectMapBase || virtualAddress >= DirectMapBase + DirectMapLength) return false; physicalAddress = virtualAddress - DirectMapBase; return true; }
83
84    /// <summary>Attempts to convert a physical byte address into the standard direct-map virtual address.</summary>
85    /// <nova.when>Use when a kernel subsystem needs stable virtual access to a physical byte inside the direct-map capacity.</nova.when>
86    /// <nova.depends>Physical address below <see cref="DirectMapLength"/>.</nova.depends>
87    /// <returns><see langword="true"/> when the physical address fits the configured direct-map capacity.</returns>
88    /// <example><code>bool ok = X64KernelAddressSpace.TryPhysicalToDirectMap(0x2000, out ulong virtualAddress);</code></example>
89    public static bool TryPhysicalToDirectMap(ulong physicalAddress, out ulong virtualAddress)
90    { virtualAddress = 0; if (physicalAddress >= DirectMapLength) return false; virtualAddress = DirectMapBase + physicalAddress; return true; }
91}
