75/// <summary>
76/// Provides the default no-heap physical-frame manager used by the editable freestanding kernel template.
77/// </summary>
78/// <remarks>
79/// The early manager consumes only UEFI ConventionalMemory. BootServicesCode and BootServicesData remain
80/// deferred even after ExitBootServices because firmware-provided bootstrap state, including the inherited
81/// stack, can still occupy those pages. Loader, runtime, ACPI, MMIO, framebuffer and defective ranges also
82/// remain unavailable. Metadata lives in fixed kernel storage, so initialization never depends on a heap.
83/// </remarks>
84public static unsafe partial class KernelPhysicalMemory
85{

87    private const Int32 MaximumExtents = 512;
88    private const Int32 MaximumAllocations = 256;
89    private const UInt32 UefiConventionalMemory = 7U;
90    private const UInt64 UefiRuntimeAttribute = 0x8000000000000000UL;

123    /// <summary>Initializes the default extent manager directly from the retained final UEFI memory map.</summary>
124    /// <returns><see langword="true"/> when every immediately allocatable firmware range was accepted.</returns>
125    public static Boolean Initialize(BootContext boot)
126    {
127        if (_initialized) return SetFailure(KernelPhysicalMemoryStatus.AlreadyInitialized);
128        if (!boot.HasFinalMemoryMap()) return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
129
130        UInt64 mapAddress = boot.GetFinalMemoryMapAddress();
131        UInt64 mapLength = boot.GetFinalMemoryMapLength();
132        UInt64 descriptorSize = boot.GetFinalMemoryDescriptorSize();
133        if (mapAddress == 0UL || mapLength == 0UL || descriptorSize < 40UL || mapLength % descriptorSize != 0UL)
134            return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
135
136        UInt64 descriptorCount = mapLength / descriptorSize;
137        if (descriptorCount == 0UL || descriptorCount > 0x7FFFFFFFUL)
138            return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
139
140        UInt64 bootstrapAddress = boot.GetBootstrapPageTableWorkspaceAddress();
141        UInt64 bootstrapPages = boot.GetBootstrapPageTableWorkspacePages();
142        if (bootstrapAddress == 0UL || bootstrapPages == 0UL || (bootstrapAddress & 0xFFFUL) != 0UL ||
143            bootstrapPages > 0xFFFFFFFFFFFFFFFFUL / PageSize || bootstrapAddress > 0xFFFFFFFFFFFFFFFFUL - bootstrapPages * PageSize)
144            return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
145
146        ResetState();
147        _bootstrapWorkspaceAddress = bootstrapAddress;
148        _bootstrapWorkspacePages = bootstrapPages;
149        for (UInt64 index = 0; index < descriptorCount; index++)
150        {
151            UInt64 offset = index * descriptorSize;
152            if (mapAddress > 0xFFFFFFFFFFFFFFFFUL - offset) return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
153            Byte* descriptor = (Byte*)(nuint)(mapAddress + offset);
154            UInt32 type = *(UInt32*)(descriptor + 0);
155            UInt64 start = *(UInt64*)(descriptor + 8);
156            UInt64 pages = *(UInt64*)(descriptor + 24);
157            UInt64 attributes = *(UInt64*)(descriptor + 32);
158            if (pages == 0UL || (start & 0xFFFUL) != 0UL || pages > 0xFFFFFFFFFFFFFFFFUL / PageSize)
159                return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
160            if ((attributes & UefiRuntimeAttribute) != 0UL) continue;
161            if (type != UefiConventionalMemory) continue;
162            if (!InsertFreeExtent(start / PageSize, pages)) return SetFailure(KernelPhysicalMemoryStatus.ExtentCapacityExhausted);
163            if (_managedPages > 0xFFFFFFFFFFFFFFFFUL - pages) return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
164            _managedPages += pages;
165            _freePages += pages;
166        }
167
168        if (_managedPages == 0UL) return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
169        _initialized = true;
170        _lastStatus = KernelPhysicalMemoryStatus.Success;
171        return true;
172    }

174    /// <summary>Allocates contiguous 4 KiB physical frames using first-fit extent selection.</summary>
175    /// <param name="pageCount">Number of contiguous pages required.</param>
176    /// <param name="alignmentPages">Power-of-two page alignment; use one for ordinary page alignment.</param>
177    /// <param name="allocation">Receives the live allocation token and physical range.</param>
178    /// <returns><see langword="true"/> when a matching free extent was split successfully.</returns>
179    public static Boolean TryAllocate(UInt64 pageCount, UInt64 alignmentPages, out KernelPhysicalAllocation allocation)
180    {
181        allocation = default;
182        if (!_initialized) return SetFailure(KernelPhysicalMemoryStatus.NotInitialized);
183        if (pageCount == 0UL || alignmentPages == 0UL || (alignmentPages & (alignmentPages - 1UL)) != 0UL)
184            return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
185        Int32 record = FindFreeAllocationRecord();
186        if (record < 0) return SetFailure(KernelPhysicalMemoryStatus.AllocationCapacityExhausted);
187
188        fixed (UInt64* starts = _state.ExtentStarts)
189        fixed (UInt64* pages = _state.ExtentPages)
190        fixed (UInt64* tokens = _state.AllocationTokens)
191        fixed (UInt64* allocationStarts = _state.AllocationStarts)
192        fixed (UInt64* allocationPages = _state.AllocationPages)
193        fixed (Byte* active = _state.AllocationActive)
194        {
195            for (Int32 index = 0; index < _extentCount; index++)
196            {
197                UInt64 extentStart = starts[index];
198                UInt64 extentPages = pages[index];
199                if (!TryAlignFrame(extentStart, alignmentPages, out UInt64 candidate)) continue;
200                if (candidate < extentStart) continue;
201                UInt64 prefix = candidate - extentStart;
202                if (prefix > extentPages || pageCount > extentPages - prefix) continue;
203
204                UInt64 suffix = extentPages - prefix - pageCount;
205                if (prefix != 0UL && suffix != 0UL && _extentCount >= MaximumExtents)
206                    return SetFailure(KernelPhysicalMemoryStatus.ExtentCapacityExhausted);
207                ReplaceAllocatedExtent(index, extentStart, prefix, candidate + pageCount, suffix);
208
209                UInt64 token = NextToken();
210                tokens[record] = token;
211                allocationStarts[record] = candidate;
212                allocationPages[record] = pageCount;
213                active[record] = 1;
214                _liveAllocationCount++;
215                _freePages -= pageCount;
216                _allocatedPages += pageCount;
217                allocation = new KernelPhysicalAllocation(token, candidate * PageSize, pageCount);
218                _lastStatus = KernelPhysicalMemoryStatus.Success;
219                return true;
220            }
221        }
222        return SetFailure(KernelPhysicalMemoryStatus.OutOfMemory);
223    }

225    /// <summary>Permanently excludes one 4 KiB physical page from ordinary early allocation when it is currently free.</summary>
226    /// <param name="physicalAddress">The 4 KiB-aligned physical page address to protect.</param>
227    /// <returns><see langword="true"/> when the page is excluded or was already unavailable to this manager.</returns>
228    public static Boolean TryExcludePage(UInt64 physicalAddress)
229    {
230        if (!_initialized) return SetFailure(KernelPhysicalMemoryStatus.NotInitialized);
231        if ((physicalAddress & 0xFFFUL) != 0UL) return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
232        UInt64 frame = physicalAddress / PageSize;
233        fixed (UInt64* starts = _state.ExtentStarts)
234        fixed (UInt64* pages = _state.ExtentPages)
235        {
236            for (Int32 index = 0; index < _extentCount; index++)
237            {
238                UInt64 start = starts[index];
239                UInt64 count = pages[index];
240                if (frame < start || frame - start >= count) continue;
241                UInt64 prefix = frame - start;
242                UInt64 suffix = count - prefix - 1UL;
243                if (prefix != 0UL && suffix != 0UL && _extentCount >= MaximumExtents)
244                    return SetFailure(KernelPhysicalMemoryStatus.ExtentCapacityExhausted);
245                if (!ReplaceAllocatedExtent(index, start, prefix, frame + 1UL, suffix))
246                    return SetFailure(KernelPhysicalMemoryStatus.ExtentCapacityExhausted);
247                _freePages--;
248                _lastStatus = KernelPhysicalMemoryStatus.Success;
249                return true;
250            }
251        }
252        _lastStatus = KernelPhysicalMemoryStatus.Success;
253        return true;
254    }

257    /// <summary>Releases a live physical allocation exactly once and coalesces adjacent free extents.</summary>
258    /// <returns><see langword="true"/> when the token identifies an active allocation.</returns>
259    public static Boolean TryRelease(KernelPhysicalAllocation allocation)
260    {
261        if (!_initialized) return SetFailure(KernelPhysicalMemoryStatus.NotInitialized);
262        fixed (UInt64* tokens = _state.AllocationTokens)
263        fixed (UInt64* starts = _state.AllocationStarts)
264        fixed (UInt64* pages = _state.AllocationPages)
265        fixed (Byte* active = _state.AllocationActive)
266        {
267            for (Int32 index = 0; index < MaximumAllocations; index++)
268            {
269                if (active[index] == 0 || tokens[index] != allocation.Token || starts[index] * PageSize != allocation.StartAddress || pages[index] != allocation.PageCount) continue;
270                if (!InsertFreeExtent(starts[index], pages[index])) return SetFailure(KernelPhysicalMemoryStatus.ExtentCapacityExhausted);
271                active[index] = 0;
272                _liveAllocationCount--;
273                _freePages += pages[index];
274                _allocatedPages -= pages[index];
275                _lastStatus = KernelPhysicalMemoryStatus.Success;
276                return true;
277            }
278        }
279        return SetFailure(KernelPhysicalMemoryStatus.AllocationNotFound);
280    }

294    private static Boolean InsertFreeExtent(UInt64 startFrame, UInt64 pageCount)
295    {
296        if (pageCount == 0UL || startFrame > 0xFFFFFFFFFFFFFFFFUL - pageCount) return false;
297        fixed (UInt64* starts = _state.ExtentStarts)
298        fixed (UInt64* pages = _state.ExtentPages)
299        {
300            Int32 insert = 0;
301            while (insert < _extentCount && starts[insert] < startFrame) insert++;
302            if (_extentCount >= MaximumExtents) return false;
303            for (Int32 move = _extentCount; move > insert; move--)
304            {
305                starts[move] = starts[move - 1];
306                pages[move] = pages[move - 1];
307            }
308            starts[insert] = startFrame;
309            pages[insert] = pageCount;
310            _extentCount++;
311            return CoalesceAround(insert);
312        }
313    }

315    private static Boolean CoalesceAround(Int32 index)
316    {
317        fixed (UInt64* starts = _state.ExtentStarts)
318        fixed (UInt64* pages = _state.ExtentPages)
319        {
320            if (index > 0)
321            {
322                UInt64 previousEnd = starts[index - 1] + pages[index - 1];
323                if (previousEnd > starts[index]) return false;
324                if (previousEnd == starts[index])
325                {
326                    pages[index - 1] += pages[index];
327                    RemoveExtent(index);
328                    index--;
329                }
330            }
331            if (index + 1 < _extentCount)
332            {
333                UInt64 end = starts[index] + pages[index];
334                if (end > starts[index + 1]) return false;
335                if (end == starts[index + 1])
336                {
337                    pages[index] += pages[index + 1];
338                    RemoveExtent(index + 1);
339                }
340            }
341        }
342        return true;
343    }

345    private static Boolean ReplaceAllocatedExtent(Int32 index, UInt64 originalStart, UInt64 prefixPages, UInt64 suffixStart, UInt64 suffixPages)
346    {
347        fixed (UInt64* starts = _state.ExtentStarts)
348        fixed (UInt64* pages = _state.ExtentPages)
349        {
350            if (prefixPages == 0UL && suffixPages == 0UL) return RemoveExtent(index);
351            if (prefixPages == 0UL)
352            {
353                starts[index] = suffixStart;
354                pages[index] = suffixPages;
355                return true;
356            }
357            starts[index] = originalStart;
358            pages[index] = prefixPages;
359            if (suffixPages == 0UL) return true;
360            for (Int32 move = _extentCount; move > index + 1; move--)
361            {
362                starts[move] = starts[move - 1];
363                pages[move] = pages[move - 1];
364            }
365            starts[index + 1] = suffixStart;
366            pages[index + 1] = suffixPages;
367            _extentCount++;
368            return true;
369        }
370    }

372    private static Boolean RemoveExtent(Int32 index)
373    {
374        fixed (UInt64* starts = _state.ExtentStarts)
375        fixed (UInt64* pages = _state.ExtentPages)
376        {
377            for (Int32 move = index; move + 1 < _extentCount; move++)
378            {
379                starts[move] = starts[move + 1];
380                pages[move] = pages[move + 1];
381            }
382            _extentCount--;
383            if (_extentCount >= 0)
384            {
385                starts[_extentCount] = 0UL;
386                pages[_extentCount] = 0UL;
387            }
388        }
389        return true;
390    }

401    private static Boolean TryAlignFrame(UInt64 frame, UInt64 alignmentPages, out UInt64 aligned)
402    {
403        aligned = 0UL;
404        UInt64 mask = alignmentPages - 1UL;
405        if (frame > 0xFFFFFFFFFFFFFFFFUL - mask) return false;
406        aligned = (frame + mask) & ~mask;
407        return true;
408    }
