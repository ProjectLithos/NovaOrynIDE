21    {
22        physicalAddress = 0UL;
23        if (!_initialized) return SetFailure(KernelPhysicalMemoryStatus.NotInitialized);
24        if (_bootstrapWorkspaceUsedPages >= _bootstrapWorkspacePages) return SetFailure(KernelPhysicalMemoryStatus.OutOfMemory);
25        UInt64 offset = _bootstrapWorkspaceUsedPages * PageSize;
26        if (_bootstrapWorkspaceAddress > 0xFFFFFFFFFFFFFFFFUL - offset) return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
27        physicalAddress = _bootstrapWorkspaceAddress + offset;
28        _bootstrapWorkspaceUsedPages++;
29        _lastStatus = KernelPhysicalMemoryStatus.Success;
30        return true;
31    }

56    /// <summary>Allocates an exact free physical range selected by an early bootstrap consumer.</summary>
57    /// <param name="physicalAddress">4 KiB-aligned physical start address.</param>
58    /// <param name="pageCount">Number of contiguous pages to claim.</param>
59    /// <param name="allocation">Receives the live allocation token and range.</param>
60    /// <returns><see langword="true"/> when the exact range was free and could be split from its extent.</returns>
61    public static Boolean TryAllocateAt(UInt64 physicalAddress, UInt64 pageCount, out KernelPhysicalAllocation allocation)
62    {
63        allocation = default;
64        if (!_initialized) return SetFailure(KernelPhysicalMemoryStatus.NotInitialized);
65        if (pageCount == 0UL || (physicalAddress & 0xFFFUL) != 0UL) return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
66        UInt64 startFrame = physicalAddress / PageSize;
67        if (startFrame > 0xFFFFFFFFFFFFFFFFUL - pageCount) return SetFailure(KernelPhysicalMemoryStatus.InvalidParameter);
68        Int32 record = FindFreeAllocationRecord();
69        if (record < 0) return SetFailure(KernelPhysicalMemoryStatus.AllocationCapacityExhausted);
70        fixed (UInt64* starts = _state.ExtentStarts)
71        fixed (UInt64* pages = _state.ExtentPages)
72        fixed (UInt64* tokens = _state.AllocationTokens)
73        fixed (UInt64* allocationStarts = _state.AllocationStarts)
74        fixed (UInt64* allocationPages = _state.AllocationPages)
75        fixed (Byte* active = _state.AllocationActive)
76        {
77            for (Int32 index = 0; index < _extentCount; index++)
78            {
79                UInt64 extentStart = starts[index];
80                UInt64 extentPages = pages[index];
81                if (startFrame < extentStart) continue;
82                UInt64 prefix = startFrame - extentStart;
83                if (prefix > extentPages || pageCount > extentPages - prefix) continue;
84                UInt64 suffix = extentPages - prefix - pageCount;
85                if (prefix != 0UL && suffix != 0UL && _extentCount >= MaximumExtents)
86                    return SetFailure(KernelPhysicalMemoryStatus.ExtentCapacityExhausted);
87                if (!ReplaceAllocatedExtent(index, extentStart, prefix, startFrame + pageCount, suffix))
88                    return SetFailure(KernelPhysicalMemoryStatus.ExtentCapacityExhausted);
89                UInt64 token = NextToken();
90                tokens[record] = token;
91                allocationStarts[record] = startFrame;
92                allocationPages[record] = pageCount;
93                active[record] = 1;
94                _liveAllocationCount++;
95                _freePages -= pageCount;
96                _allocatedPages += pageCount;
97                allocation = new KernelPhysicalAllocation(token, physicalAddress, pageCount);
98                _lastStatus = KernelPhysicalMemoryStatus.Success;
99                return true;
100            }
101        }
102        return SetFailure(KernelPhysicalMemoryStatus.OutOfMemory);
103    }
