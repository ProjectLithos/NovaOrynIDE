32/// <summary>Provides x64 port-mapped input/output operations.</summary>
33/// <nova.when>Use only for hardware that exposes legacy x64 I/O ports.</nova.when>
34/// <nova.depends>NovaOryn x64 native port-I/O entry points</nova.depends>
35public static class Port
36{
37    /// <summary>Writes one byte to an x64 I/O port.</summary>

39    /// <param name="value">The byte to write.</param>

44    /// <summary>Attempts to read one byte from an x64 I/O port.</summary>
45    /// <param name="port">The source I/O port.</param>
