7/// <summary>Defines the result returned by an interrupt handler.</summary>
8public enum InterruptResult : byte
9{
10    /// <summary>The handler did not claim the vector.</summary>
11    Unhandled = 0,
12    /// <summary>The interrupt was handled and execution may resume.</summary>
13    Handled = 1,
14    /// <summary>The interrupt was handled and scheduling should occur.</summary>
15    Reschedule = 2,
16    /// <summary>The processor must stop because execution cannot safely resume.</summary>

98/// <summary>Provides the stable managed/native x64 interrupt frame.</summary>
99[StructLayout(LayoutKind.Sequential, Pack = 8)]
100public struct InterruptContext
101{
102    /// <summary>Gets or sets the vector number.</summary>
103    public ulong Vector;
104    /// <summary>Gets or sets the normalised architectural error code.</summary>
105    public ulong ErrorCode;
106    /// <summary>Gets or sets the interrupted RIP.</summary>
107    public ulong InstructionPointer;
108    /// <summary>Gets or sets the interrupted CS.</summary>
109    public ulong CodeSegment;
110    /// <summary>Gets or sets the interrupted RFLAGS.</summary>
111    public ulong Flags;
112    /// <summary>Gets or sets the interrupted RSP.</summary>
113    public ulong StackPointer;
114    /// <summary>Gets or sets the interrupted SS.</summary>
115    public ulong StackSegment;
116    /// <summary>Gets or sets the captured CR0.</summary>
117    public ulong ControlRegister0;
118    /// <summary>Gets or sets the captured CR2 fault address.</summary>
119    public ulong ControlRegister2;
120    /// <summary>Gets or sets the captured CR3.</summary>
121    public ulong ControlRegister3;
122    /// <summary>Gets or sets the captured CR4.</summary>
123    public ulong ControlRegister4;
124    /// <summary>Gets or sets the current processor identifier.</summary>
125    public ulong ProcessorId;
126    /// <summary>Gets or sets the one when hardware changed privilege levels.</summary>
127    public ulong PrivilegeTransition;
128    /// <summary>Gets or sets the RAX.</summary>
129    public ulong Rax;
130    /// <summary>Gets or sets the RBX.</summary>
131    public ulong Rbx;
132    /// <summary>Gets or sets the RCX.</summary>
133    public ulong Rcx;
134    /// <summary>Gets or sets the RDX.</summary>
135    public ulong Rdx;
136    /// <summary>Gets or sets the RSI.</summary>
137    public ulong Rsi;
138    /// <summary>Gets or sets the RDI.</summary>
139    public ulong Rdi;
140    /// <summary>Gets or sets the RBP.</summary>
141    public ulong Rbp;
142    /// <summary>Gets or sets the R8.</summary>
143    public ulong R8;
144    /// <summary>Gets or sets the R9.</summary>
145    public ulong R9;
146    /// <summary>Gets or sets the R10.</summary>
147    public ulong R10;
148    /// <summary>Gets or sets the R11.</summary>
149    public ulong R11;
150    /// <summary>Gets or sets the R12.</summary>
151    public ulong R12;
152    /// <summary>Gets or sets the R13.</summary>
153    public ulong R13;
154    /// <summary>Gets or sets the R14.</summary>
155    public ulong R14;
156    /// <summary>Gets or sets the R15.</summary>
157    public ulong R15;
158
159    /// <summary>Gets whether the interrupt crossed a privilege boundary.</summary>
160    public readonly bool HasPrivilegeTransition() => (PrivilegeTransition & 1) != 0;
161
162    /// <summary>Gets whether hardware switched stacks because of privilege or IST policy.</summary>
163    public readonly bool HasStackSwitch() => (PrivilegeTransition & 2) != 0;
164
165    /// <summary>Gets whether this frame describes a page fault.</summary>
166    public readonly bool IsPageFault() => Vector == (byte)CpuExceptionVector.PageFault;
167}
