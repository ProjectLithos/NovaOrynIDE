6/// <summary>Creates one of NovaOryn's supported memory-map normalisation implementations.</summary>
7/// <nova.when>Use when selecting strict, safety-priority, or conservative firmware-overlap handling.</nova.when>
8/// <nova.depends>IMemoryMapNormaliser</nova.depends>
9public static class MemoryMapNormaliserFactory
10{
11    /// <summary>Creates the requested stateless normaliser.</summary>
12    /// <nova.when>Use during boot-policy selection or SDK configuration.</nova.when>
13    /// <nova.depends>MemoryMapNormalisationMethod</nova.depends>
14    /// <returns>A normaliser implementing the selected overlap policy.</returns>
15    /// <example><code>IMemoryMapNormaliser normaliser = MemoryMapNormaliserFactory.Create(MemoryMapNormalisationMethod.SafetyPriority);</code></example>
16    public static IMemoryMapNormaliser Create(MemoryMapNormalisationMethod method)
17    {
18        return method switch
19        {
20            MemoryMapNormalisationMethod.Strict => new StrictMemoryMapNormaliser(),
21            MemoryMapNormalisationMethod.SafetyPriority => new SafetyPriorityMemoryMapNormaliser(),
22            MemoryMapNormalisationMethod.Conservative => new ConservativeMemoryMapNormaliser(),
23            _ => new StrictMemoryMapNormaliser()
24        };
25    }

84    internal static MemoryMapNormalisationResult Normalise(
85        IMemoryMapSource source,
86        MemoryReservation[] reservations,
87        MemoryMapNormalisationWorkspace workspace,
88        MemoryMapNormalisationMethod method,
89        out NormalisedMemoryMap? map)
90    {
91        map = null;
92        if (source is null || reservations is null || workspace is null || source.Count < 1)
93            return Result(MemoryMapNormalisationStatus.InvalidInput, source?.Count ?? 0, reservations?.Length ?? 0);
94        if (!source.IsFinal)
95            return Result(MemoryMapNormalisationStatus.FinalMapRequired, source.Count, reservations.Length);
96
97        int sourceCount = source.Count;
98        if (sourceCount > workspace.Inputs.Length || reservations.Length > workspace.Inputs.Length - sourceCount)
99            return Result(MemoryMapNormalisationStatus.InsufficientCapacity, sourceCount, reservations.Length);
100        int totalCount = sourceCount + reservations.Length;
101        if (totalCount > workspace.Boundaries.Length / 2)
102            return Result(MemoryMapNormalisationStatus.InsufficientCapacity, sourceCount, reservations.Length);
103
104        for (int index = 0; index < sourceCount; index++)
105        {
106            if (!source.TryGetDescriptor(index, out MemoryDescriptor descriptor))
107                return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
108            workspace.Inputs[index] = descriptor;
109        }
110        for (int index = 0; index < reservations.Length; index++)
111            workspace.Inputs[sourceCount + index] = reservations[index].Descriptor;
112
113        int boundaryCount = 0;
114        for (int index = 0; index < totalCount; index++)
115        {
116            MemoryDescriptor descriptor = workspace.Inputs[index];
117            if (descriptor.Length == 0 || descriptor.EndExclusive <= descriptor.PhysicalStart.Value)
118                return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
119            workspace.Boundaries[boundaryCount++] = descriptor.PhysicalStart.Value;
120            workspace.Boundaries[boundaryCount++] = descriptor.EndExclusive;
121        }
122        SortAndDeduplicate(workspace.Boundaries, ref boundaryCount);
123
124        int outputCount = 0;
125        int splitCount = 0;
126        int mergeCount = 0;
127        int resolvedOverlapCount = 0;
128        for (int boundaryIndex = 0; boundaryIndex + 1 < boundaryCount; boundaryIndex++)
129        {
130            ulong start = workspace.Boundaries[boundaryIndex];
131            ulong end = workspace.Boundaries[boundaryIndex + 1];
132            if (start == end) continue;
133
134            int sourceSelection = -1;
135            bool sourceConflict = false;
136            int sourceCoverage = 0;
137            for (int index = 0; index < sourceCount; index++)
138            {
139                MemoryDescriptor candidate = workspace.Inputs[index];
140                if (!Covers(candidate, start, end)) continue;
141                sourceCoverage++;
142                if (sourceSelection < 0)
143                {
144                    sourceSelection = index;
145                    continue;
146                }
147                MemoryDescriptor selected = workspace.Inputs[sourceSelection];
148                if (!SameMetadata(selected, candidate)) sourceConflict = true;
149                if (IsSafer(candidate, selected)) sourceSelection = index;
150            }
151
152            int reservationSelection = -1;
153            for (int index = sourceCount; index < totalCount; index++)
154            {
155                MemoryDescriptor reservation = workspace.Inputs[index];
156                if (!Covers(reservation, start, end)) continue;
157                if (reservationSelection < 0 || IsSafer(reservation, workspace.Inputs[reservationSelection]))
158                    reservationSelection = index;
159            }
160
161            if (sourceSelection < 0 && reservationSelection < 0) continue;
162            if (sourceCoverage > 1) resolvedOverlapCount++;
163            if (sourceConflict && method == MemoryMapNormalisationMethod.Strict)
164                return new MemoryMapNormalisationResult(MemoryMapNormalisationStatus.OverlapConflict, sourceCount, reservations.Length, outputCount, splitCount, mergeCount, resolvedOverlapCount);
165
166            MemoryDescriptor selectedDescriptor;
167            if (sourceSelection < 0)
168            {
169                if (!workspace.Inputs[reservationSelection].TrySlice(start, end, out selectedDescriptor))
170                    return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
171            }
172            else if (sourceConflict && method == MemoryMapNormalisationMethod.Conservative)
173            {
174                if (!TryCreateConservativeInterval(workspace.Inputs, sourceCount, start, end, out selectedDescriptor))
175                    return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
176            }
177            else if (sourceConflict && method == MemoryMapNormalisationMethod.SafetyPriority)
178            {
179                if (!TryCreateSafetyInterval(workspace.Inputs, sourceCount, sourceSelection, start, end, out selectedDescriptor))
180                    return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
181            }
182            else if (!workspace.Inputs[sourceSelection].TrySlice(start, end, out selectedDescriptor))
183            {
184                return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
185            }
186
187            if (reservationSelection >= 0 && !TryCreateReservationInterval(workspace.Inputs, sourceCount, reservationSelection, start, end, out selectedDescriptor))
188                return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
189
190            splitCount++;
191            if (outputCount > 0 && workspace.Outputs[outputCount - 1].IsMergeCompatible(selectedDescriptor))
192            {
193                MemoryDescriptor previous = workspace.Outputs[outputCount - 1];
194                if (previous.PageCount > ulong.MaxValue - selectedDescriptor.PageCount)
195                    return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
196                ulong pages = previous.PageCount + selectedDescriptor.PageCount;
197                if (!MemoryDescriptor.TryCreate(previous.PhysicalStart, pages, previous.MemoryType, previous.CacheAttributes, previous.RuntimeStatus, previous.Availability, previous.HasNumaNode, previous.NumaNode, out MemoryDescriptor merged))
198                    return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
199                workspace.Outputs[outputCount - 1] = merged;
200                mergeCount++;
201                continue;
202            }
203            if (outputCount >= workspace.Outputs.Length)
204                return new MemoryMapNormalisationResult(MemoryMapNormalisationStatus.InsufficientCapacity, sourceCount, reservations.Length, outputCount, splitCount, mergeCount, resolvedOverlapCount);
205            workspace.Outputs[outputCount++] = selectedDescriptor;
206        }
207
208        if (outputCount == 0)
209            return Result(MemoryMapNormalisationStatus.InvalidDescriptor, sourceCount, reservations.Length);
210        map = new NormalisedMemoryMap(workspace.Outputs, outputCount, method);
211        return new MemoryMapNormalisationResult(MemoryMapNormalisationStatus.Success, sourceCount, reservations.Length, outputCount, splitCount, mergeCount, resolvedOverlapCount);
212    }

214    private static bool TryCreateReservationInterval(
215        MemoryDescriptor[] descriptors,
216        int sourceCount,
217        int reservationIndex,
218        ulong start,
219        ulong end,
220        out MemoryDescriptor descriptor)
221    {
222        MemoryDescriptor reservation = descriptors[reservationIndex];
223        MemoryRuntimeStatus runtimeStatus = CombineRuntimeStatus(descriptors, sourceCount, start, end);
224        MemoryCacheAttributes attributes = CombineReservationAttributes(
225            CombineAttributes(descriptors, sourceCount, start, end),
226            reservation.CacheAttributes);
227        bool hasNumaNode = TryGetConsistentNumaNode(descriptors, sourceCount, start, end, out uint numaNode);
228        return MemoryDescriptor.TryCreate(
229            new PhysicalAddress(start),
230            (end - start) / 4096UL,
231            reservation.MemoryType,
232            attributes,
233            runtimeStatus,
234            runtimeStatus == MemoryRuntimeStatus.NotRuntime ? reservation.Availability : MemoryAvailability.RuntimeOwned,
235            hasNumaNode,
236            numaNode,
237            out descriptor);
238    }

289    private static MemoryCacheAttributes CombineAttributes(
290        MemoryDescriptor[] descriptors,
291        int sourceCount,
292        ulong start,
293        ulong end)
294    {
295        const MemoryCacheAttributes cacheMask =
296            MemoryCacheAttributes.Uncacheable |
297            MemoryCacheAttributes.WriteCombining |
298            MemoryCacheAttributes.WriteThrough |
299            MemoryCacheAttributes.WriteBack |
300            MemoryCacheAttributes.UncachedExported;
301        MemoryCacheAttributes nonCacheAttributes = MemoryCacheAttributes.None;
302        bool uncacheable = false;
303        bool uncachedExported = false;
304        bool writeThrough = false;
305        bool writeCombining = false;
306        bool writeBack = false;
307        for (int index = 0; index < sourceCount; index++)
308        {
309            MemoryDescriptor candidate = descriptors[index];
310            if (!Covers(candidate, start, end)) continue;
311            MemoryCacheAttributes attributes = candidate.CacheAttributes;
312            nonCacheAttributes |= attributes & ~cacheMask;
313            uncacheable |= (attributes & MemoryCacheAttributes.Uncacheable) != 0;
314            uncachedExported |= (attributes & MemoryCacheAttributes.UncachedExported) != 0;
315            writeThrough |= (attributes & MemoryCacheAttributes.WriteThrough) != 0;
316            writeCombining |= (attributes & MemoryCacheAttributes.WriteCombining) != 0;
317            writeBack |= (attributes & MemoryCacheAttributes.WriteBack) != 0;
318        }
319        MemoryCacheAttributes cacheAttribute = uncacheable ? MemoryCacheAttributes.Uncacheable
320            : uncachedExported ? MemoryCacheAttributes.UncachedExported
321            : writeThrough ? MemoryCacheAttributes.WriteThrough
322            : writeCombining ? MemoryCacheAttributes.WriteCombining
323            : writeBack ? MemoryCacheAttributes.WriteBack
324            : MemoryCacheAttributes.None;
325        return nonCacheAttributes | cacheAttribute;
326    }

328    private static MemoryCacheAttributes CombineReservationAttributes(
329        MemoryCacheAttributes sourceAttributes,
330        MemoryCacheAttributes reservationAttributes)
331    {
332        const MemoryCacheAttributes cacheMask =
333            MemoryCacheAttributes.Uncacheable |
334            MemoryCacheAttributes.WriteCombining |
335            MemoryCacheAttributes.WriteThrough |
336            MemoryCacheAttributes.WriteBack |
337            MemoryCacheAttributes.UncachedExported;
338        MemoryCacheAttributes combined = sourceAttributes | reservationAttributes;
339        MemoryCacheAttributes nonCacheAttributes = combined & ~cacheMask;
340        MemoryCacheAttributes cacheAttribute = (combined & MemoryCacheAttributes.Uncacheable) != 0 ? MemoryCacheAttributes.Uncacheable
341            : (combined & MemoryCacheAttributes.UncachedExported) != 0 ? MemoryCacheAttributes.UncachedExported
342            : (combined & MemoryCacheAttributes.WriteThrough) != 0 ? MemoryCacheAttributes.WriteThrough
343            : (combined & MemoryCacheAttributes.WriteCombining) != 0 ? MemoryCacheAttributes.WriteCombining
344            : (combined & MemoryCacheAttributes.WriteBack) != 0 ? MemoryCacheAttributes.WriteBack
345            : MemoryCacheAttributes.None;
346        return nonCacheAttributes | cacheAttribute;
347    }

349    private static bool TryGetConsistentNumaNode(
350        MemoryDescriptor[] descriptors,
351        int sourceCount,
352        ulong start,
353        ulong end,
354        out uint numaNode)
355    {
356        numaNode = 0;
357        bool found = false;
358        for (int index = 0; index < sourceCount; index++)
359        {
360            MemoryDescriptor candidate = descriptors[index];
361            if (!Covers(candidate, start, end)) continue;
362            if (!candidate.HasNumaNode) return false;
363            if (!found)
364            {
365                numaNode = candidate.NumaNode;
366                found = true;
367            }
368            else if (candidate.NumaNode != numaNode)
369            {
370                numaNode = 0;
371                return false;
372            }
373        }
374        return found;
375    }

377    private static MemoryRuntimeStatus CombineRuntimeStatus(
378        MemoryDescriptor[] descriptors,
379        int sourceCount,
380        ulong start,
381        ulong end)
382    {
383        bool hasCode = false;
384        bool hasData = false;
385        for (int index = 0; index < sourceCount; index++)
386        {
387            MemoryDescriptor candidate = descriptors[index];
388            if (!Covers(candidate, start, end)) continue;
389            hasCode |= candidate.RuntimeStatus is MemoryRuntimeStatus.RuntimeCode or MemoryRuntimeStatus.RuntimeCodeAndData;
390            hasData |= candidate.RuntimeStatus is MemoryRuntimeStatus.RuntimeData or MemoryRuntimeStatus.RuntimeCodeAndData;
391        }
392        if (hasCode && hasData) return MemoryRuntimeStatus.RuntimeCodeAndData;
393        if (hasCode) return MemoryRuntimeStatus.RuntimeCode;
394        if (hasData) return MemoryRuntimeStatus.RuntimeData;
395        return MemoryRuntimeStatus.NotRuntime;
396    }

406    private static bool IsSafer(MemoryDescriptor candidate, MemoryDescriptor selected)
407    {
408        int candidateType = GetPriority(candidate.MemoryType);
409        int selectedType = GetPriority(selected.MemoryType);
410        if (candidateType != selectedType) return candidateType > selectedType;
411        int candidateAvailability = GetAvailabilityPriority(candidate.Availability);
412        int selectedAvailability = GetAvailabilityPriority(selected.Availability);
413        if (candidateAvailability != selectedAvailability) return candidateAvailability > selectedAvailability;
414        if (candidate.RuntimeStatus != selected.RuntimeStatus) return (int)candidate.RuntimeStatus > (int)selected.RuntimeStatus;
415        if (candidate.CacheAttributes != selected.CacheAttributes) return (ulong)candidate.CacheAttributes > (ulong)selected.CacheAttributes;
416        if (candidate.HasNumaNode != selected.HasNumaNode) return !candidate.HasNumaNode;
417        if (candidate.HasNumaNode && candidate.NumaNode != selected.NumaNode) return candidate.NumaNode < selected.NumaNode;
418        return false;
419    }

457    private static void SortAndDeduplicate(ulong[] values, ref int count)
458    {
459        for (int index = 1; index < count; index++)
460        {
461            ulong value = values[index];
462            int insert = index - 1;
463            while (insert >= 0 && values[insert] > value)
464            {
465                values[insert + 1] = values[insert];
466                insert--;
467            }
468            values[insert + 1] = value;
469        }
470        if (count < 2) return;
471        int write = 1;
472        for (int read = 1; read < count; read++)
473        {
474            if (values[read] == values[write - 1]) continue;
475            values[write++] = values[read];
476        }
477        count = write;
478    }
