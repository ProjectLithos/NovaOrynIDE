7/// <summary>Reads the retained native UEFI descriptor buffer exposed by a final boot context.</summary>
8/// <nova.when>Use during identity-mapped early boot, or pass an explicitly mapped virtual address after paging changes.</nova.when>
9/// <nova.depends>BootContext final-map metadata and a readable mapping of the retained descriptor buffer</nova.depends>
10public sealed unsafe class NativeUefiMemoryMapSource : IMemoryMapSource
11{

13    private readonly ulong _descriptorSize;

15    private NativeUefiMemoryMapSource(

29    /// <summary>Gets whether this source was created from a final accepted UEFI map.</summary>

45    /// <summary>Creates a source while the retained physical buffer remains identity mapped.</summary>
46    /// <nova.when>Use before replacing the firmware page tables.</nova.when>
47    /// <nova.depends>BootContext.MemoryMapAddress must be directly readable</nova.depends>
48    /// <returns><see langword="true"/> when all final-map metadata is valid.</returns>
49    /// <example><code>bool ready = NativeUefiMemoryMapSource.TryCreate(boot, out NativeUefiMemoryMapSource? source);</code></example>
50    public static bool TryCreate(BootContext boot, out NativeUefiMemoryMapSource? source)
51        => TryCreate(boot, boot.MemoryMapAddress.Value, out source);
52
53    /// <summary>Creates a source using the virtual address at which the retained physical buffer is mapped.</summary>
54    /// <nova.when>Use after installing page tables that map the final UEFI buffer at a non-identity address.</nova.when>
55    /// <nova.depends>The complete buffer must be readable at mappedAddress</nova.depends>
56    /// <returns><see langword="true"/> when all final-map metadata and address arithmetic are valid.</returns>
57    /// <example><code>bool ready = NativeUefiMemoryMapSource.TryCreate(boot, mappedAddress, out NativeUefiMemoryMapSource? source);</code></example>
58    public static bool TryCreate(BootContext boot, ulong mappedAddress, out NativeUefiMemoryMapSource? source)
59    {
60        source = null;
61        if (boot.Protocol != BootProtocol.Uefi || !boot.IsFinalMemoryMap) return false;
62        if (mappedAddress == 0 || boot.MemoryMapLength == 0 || boot.MemoryDescriptorSize < 40) return false;
63        if ((boot.MemoryDescriptorSize & 7UL) != 0 || boot.MemoryMapLength % boot.MemoryDescriptorSize != 0) return false;
64        if (mappedAddress > ulong.MaxValue - boot.MemoryMapLength) return false;
65        ulong count = boot.MemoryMapLength / boot.MemoryDescriptorSize;
66        if (count == 0 || count > int.MaxValue) return false;
67        source = new NativeUefiMemoryMapSource(mappedAddress, (int)count, boot.MemoryMapKey, boot.MemoryDescriptorSize, boot.MemoryDescriptorVersion);
68        return true;
69    }
70
71    /// <summary>Attempts to translate one descriptor from the retained native buffer.</summary>
72    /// <nova.when>Used by every memory-map normaliser implementation.</nova.when>
73    /// <nova.depends>The buffer mapping must remain valid and immutable</nova.depends>
74    /// <returns><see langword="true"/> when the index and descriptor are valid.</returns>
75    /// <example><code>bool found = source.TryGetDescriptor(0, out MemoryDescriptor descriptor);</code></example>
76    public bool TryGetDescriptor(int index, out MemoryDescriptor descriptor)
77    {
78        descriptor = default;
79        if (!TryGetUefiDescriptor(index, out UefiMemoryDescriptor source)) return false;
80        return UefiMemoryDescriptorMapper.TryMap(source, out descriptor);
81    }
82
83    /// <summary>Attempts to read one original UEFI descriptor from the retained native buffer.</summary>
84    /// <nova.when>Use for immutable firmware diagnostics before reclaiming or unmapping the buffer.</nova.when>
85    /// <nova.depends>UEFI descriptor layout version one fields at offsets 0 through 39</nova.depends>
86    /// <returns><see langword="true"/> when the index exists and its address is representable.</returns>
87    /// <example><code>bool found = source.TryGetUefiDescriptor(0, out UefiMemoryDescriptor descriptor);</code></example>
88    public bool TryGetUefiDescriptor(int index, out UefiMemoryDescriptor descriptor)
89    {
90        descriptor = default;
91        if ((uint)index >= (uint)Count) return false;
92        ulong offset = (ulong)(uint)index * _descriptorSize;
93        if (_mappedAddress > ulong.MaxValue - offset) return false;
94        byte* address = (byte*)(nuint)(_mappedAddress + offset);
95        UefiMemoryType type = (UefiMemoryType)(*(uint*)(address + 0));
96        ulong physicalStart = *(ulong*)(address + 8);
97        ulong virtualStart = *(ulong*)(address + 16);
98        ulong pageCount = *(ulong*)(address + 24);
99        UefiMemoryAttributes attributes = (UefiMemoryAttributes)(*(ulong*)(address + 32));
100        descriptor = new UefiMemoryDescriptor(type, new PhysicalAddress(physicalStart), virtualStart, pageCount, attributes);
101        return true;
102    }
