106    /// <summary>Validates the core and platform-dependent reservations required for normalisation.</summary>
107    /// <nova.when>Call before passing this plan to a normaliser.</nova.when>
108    /// <nova.depends>Kernel image and boot structures are always required; framebuffer and MMIO are conditional.</nova.depends>
109    /// <returns><see langword="true"/> when every requested reservation category is present.</returns>
110    /// <example><code>bool complete = plan.TryValidateRequiredReservations(framebufferPresent, mmioPresent, out MemoryType missingType);</code></example>
111    public bool TryValidateRequiredReservations(bool framebufferRequired, bool memoryMappedIoRequired, out MemoryType missingType)
112    {
113        if (!_hasKernelImage)
114        {
115            missingType = MemoryType.LoaderKernelImage;
116            return false;
117        }
118        if (!_hasBootStructures)
119        {
120            missingType = MemoryType.BootStructures;
121            return false;
122        }
123        if (framebufferRequired && !_hasFramebuffer)
124        {
125            missingType = MemoryType.Framebuffer;
126            return false;
127        }
128        if (memoryMappedIoRequired && !_hasMemoryMappedIo)
129        {
130            missingType = MemoryType.MemoryMappedIo;
131            return false;
132        }
133        missingType = MemoryType.Unknown;
134        return true;
135    }

137    /// <summary>Returns an independent copy of the configured reservations.</summary>
138    /// <nova.when>Pass the returned array to any normaliser implementation.</nova.when>
139    /// <nova.depends>Count</nova.depends>
140    /// <returns>A copy that callers may retain independently of the plan.</returns>
141    /// <example><code>MemoryReservation[] reservations = plan.ToArray();</code></example>
142    public MemoryReservation[] ToArray()
143    {
144        MemoryReservation[] result = new MemoryReservation[_count];
145        Array.Copy(_reservations, result, _count);
146        return result;
147    }

149    private bool TryAdd(PhysicalAddress start, ulong pages, MemoryType type, MemoryCacheAttributes attributes)
150    {
151        if (_count >= _reservations.Length) return false;
152        if (!MemoryReservation.TryCreate(start, pages, type, attributes, out MemoryReservation reservation)) return false;
153        _reservations[_count++] = reservation;
154        switch (type)
155        {
156            case MemoryType.LoaderKernelImage: _hasKernelImage = true; break;
157            case MemoryType.BootStructures: _hasBootStructures = true; break;
158            case MemoryType.Framebuffer: _hasFramebuffer = true; break;
159            case MemoryType.MemoryMappedIo: _hasMemoryMappedIo = true; break;
160            case MemoryType.PageTables: _hasPageTables = true; break;
161            case MemoryType.EarlyAllocatorAllocations: _hasEarlyAllocations = true; break;
162        }
163        return true;
164    }
