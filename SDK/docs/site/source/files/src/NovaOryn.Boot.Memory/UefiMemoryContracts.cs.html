141/// <summary>Translates UEFI descriptors into architecture-independent NovaOryn descriptors.</summary>
142/// <nova.when>Used by FinalUefiMemoryMapSnapshot when a normaliser reads the retained map.</nova.when>
143/// <nova.depends>NovaOryn.Memory.Contracts</nova.depends>
144public static class UefiMemoryDescriptorMapper
145{
146    /// <summary>Attempts to translate one UEFI descriptor while rejecting overflow and misalignment.</summary>
147    /// <nova.when>Use for every retained firmware descriptor before normalisation.</nova.when>
148    /// <nova.depends>MemoryDescriptor.TryCreate</nova.depends>
149    /// <returns><see langword="true"/> when the UEFI descriptor is valid and recognised.</returns>
150    /// <example><code>bool mapped = UefiMemoryDescriptorMapper.TryMap(uefi, out MemoryDescriptor descriptor);</code></example>
151    public static bool TryMap(UefiMemoryDescriptor source, out MemoryDescriptor descriptor)
152    {
153        MemoryType type;
154        MemoryRuntimeStatus runtime = MemoryRuntimeStatus.NotRuntime;
155        MemoryAvailability availability;
156        switch (source.Type)
157        {
158            case UefiMemoryType.ConventionalMemory:
159                type = MemoryType.UsableConventional;
160                availability = MemoryAvailability.AvailableAfterExitBootServices;
161                break;
162            case UefiMemoryType.LoaderCode:
163            case UefiMemoryType.LoaderData:
164                type = MemoryType.LoaderKernelImage;
165                availability = MemoryAvailability.Unavailable;
166                break;
167            case UefiMemoryType.BootServicesCode:
168            case UefiMemoryType.BootServicesData:
169                type = MemoryType.BootServices;
170                availability = MemoryAvailability.AvailableAfterExitBootServices;
171                break;
172            case UefiMemoryType.RuntimeServicesCode:
173                type = MemoryType.RuntimeServices;
174                runtime = MemoryRuntimeStatus.RuntimeCode;
175                availability = MemoryAvailability.RuntimeOwned;
176                break;
177            case UefiMemoryType.RuntimeServicesData:
178                type = MemoryType.RuntimeServices;
179                runtime = MemoryRuntimeStatus.RuntimeData;
180                availability = MemoryAvailability.RuntimeOwned;
181                break;
182            case UefiMemoryType.AcpiReclaimMemory:
183                type = MemoryType.AcpiReclaimable;
184                availability = MemoryAvailability.ReclaimableAfterAcpiInitialization;
185                break;
186            case UefiMemoryType.AcpiMemoryNvs:
187                type = MemoryType.AcpiNvs;
188                availability = MemoryAvailability.PermanentlyReserved;
189                break;
190            case UefiMemoryType.MemoryMappedIo:
191            case UefiMemoryType.MemoryMappedIoPortSpace:
192                type = MemoryType.MemoryMappedIo;
193                availability = MemoryAvailability.PermanentlyReserved;
194                break;
195            case UefiMemoryType.UnusableMemory:
196            case UefiMemoryType.UnacceptedMemory:
197                type = MemoryType.BadMemory;
198                availability = MemoryAvailability.Defective;
199                break;
200            case UefiMemoryType.PersistentMemory:
201                type = MemoryType.PersistentMemory;
202                availability = MemoryAvailability.Unavailable;
203                break;
204            default:
205                type = MemoryType.FirmwareReserved;
206                availability = MemoryAvailability.PermanentlyReserved;
207                break;
208        }
209
210        if ((source.Attributes & UefiMemoryAttributes.Runtime) != 0 && runtime == MemoryRuntimeStatus.NotRuntime)
211        {
212            runtime = MemoryRuntimeStatus.RuntimeData;
213            availability = MemoryAvailability.RuntimeOwned;
214            if (type != MemoryType.MemoryMappedIo) type = MemoryType.RuntimeServices;
215        }
216
217        MemoryCacheAttributes attributes = TranslateAttributes(source.Attributes);
218        return MemoryDescriptor.TryCreate(source.PhysicalStart, source.PageCount, type, attributes, runtime, availability, false, 0, out descriptor);
219    }
220
221    private static MemoryCacheAttributes TranslateAttributes(UefiMemoryAttributes source)
222    {
223        MemoryCacheAttributes result = MemoryCacheAttributes.None;
224        if ((source & UefiMemoryAttributes.Uncacheable) != 0) result |= MemoryCacheAttributes.Uncacheable;
225        if ((source & UefiMemoryAttributes.WriteCombining) != 0) result |= MemoryCacheAttributes.WriteCombining;
226        if ((source & UefiMemoryAttributes.WriteThrough) != 0) result |= MemoryCacheAttributes.WriteThrough;
227        if ((source & UefiMemoryAttributes.WriteBack) != 0) result |= MemoryCacheAttributes.WriteBack;
228        if ((source & UefiMemoryAttributes.UncachedExported) != 0) result |= MemoryCacheAttributes.UncachedExported;
229        if ((source & UefiMemoryAttributes.WriteProtected) != 0) result |= MemoryCacheAttributes.WriteProtected;
230        if ((source & UefiMemoryAttributes.ReadProtected) != 0) result |= MemoryCacheAttributes.ReadProtected;
231        if ((source & UefiMemoryAttributes.ExecuteProtected) != 0) result |= MemoryCacheAttributes.ExecuteProtected;
232        if ((source & UefiMemoryAttributes.NonVolatile) != 0) result |= MemoryCacheAttributes.NonVolatile;
233        if ((source & UefiMemoryAttributes.MoreReliable) != 0) result |= MemoryCacheAttributes.MoreReliable;
234        if ((source & UefiMemoryAttributes.ReadOnly) != 0) result |= MemoryCacheAttributes.ReadOnly;
235        if ((source & UefiMemoryAttributes.SpecificPurpose) != 0) result |= MemoryCacheAttributes.SpecificPurpose;
236        if ((source & UefiMemoryAttributes.CpuCrypto) != 0) result |= MemoryCacheAttributes.CpuCrypto;
237        return result;
238    }
