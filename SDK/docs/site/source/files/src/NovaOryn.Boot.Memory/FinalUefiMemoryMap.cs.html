5/// <summary>Owns all storage needed before the final GetMemoryMap and ExitBootServices sequence.</summary>
6/// <nova.when>Create with headroom before the final firmware call so no allocation is required between map capture and exit.</nova.when>
7/// <nova.depends>IUefiMemoryMapProvider</nova.depends>
8public sealed class UefiMemoryMapWorkspace
9{
10    private readonly UefiMemoryDescriptor[] _captureDescriptors;
11    private readonly UefiMemoryDescriptor[] _snapshotDescriptors;
12    private readonly FinalUefiMemoryMapSnapshot _snapshot;
13
14    /// <summary>Creates a fixed-capacity final-map workspace.</summary>
15    /// <nova.when>Create before the last firmware allocations and retain through ExitBootServices.</nova.when>
16    /// <nova.depends>Caller-selected UEFI descriptor headroom</nova.depends>
17    /// <returns>A workspace with separate capture and immutable snapshot storage.</returns>
18    /// <example><code>UefiMemoryMapWorkspace workspace = new(256);</code></example>
19    /// <param name="descriptorCapacity">Maximum retained UEFI descriptor count.</param>
20    public UefiMemoryMapWorkspace(int descriptorCapacity)
21    {
22        if (descriptorCapacity < 1) throw new ArgumentOutOfRangeException(nameof(descriptorCapacity));
23        _captureDescriptors = new UefiMemoryDescriptor[descriptorCapacity];
24        _snapshotDescriptors = new UefiMemoryDescriptor[descriptorCapacity];
25        _snapshot = new FinalUefiMemoryMapSnapshot(_snapshotDescriptors);
26    }
27
28    /// <summary>Gets the preallocated descriptor capacity.</summary>
29    /// <nova.when>Use to verify final-map capture headroom before firmware exit.</nova.when>
30    public int DescriptorCapacity => _captureDescriptors.Length;
31
32    internal UefiMemoryDescriptor[] Descriptors => _captureDescriptors;
33    internal FinalUefiMemoryMapSnapshot Snapshot => _snapshot;

36/// <summary>Retains the exact typed UEFI map whose key succeeded in ExitBootServices.</summary>
37/// <nova.when>Use as the sole firmware source for memory-map normalisation.</nova.when>
38/// <nova.depends>FinalUefiMemoryMapAcquirer</nova.depends>
39public sealed class FinalUefiMemoryMapSnapshot : IMemoryMapSource
40{
41    private readonly UefiMemoryDescriptor[] _storage;
42    private int _count;
43
44    internal FinalUefiMemoryMapSnapshot(UefiMemoryDescriptor[] storage) => _storage = storage;
45
46    /// <summary>Gets whether ExitBootServices succeeded with this exact map key.</summary>
47    /// <nova.when>Check before normalisation or diagnostic enumeration.</nova.when>
48    public bool IsFinal { get; private set; }
49    /// <summary>Gets the retained descriptor count.</summary>
50    /// <nova.when>Use to bound immutable final-map diagnostics.</nova.when>
51    public int Count => _count;
52    /// <summary>Gets the map key accepted by ExitBootServices.</summary>
53    /// <nova.when>Use to audit the exact final firmware hand-off.</nova.when>
54    public ulong MapKey { get; private set; }
55    /// <summary>Gets the firmware descriptor version.</summary>
56    /// <nova.when>Use when diagnosing firmware map format compatibility.</nova.when>
57    public uint DescriptorVersion { get; private set; }
58    /// <summary>Gets the number of GetMemoryMap/ExitBootServices attempts.</summary>
59    /// <nova.when>Use to report stale-key retries.</nova.when>
60    public int CaptureAttempts { get; private set; }
61
62    /// <summary>Attempts to translate one retained UEFI descriptor.</summary>
63    /// <nova.when>Used by all normaliser implementations and diagnostic tools.</nova.when>
64    /// <nova.depends>UefiMemoryDescriptorMapper</nova.depends>
65    /// <returns><see langword="true"/> when the index and descriptor are valid.</returns>
66    /// <example><code>bool found = snapshot.TryGetDescriptor(0, out MemoryDescriptor descriptor);</code></example>
67    public bool TryGetDescriptor(int index, out MemoryDescriptor descriptor)
68    {
69        descriptor = default;
70        if (!IsFinal || (uint)index >= (uint)_count) return false;
71        return UefiMemoryDescriptorMapper.TryMap(_storage[index], out descriptor);
72    }
73
74    /// <summary>Attempts to read the unmodified typed UEFI descriptor.</summary>
75    /// <nova.when>Use for firmware diagnostics that need original UEFI type and attribute values.</nova.when>
76    /// <nova.depends>Count</nova.depends>
77    /// <returns><see langword="true"/> when the index exists in the final snapshot.</returns>
78    /// <example><code>bool found = snapshot.TryGetUefiDescriptor(0, out UefiMemoryDescriptor descriptor);</code></example>
79    public bool TryGetUefiDescriptor(int index, out UefiMemoryDescriptor descriptor)
80    {
81        descriptor = default;
82        if (!IsFinal || (uint)index >= (uint)_count) return false;
83        descriptor = _storage[index];
84        return true;
85    }
86
87    internal bool Seal(UefiMemoryDescriptor[] source, int count, ulong mapKey, uint descriptorVersion, int captureAttempts)
88    {
89        if (IsFinal || source is null || count < 1 || count > source.Length || count > _storage.Length) return false;
90        Array.Copy(source, _storage, count);
91        _count = count;
92        MapKey = mapKey;
93        DescriptorVersion = descriptorVersion;
94        CaptureAttempts = captureAttempts;
95        IsFinal = true;
96        return true;
97    }
98}

100/// <summary>Performs the required final-map retry sequence without allocating between firmware calls.</summary>
101/// <nova.when>Call once after all boot-services allocations and immediately before managed post-UEFI boot.</nova.when>
102/// <nova.depends>IUefiMemoryMapProvider and UefiMemoryMapWorkspace</nova.depends>
103public static class FinalUefiMemoryMapAcquirer
104{
105    /// <summary>Captures the final UEFI map and exits boot services with its exact key.</summary>
106    /// <nova.when>Use after the framebuffer, loaded image, and all NovaOryn boot structures are final.</nova.when>
107    /// <nova.depends>No allocation or firmware call between provider methods</nova.depends>
108    /// <returns><see langword="true"/> when ExitBootServices accepted the retained map key.</returns>
109    /// <example><code>bool exited = FinalUefiMemoryMapAcquirer.TryCaptureAndExit(provider, imageHandle, workspace, 8, out FinalUefiMemoryMapSnapshot? snapshot, out UefiMemoryMapStatus mapStatus, out UefiExitBootServicesStatus exitStatus);</code></example>
110    public static bool TryCaptureAndExit(
111        IUefiMemoryMapProvider provider,
112        ulong imageHandle,
113        UefiMemoryMapWorkspace workspace,
114        int maximumAttempts,
115        out FinalUefiMemoryMapSnapshot? snapshot,
116        out UefiMemoryMapStatus mapStatus,
117        out UefiExitBootServicesStatus exitStatus)
118    {
119        snapshot = null;
120        mapStatus = UefiMemoryMapStatus.InvalidParameter;
121        exitStatus = UefiExitBootServicesStatus.FirmwareError;
122        if (provider is null || workspace is null || imageHandle == 0 || maximumAttempts < 1 || workspace.Snapshot.IsFinal) return false;
123
124        for (int attempt = 1; attempt <= maximumAttempts; attempt++)
125        {
126            mapStatus = provider.GetMemoryMap(workspace.Descriptors, out int count, out ulong mapKey, out uint descriptorVersion);
127            if (mapStatus != UefiMemoryMapStatus.Success) return false;
128            if (count < 1 || count > workspace.DescriptorCapacity)
129            {
130                mapStatus = UefiMemoryMapStatus.InvalidParameter;
131                return false;
132            }
133
134            exitStatus = provider.ExitBootServices(imageHandle, mapKey);
135            if (exitStatus == UefiExitBootServicesStatus.InvalidMapKey) continue;
136            if (exitStatus != UefiExitBootServicesStatus.Success) return false;
137            if (!workspace.Snapshot.Seal(workspace.Descriptors, count, mapKey, descriptorVersion, attempt)) return false;
138            snapshot = workspace.Snapshot;
139            return true;
140        }
141        return false;
142    }
143}
