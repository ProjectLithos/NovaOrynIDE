5/// <summary>Registers readable fatal diagnostics for essential x64 exceptions.</summary>
6public sealed class EssentialExceptionHandlers
7{
8    private readonly IInterruptDescriptorTable table;
9    private readonly IExceptionDiagnosticSink sink;
10
11    /// <summary>Creates the essential exception handler set.</summary>
12    public EssentialExceptionHandlers(IInterruptDescriptorTable table, IExceptionDiagnosticSink sink)
13    {
14        this.table = table;
15        this.sink = sink;
16    }
17
18    /// <summary>Registers divide, invalid-opcode, protection, page, double-fault, stack, NMI and machine-check handlers.</summary>
19    public bool RegisterAll()
20    {
21        foreach (byte vector in new byte[] { 0, 2, 6, 8, 12, 13, 14, 18 })
22        {
23            byte ist = vector switch { 8 => 1, 2 => 2, 18 => 3, _ => 0 };
24            InterruptRegistrationResult result = table.Register(vector, HandleFatal,
25                new InterruptRegistrationOptions(NovaOryn.Architecture.DescriptorPrivilegeLevel.Kernel,
26                    InterruptGateType.Interrupt, ist, true));
27            if (!result.Succeeded) return false;
28        }
29        return true;
30    }
31
32    private InterruptResult HandleFatal(ref InterruptContext context)
33    {
34        sink.WriteLine("FATAL CPU EXCEPTION");
35        sink.WriteLine($"Exception: {GetName((byte)context.Vector)} ({context.Vector})");
36        sink.WriteLine($"Error code: 0x{context.ErrorCode:X16} {DecodeErrorCode(ref context)}");
37        sink.WriteLine($"RIP=0x{context.InstructionPointer:X16} CS=0x{context.CodeSegment:X4} RFLAGS=0x{context.Flags:X16}");
38        sink.WriteLine($"RSP=0x{context.StackPointer:X16} SS=0x{context.StackSegment:X4} CPL transition={context.HasPrivilegeTransition()}");
39        sink.WriteLine($"CR0=0x{context.ControlRegister0:X16} CR2=0x{context.ControlRegister2:X16}");
40        sink.WriteLine($"CR3=0x{context.ControlRegister3:X16} CR4=0x{context.ControlRegister4:X16}");
41        sink.WriteLine($"RAX={context.Rax:X16} RBX={context.Rbx:X16} RCX={context.Rcx:X16} RDX={context.Rdx:X16}");
42        sink.WriteLine($"RSI={context.Rsi:X16} RDI={context.Rdi:X16} RBP={context.Rbp:X16}");
43        sink.WriteLine($"R8={context.R8:X16} R9={context.R9:X16} R10={context.R10:X16} R11={context.R11:X16}");
44        sink.WriteLine($"R12={context.R12:X16} R13={context.R13:X16} R14={context.R14:X16} R15={context.R15:X16}");
45        sink.WriteLine($"CPU: {context.ProcessorId}");
46        sink.WriteLine("Current thread/process: unavailable until scheduler integration.");
47        if (context.IsPageFault()) sink.WriteLine($"Page fault address=0x{context.ControlRegister2:X16}; {DecodePageFault(context.ErrorCode)}");
48        sink.WriteLine("Stack trace: unavailable until unwind metadata integration.");
49        sink.StopProcessor();
50        return InterruptResult.Fatal;
51    }
52
53    private static string GetName(byte vector) => vector switch
54    {
55        0 => "Divide error", 2 => "Non-maskable interrupt", 6 => "Invalid opcode",
56        8 => "Double fault", 12 => "Stack-segment fault", 13 => "General protection fault",
57        14 => "Page fault", 18 => "Machine check", _ => "CPU exception"
58    };
59
60    private static string DecodeErrorCode(ref InterruptContext context)
61    {
62        if (context.Vector == 14) return DecodePageFault(context.ErrorCode);
63        if (context.Vector is 10 or 11 or 12 or 13)
64        {
65            ulong code = context.ErrorCode;
66            return $"external={(code & 1) != 0}, table={((code >> 1) & 3)}, selector-index={code >> 3}";
67        }
68        return "architectural error code not applicable";
69    }
70
71    private static string DecodePageFault(ulong code) =>
72        $"present={(code & 1) != 0}, access={((code & 2) != 0 ? "write" : "read")}, " +
73        $"mode={((code & 4) != 0 ? "user" : "supervisor")}, reserved={(code & 8) != 0}, " +
74        $"instruction-fetch={(code & 16) != 0}, protection-key={(code & 32) != 0}, shadow-stack={(code & 64) != 0}";
75}
