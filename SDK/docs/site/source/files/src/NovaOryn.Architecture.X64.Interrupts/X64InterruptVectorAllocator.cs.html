5/// <summary>Allocates x64 driver vectors from 0x40 through 0xEF.</summary>
6public sealed class X64InterruptVectorAllocator : IInterruptVectorAllocator
7{
8    private const byte First = 0x40;
9    private const byte Last = 0xEF;
10    private readonly bool[] allocated = new bool[256];
11
12    /// <inheritdoc />
13    public byte Allocate()
14    {
15        for (int vector = First; vector <= Last; vector++)
16        {
17            if (allocated[vector]) continue;
18            allocated[vector] = true;
19            return (byte)vector;
20        }
21        return 0;
22    }
23
24    /// <inheritdoc />
25    public bool Release(byte vector)
26    {
27        if (vector < First || vector > Last || !allocated[vector]) return false;
28        allocated[vector] = false;
29        return true;
30    }
31
32    /// <inheritdoc />
33    public bool IsAllocated(byte vector) => allocated[vector];
34}
