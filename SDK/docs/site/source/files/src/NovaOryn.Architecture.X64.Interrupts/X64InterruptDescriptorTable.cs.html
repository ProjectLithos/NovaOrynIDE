7/// <summary>Builds and dispatches a complete processor-local x64 IDT.</summary>
8public sealed unsafe class X64InterruptDescriptorTable : IInterruptDescriptorTable
9{
10    private const uint EntryCount = 256;
11    private const uint EntrySize = 16;
12    private const uint RequiredBytes = EntryCount * EntrySize;
13    private static X64InterruptDescriptorTable? installed;
14    private readonly InterruptHandler?[] handlers = new InterruptHandler?[EntryCount];
15    private readonly InterruptRegistrationOptions[] options = new InterruptRegistrationOptions[EntryCount];
16    private readonly ulong[] generations = new ulong[EntryCount];
17    private InterruptDescriptorTableConfiguration configuration;
18    private bool configured;
19
20    /// <inheritdoc />
21    public bool Configure(InterruptDescriptorTableConfiguration value)
22    {
23        if (value.TableAddress.Value == 0 || value.TableCapacity < RequiredBytes || value.KernelCodeSelector.Value == 0) return false;
24        if (!IsIst(value.DoubleFaultInterruptStack) || !IsIst(value.NonMaskableInterruptStack) || !IsIst(value.MachineCheckInterruptStack)) return false;
25        configuration = value;
26        for (int vector = 0; vector < EntryCount; vector++)
27        {
28            byte ist = GetDefaultIst((byte)vector, value);
29            WriteGate((byte)vector, NativeMethods.GetInterruptStub((byte)vector), value.KernelCodeSelector.Value,
30                InterruptGateType.Interrupt, DescriptorPrivilegeLevel.Kernel, ist);
31            options[vector] = new InterruptRegistrationOptions(DescriptorPrivilegeLevel.Kernel, InterruptGateType.Interrupt, ist, false);
32        }
33        configured = true;
34        return true;
35    }
36
37    /// <inheritdoc />
38    public bool Install()
39    {
40        if (!configured) return false;
41        installed = this;
42        delegate* unmanaged[Cdecl]<InterruptContext*, int> dispatcher = &DispatchNative;
43        if (!NativeMethods.SetInterruptDispatcher((ulong)dispatcher)) return false;
44        return NativeMethods.LoadInterruptDescriptorTable(configuration.TableAddress.Value, (ushort)(RequiredBytes - 1));
45    }
46
47    /// <inheritdoc />
48    public InterruptRegistrationResult Register(byte vector, InterruptHandler handler, InterruptRegistrationOptions value)
49    {
50        if (!configured) return Failed("The IDT is not configured.");
51        if (handler is null) return Failed("The handler is required.");
52        if (!IsIst(value.InterruptStackTable)) return Failed("IST must be between zero and seven.");
53        if (handlers[vector] is not null && !value.ReplaceExisting) return Failed("The vector already has a handler.");
54        generations[vector]++;
55        handlers[vector] = handler;
56        options[vector] = value;
57        WriteGate(vector, NativeMethods.GetInterruptStub(vector), configuration.KernelCodeSelector.Value,
58            value.GateType, value.PrivilegeLevel, value.InterruptStackTable);
59        return new InterruptRegistrationResult(true, new InterruptRegistrationHandle(vector, generations[vector]), string.Empty);
60    }
61
62    /// <inheritdoc />
63    public bool Remove(InterruptRegistrationHandle handle)
64    {
65        byte vector = handle.Vector;
66        if (handlers[vector] is null || generations[vector] != handle.Generation) return false;
67        handlers[vector] = null;
68        generations[vector]++;
69        byte ist = GetDefaultIst(vector, configuration);
70        options[vector] = new InterruptRegistrationOptions(DescriptorPrivilegeLevel.Kernel, InterruptGateType.Interrupt, ist, false);
71        WriteGate(vector, NativeMethods.GetInterruptStub(vector), configuration.KernelCodeSelector.Value,
72            InterruptGateType.Interrupt, DescriptorPrivilegeLevel.Kernel, ist);
73        return true;
74    }
75
76    /// <inheritdoc />
77    public bool IsRegistered(byte vector) => handlers[vector] is not null;
78
79    /// <inheritdoc />
80    public InterruptResult Dispatch(ref InterruptContext context)
81    {
82        if (context.Vector > byte.MaxValue) return InterruptResult.Fatal;
83        InterruptHandler? handler = handlers[(byte)context.Vector];
84        if (handler is null) return InterruptResult.Fatal;
85        InterruptResult result = handler(ref context);
86        return result == InterruptResult.Unhandled && context.Vector < 32 ? InterruptResult.Fatal : result;
87    }
88
89    /// <summary>Stops the current processor through the native terminal halt loop.</summary>
90    public bool StopProcessor() => NativeMethods.StopProcessor();
91
92    [UnmanagedCallersOnly(CallConvs = [typeof(CallConvCdecl)])]
93    private static int DispatchNative(InterruptContext* context)
94    {
95        if (installed is null || context is null) return (int)InterruptResult.Fatal;
96        return (int)installed.Dispatch(ref *context);
97    }
98
99    private static bool IsIst(byte value) => value <= 7;
100
101    private static byte GetDefaultIst(byte vector, InterruptDescriptorTableConfiguration value) => vector switch
102    {
103        (byte)CpuExceptionVector.DoubleFault => value.DoubleFaultInterruptStack,
104        (byte)CpuExceptionVector.NonMaskableInterrupt => value.NonMaskableInterruptStack,
105        (byte)CpuExceptionVector.MachineCheck => value.MachineCheckInterruptStack,
106        _ => 0
107    };
108
109    private static InterruptRegistrationResult Failed(string error) =>
110        new(false, new InterruptRegistrationHandle(0, 0), error);
111
112    private void WriteGate(byte vector, ulong target, ushort selector, InterruptGateType gateType,
113        DescriptorPrivilegeLevel privilegeLevel, byte ist)
114    {
115        byte* entry = (byte*)configuration.TableAddress.Value + (vector * EntrySize);
116        *(ushort*)(entry + 0) = (ushort)target;
117        *(ushort*)(entry + 2) = selector;
118        entry[4] = (byte)(ist & 0x7);
119        entry[5] = (byte)(0x80 | (((byte)privilegeLevel & 0x3) << 5) | ((byte)gateType & 0xF));
120        *(ushort*)(entry + 6) = (ushort)(target >> 16);
121        *(uint*)(entry + 8) = (uint)(target >> 32);
122        *(uint*)(entry + 12) = 0;
123        NativeMethods.SetInterruptStackSwitch(vector, ist != 0);
124    }
