21/// <summary>Exposes the standard NovaOryn x64 kernel virtual-address reservations to freestanding kernels.</summary>
22public static class KernelAddressSpace
23{
24    /// <summary>First user virtual byte; low 64 KiB remain a guard.</summary>
25    public const UInt64 UserBase = 0x0000000000010000UL;
26    /// <summary>Exclusive end of the user half.</summary>
27    public const UInt64 UserEndExclusive = 0x0000800000000000UL;
28    /// <summary>Higher-half kernel image base.</summary>
29    public const UInt64 KernelImageBase = 0xFFFF800000000000UL;
30    /// <summary>Kernel image reservation length.</summary>
31    public const UInt64 KernelImageLength = 0x0000000100000000UL;
32    /// <summary>Future kernel heap base.</summary>
33    public const UInt64 KernelHeapBase = 0xFFFF810000000000UL;
34    /// <summary>Future kernel heap reservation length.</summary>
35    public const UInt64 KernelHeapLength = 0x0000010000000000UL;
36    /// <summary>Kernel stack arena base.</summary>
37    public const UInt64 KernelStacksBase = 0xFFFF820000000000UL;
38    /// <summary>Kernel stack arena length.</summary>
39    public const UInt64 KernelStacksLength = 0x0000010000000000UL;
40    /// <summary>Direct physical map base.</summary>
41    public const UInt64 DirectMapBase = 0xFFFF900000000000UL;
42    /// <summary>Direct physical map capacity.</summary>
43    public const UInt64 DirectMapLength = 0x0000400000000000UL;
44    /// <summary>MMIO window base.</summary>
45    public const UInt64 MmioBase = 0xFFFFD00000000000UL;
46    /// <summary>MMIO window length.</summary>
47    public const UInt64 MmioLength = 0x0000100000000000UL;
48    /// <summary>Page-table access window base.</summary>
49    public const UInt64 PageTableWindowBase = 0xFFFFFF0000000000UL;
50    /// <summary>Page-table access window length.</summary>
51    public const UInt64 PageTableWindowLength = 0x0000008000000000UL;
52    private static Boolean _initialized;
53    private static KernelAddressSpaceStatus _status = KernelAddressSpaceStatus.VirtualMemoryNotInitialized;
54
55    /// <summary>Gets whether the address-space policy has been initialized.</summary>
56    public static Boolean IsInitialized() => _initialized;
57    /// <summary>Gets the last initialization status.</summary>
58    public static KernelAddressSpaceStatus GetLastStatus() => _status;
59    /// <summary>Gets a freestanding-safe symbolic name for the last initialization status.</summary>
60    /// <returns>A stable status name that can be written directly without enum formatting or <c>string.Format</c>.</returns>
61    public static String GetLastStatusName()
62    {
63        if (_status == KernelAddressSpaceStatus.Success) return "Success";
64        if (_status == KernelAddressSpaceStatus.VirtualMemoryNotInitialized) return "VirtualMemoryNotInitialized";
65        if (_status == KernelAddressSpaceStatus.AlreadyInitialized) return "AlreadyInitialized";
66        if (_status == KernelAddressSpaceStatus.InvalidLayout) return "InvalidLayout";
67        if (_status == KernelAddressSpaceStatus.DirectMapInitializationFailed) return "DirectMapInitializationFailed";
68        return "Unknown";
69    }
70
71    /// <summary>Validates and activates the standard address-space policy for later kernel subsystems.</summary>
72    /// <returns><see langword="true"/> when the VMM is ready and every standard region satisfies the compiled invariants.</returns>
73    public static Boolean Initialize()
74    {
75        if (_initialized) { _status = KernelAddressSpaceStatus.AlreadyInitialized; return false; }
76        if (!KernelVirtualMemory.IsInitialized()) { _status = KernelAddressSpaceStatus.VirtualMemoryNotInitialized; return false; }
77        if (!ValidateRange(UserBase, UserEndExclusive - UserBase, false) ||
78            !ValidateRange(KernelImageBase, KernelImageLength, true) || !ValidateRange(KernelHeapBase, KernelHeapLength, true) ||
79            !ValidateRange(KernelStacksBase, KernelStacksLength, true) || !ValidateRange(DirectMapBase, DirectMapLength, true) ||
80            !ValidateRange(MmioBase, MmioLength, true) || !ValidateRange(PageTableWindowBase, PageTableWindowLength, true) ||
81            Overlaps(KernelImageBase, KernelImageLength, KernelHeapBase, KernelHeapLength) ||
82            Overlaps(KernelHeapBase, KernelHeapLength, KernelStacksBase, KernelStacksLength) ||
83            Overlaps(KernelStacksBase, KernelStacksLength, DirectMapBase, DirectMapLength) ||
84            Overlaps(DirectMapBase, DirectMapLength, MmioBase, MmioLength) ||
85            Overlaps(MmioBase, MmioLength, PageTableWindowBase, PageTableWindowLength))
86        { _status = KernelAddressSpaceStatus.InvalidLayout; return false; }
87        if (!KernelVirtualMemory.InitializeDirectMap(DirectMapBase, DirectMapLength))
88        { _status = KernelAddressSpaceStatus.DirectMapInitializationFailed; return false; }
89        _initialized = true; _status = KernelAddressSpaceStatus.Success; return true;
90    }

92    /// <summary>Attempts to convert a physical byte address into the standard direct-map virtual address.</summary>
93    /// <returns><see langword="true"/> when the physical address fits the 64 TiB direct-map capacity.</returns>
94    public static Boolean TryPhysicalToDirectMap(UInt64 physicalAddress, out UInt64 virtualAddress)
95    { virtualAddress = 0; if (!_initialized || physicalAddress >= DirectMapLength) return false; virtualAddress = DirectMapBase + physicalAddress; return true; }
96
97    /// <summary>Attempts to convert one direct-map virtual address to a physical byte address.</summary>
98    /// <returns><see langword="true"/> when the virtual address lies inside the direct-map reservation.</returns>
99    public static Boolean TryDirectMapToPhysical(UInt64 virtualAddress, out UInt64 physicalAddress)
100    { physicalAddress = 0; if (!_initialized || virtualAddress < DirectMapBase || virtualAddress >= DirectMapBase + DirectMapLength) return false; physicalAddress = virtualAddress - DirectMapBase; return true; }

102    private static Boolean ValidateRange(UInt64 address, UInt64 length, Boolean high)
103    {
104        if (length == 0 || (address & 0xFFFUL) != 0 || (length & 0xFFFUL) != 0 || address > 0xFFFFFFFFFFFFFFFFUL - length) return false;
105        UInt64 last = address + length - 1UL;
106        if (!IsCanonical(address) || !IsCanonical(last)) return false;
107        return high ? address >= 0xFFFF800000000000UL : last < 0x0000800000000000UL;
108    }
109    private static Boolean IsCanonical(UInt64 address) { UInt64 upper = address >> 48; return ((address >> 47) & 1UL) != 0 ? upper == 0xFFFFUL : upper == 0UL; }
110    private static Boolean Overlaps(UInt64 a, UInt64 al, UInt64 b, UInt64 bl) => a < b + bl && b < a + al;
111}
