6/// <summary>Provides checked x64 canonical-address and page-table index calculations.</summary>
7/// <nova.when>Use when building, walking, or validating x64 four-level page tables.</nova.when>
8/// <nova.depends>x64 48-bit canonical virtual-address form</nova.depends>
9[SupportedArchitecture(SupportedArchitecture.X64)]
10[BootStage(BootStage.ManagedBootstrap)]
11public static class X64VirtualAddress
12{
13    /// <summary>Determines whether a 64-bit address is canonical for four-level x64 paging.</summary>
14    /// <nova.when>Use before dereferencing or installing any x64 virtual mapping.</nova.when>
15    /// <nova.depends>Bits 63:48 replicate bit 47</nova.depends>
16    /// <returns><see langword="true"/> when the address is in the low or high canonical half.</returns>
17    /// <example><code>bool canonical = X64VirtualAddress.IsCanonical(0xFFFF800000000000);</code></example>
18    public static bool IsCanonical(ulong address)
19    {
20        ulong upper = address >> 48;
21        bool high = ((address >> 47) & 1UL) != 0;
22        return high ? upper == 0xFFFFUL : upper == 0UL;
23    }
24
25    /// <summary>Attempts to obtain the four 9-bit page-table indices for one canonical address.</summary>
26    /// <nova.when>Use when walking PML4, PDPT, page-directory, and page-table levels.</nova.when>
27    /// <nova.depends>Successful canonical-address validation</nova.depends>
28    /// <returns><see langword="true"/> when the address is canonical and all indices were produced.</returns>
29    /// <example><code>bool indexed = X64VirtualAddress.TryGetIndices(address, out ushort pml4, out ushort pdpt, out ushort pd, out ushort pt);</code></example>
30    public static bool TryGetIndices(ulong address, out ushort pml4, out ushort pdpt, out ushort pd, out ushort pt)
31    {
32        pml4 = 0;
33        pdpt = 0;
34        pd = 0;
35        pt = 0;
36        if (!IsCanonical(address)) return false;
37        pml4 = (ushort)((address >> 39) & 0x1FFUL);
38        pdpt = (ushort)((address >> 30) & 0x1FFUL);
39        pd = (ushort)((address >> 21) & 0x1FFUL);
40        pt = (ushort)((address >> 12) & 0x1FFUL);
41        return true;
42    }
43}

45/// <summary>Encodes and decodes x64 long-mode page-table leaf entries without exposing raw bit arithmetic to SDK consumers.</summary>
46/// <nova.when>Use inside x64 virtual-memory implementations and diagnostics.</nova.when>
47/// <nova.depends>VirtualMemoryProtection and x64 long-mode page-table encoding</nova.depends>
48[SupportedArchitecture(SupportedArchitecture.X64)]
49[BootStage(BootStage.ManagedBootstrap)]
50public static class X64PageTableCodec
51{
52    private const ulong Present = 1UL << 0;
53    private const ulong Writable = 1UL << 1;
54    private const ulong User = 1UL << 2;
55    private const ulong WriteThrough = 1UL << 3;
56    private const ulong CacheDisable = 1UL << 4;
57    private const ulong LargePage = 1UL << 7;
58    private const ulong Global = 1UL << 8;
59    private const ulong NoExecute = 1UL << 63;
60    private const ulong AddressMask4KiB = 0x000FFFFFFFFFF000UL;
61    private const ulong AddressMask2MiB = 0x000FFFFFFFE00000UL;
62    private const ulong AddressMask1GiB = 0x000FFFFFC0000000UL;
63
64    /// <summary>Attempts to encode one x64 present leaf entry for the selected page size.</summary>
65    /// <nova.when>Use after validating processor support for the selected page size and NX policy.</nova.when>
66    /// <nova.depends>Aligned physical address and architecture-neutral protection</nova.depends>
67    /// <returns><see langword="true"/> when the address and protection can be represented safely.</returns>
68    /// <example><code>bool encoded = X64PageTableCodec.TryEncodeLeaf(new PhysicalAddress(0x200000), VirtualPageSize.Page2MiB, VirtualMemoryProtection.Read | VirtualMemoryProtection.Write, out ulong entry);</code></example>
69    public static bool TryEncodeLeaf(PhysicalAddress physicalAddress, VirtualPageSize pageSize, VirtualMemoryProtection protection, out ulong entry)
70    {
71        entry = 0;
72        ulong size = (ulong)pageSize;
73        if (size != 4096UL && size != 2097152UL && size != 1073741824UL) return false;
74        if ((physicalAddress.Value & (size - 1UL)) != 0) return false;
75        if ((protection & VirtualMemoryProtection.Read) == 0) return false;
76
77        ulong addressMask = pageSize switch
78        {
79            VirtualPageSize.Page4KiB => AddressMask4KiB,
80            VirtualPageSize.Page2MiB => AddressMask2MiB,
81            VirtualPageSize.Page1GiB => AddressMask1GiB,
82            _ => 0UL
83        };
84        if ((physicalAddress.Value & ~addressMask) != 0) return false;
85
86        ulong flags = Present;
87        if ((protection & VirtualMemoryProtection.Write) != 0) flags |= Writable;
88        if ((protection & VirtualMemoryProtection.User) != 0) flags |= User;
89        if ((protection & VirtualMemoryProtection.Global) != 0) flags |= Global;
90        if ((protection & VirtualMemoryProtection.Device) != 0) flags |= CacheDisable;
91        if ((protection & VirtualMemoryProtection.WriteThrough) != 0) flags |= WriteThrough;
92        if ((protection & VirtualMemoryProtection.Execute) == 0) flags |= NoExecute;
93        if (pageSize != VirtualPageSize.Page4KiB) flags |= LargePage;
94        entry = (physicalAddress.Value & addressMask) | flags;
95        return true;
96    }

98    /// <summary>Attempts to decode a present x64 leaf entry and exact translated byte address.</summary>
99    /// <nova.when>Use when translating or inspecting an already-walked leaf entry.</nova.when>
100    /// <nova.depends>Correct leaf level supplied by the caller</nova.depends>
101    /// <returns><see langword="true"/> when the entry is present and consistent with the supplied page size.</returns>
102    /// <example><code>bool decoded = X64PageTableCodec.TryDecodeLeaf(entry, virtualAddress, VirtualPageSize.Page4KiB, out VirtualTranslation translation);</code></example>
103    public static bool TryDecodeLeaf(ulong entry, ulong virtualAddress, VirtualPageSize pageSize, out VirtualTranslation translation)
104    {
105        translation = default;
106        if ((entry & Present) == 0 || !X64VirtualAddress.IsCanonical(virtualAddress)) return false;
107        if (pageSize == VirtualPageSize.Page4KiB && (entry & LargePage) != 0) return false;
108        if (pageSize != VirtualPageSize.Page4KiB && (entry & LargePage) == 0) return false;
109
110        ulong size = (ulong)pageSize;
111        ulong mask = pageSize switch
112        {
113            VirtualPageSize.Page4KiB => AddressMask4KiB,
114            VirtualPageSize.Page2MiB => AddressMask2MiB,
115            VirtualPageSize.Page1GiB => AddressMask1GiB,
116            _ => 0UL
117        };
118        if (mask == 0) return false;
119
120        ulong physicalBase = entry & mask;
121        ulong offset = virtualAddress & (size - 1UL);
122        VirtualMemoryProtection protection = VirtualMemoryProtection.Read;
123        if ((entry & Writable) != 0) protection |= VirtualMemoryProtection.Write;
124        if ((entry & User) != 0) protection |= VirtualMemoryProtection.User;
125        if ((entry & Global) != 0) protection |= VirtualMemoryProtection.Global;
126        if ((entry & CacheDisable) != 0) protection |= VirtualMemoryProtection.Device;
127        if ((entry & WriteThrough) != 0) protection |= VirtualMemoryProtection.WriteThrough;
128        if ((entry & NoExecute) == 0) protection |= VirtualMemoryProtection.Execute;
129        translation = new VirtualTranslation(virtualAddress, new PhysicalAddress(physicalBase + offset), pageSize, protection);
130        return true;
131    }

133    /// <summary>Attempts to encode an x64 non-leaf page-table pointer.</summary>
134    /// <nova.when>Use when allocating an intermediate PML4, PDPT, page-directory, or page-table child.</nova.when>
135    /// <nova.depends>4 KiB-aligned physical page-table storage</nova.depends>
136    /// <returns><see langword="true"/> when the physical address fits the x64 page-table address field.</returns>
137    /// <example><code>bool encoded = X64PageTableCodec.TryEncodeTablePointer(new PhysicalAddress(0x3000), true, true, out ulong entry);</code></example>
138    public static bool TryEncodeTablePointer(PhysicalAddress physicalAddress, bool writable, bool userAccessible, out ulong entry)
139    {
140        entry = 0;
141        if ((physicalAddress.Value & 0xFFFUL) != 0 || (physicalAddress.Value & ~AddressMask4KiB) != 0) return false;
142        ulong flags = Present;
143        if (writable) flags |= Writable;
144        if (userAccessible) flags |= User;
145        entry = (physicalAddress.Value & AddressMask4KiB) | flags;
146        return true;
147    }

149    /// <summary>Gets the 4 KiB-aligned physical page-table address stored in a present non-leaf entry.</summary>
150    /// <nova.when>Use while walking intermediate x64 page-table levels.</nova.when>
151    /// <nova.depends>Caller has verified the entry is a present non-leaf pointer</nova.depends>
152    /// <returns>The physical child-table address, or zero when the entry is not present.</returns>
153    /// <example><code>PhysicalAddress child = X64PageTableCodec.GetTableAddress(entry);</code></example>
154    public static PhysicalAddress GetTableAddress(ulong entry) => new((entry & Present) != 0 ? entry & AddressMask4KiB : 0UL);
155
156    /// <summary>Determines whether an x64 entry is present.</summary>
157    /// <nova.when>Use before interpreting an entry as a table pointer or leaf.</nova.when>
158    /// <nova.depends>x64 present bit</nova.depends>
159    /// <returns><see langword="true"/> when the present bit is set.</returns>
160    /// <example><code>bool present = X64PageTableCodec.IsPresent(entry);</code></example>
161    public static bool IsPresent(ulong entry) => (entry & Present) != 0;
162
163    /// <summary>Determines whether an x64 entry marks a large leaf at PDPT or page-directory level.</summary>
164    /// <nova.when>Use while walking x64 page tables before descending to a child table.</nova.when>
165    /// <nova.depends>x64 page-size bit</nova.depends>
166    /// <returns><see langword="true"/> when the large-page bit is set.</returns>
167    /// <example><code>bool large = X64PageTableCodec.IsLargePage(entry);</code></example>
168    public static bool IsLargePage(ulong entry) => (entry & LargePage) != 0;
