5/// <summary>Identifies a selector loaded into an x64 segment register.</summary>
6public readonly record struct SegmentSelector(ushort Value)
7{
8    /// <summary>Gets the descriptor-table index encoded by this selector.</summary>
9    public ushort Index => (ushort)(Value >> 3);
10
11    /// <summary>Gets the requested privilege level encoded by this selector.</summary>
12    public DescriptorPrivilegeLevel PrivilegeLevel => (DescriptorPrivilegeLevel)(Value & 0x3);
13
14    /// <summary>Creates a selector for a GDT entry.</summary>
15    /// <returns>The encoded selector value.</returns>
16    public static SegmentSelector Create(ushort index, DescriptorPrivilegeLevel privilegeLevel) =>
17        new((ushort)((index << 3) | ((ushort)privilegeLevel & 0x3)));
18}
