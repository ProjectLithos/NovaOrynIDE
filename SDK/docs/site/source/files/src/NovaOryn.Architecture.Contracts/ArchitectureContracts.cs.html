5/// <summary>Defines lifecycle and discovery services implemented by a processor architecture.</summary>
6/// <nova.when>Use this contract when kernel code selects or initialises an architecture implementation.</nova.when>

17    /// <summary>Initialises architecture state required before memory and interrupt services.</summary>
18    /// <nova.when>Call once on the bootstrap processor during managed bootstrap.</nova.when>

20    /// <returns><see langword="true"/> when early initialisation completed.</returns>
21    /// <example><code>bool ready = architecture.InitialiseEarly();</code></example>

24    /// <summary>Initialises architecture state for one logical processor.</summary>

31    /// <summary>Enables maskable interrupts on the current processor.</summary>

43    bool DisableInterrupts();

45    /// <summary>Halts the current processor according to architecture policy.</summary>
