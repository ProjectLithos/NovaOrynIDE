5/// <summary>Defines the architecture-neutral processor operations exposed to kernel services.</summary>
6/// <nova.when>Use this contract when code needs processor control without depending on x64 or ARM64 implementation details.</nova.when>
7/// <nova.depends>NovaOryn.Primitives.ProcessorId</nova.depends>
8public interface ICpu
9{
10    /// <summary>Enables maskable interrupts on the current processor.</summary>
11    /// <returns><see langword="true"/> when interrupts are enabled.</returns>
12    /// <example><code>bool enabled = cpu.EnableInterrupts();</code></example>
13    bool EnableInterrupts();
14
15    /// <summary>Disables maskable interrupts on the current processor.</summary>
16    /// <returns><see langword="true"/> when interrupts are disabled.</returns>
17    /// <example><code>bool disabled = cpu.DisableInterrupts();</code></example>
18    bool DisableInterrupts();
19
20    /// <summary>Determines whether maskable interrupts are enabled on the current processor.</summary>
21    /// <returns><see langword="true"/> when interrupts are enabled; otherwise, <see langword="false"/>.</returns>
22    /// <example><code>bool enabled = cpu.AreInterruptsEnabled();</code></example>
23    bool AreInterruptsEnabled();
24
25    /// <summary>Gets the stable NovaOryn identifier for the current logical processor.</summary>

27    /// <example><code>ProcessorId processor = cpu.GetProcessorId();</code></example>
28    ProcessorId GetProcessorId();

30    /// <summary>Halts the current processor until an interrupt or architecture-defined wake event occurs.</summary>
31    /// <returns><see langword="true"/> if execution resumes after the halt operation.</returns>
32    /// <example><code>bool resumed = cpu.Halt();</code></example>
33    bool Halt();
34}
