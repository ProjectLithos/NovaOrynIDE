6/// <summary>Allocates physical memory as aligned power-of-two buddy blocks with recursive split and coalescing.</summary>
7/// <nova.when>Choose when fast coalescing and naturally aligned power-of-two blocks are preferred and bounded internal fragmentation is acceptable.</nova.when>

35    /// <summary>Calculates one-byte-per-frame buddy metadata plus bounded ownership record tables.</summary>
36    /// <nova.when>Call before reserving caller-owned buddy metadata storage.</nova.when>
37    /// <nova.depends>Highest and lowest immediately allocatable physical frame</nova.depends>
38    /// <returns><see langword="true"/> when the metadata size can be represented.</returns>
39    /// <example><code>bool sized = BuddyPhysicalMemoryManager.TryGetRequiredWorkspaceBytes(map, 128, 32, out ulong bytes);</code></example>
40    public static bool TryGetRequiredWorkspaceBytes(NormalisedMemoryMap map, int allocationCapacity, int reservationCapacity, out ulong bytes)
41    {
42        bytes = 0;
43        if (!PhysicalAllocatorInternals.TryGetManagedFrameBounds(map, out ulong minimumFrame, out ulong maximumFrameExclusive, out _, out _)) return false;
44        ulong span = maximumFrameExclusive - minimumFrame;
45        if (!PhysicalAllocatorInternals.TryAlignWorkspaceBytes(span, out ulong primaryBytes)) return false;
46        return PhysicalAllocatorInternals.TryWorkspaceBytes(primaryBytes, allocationCapacity, reservationCapacity, out bytes);
47    }

49    /// <summary>Initialises free buddy blocks from immediately allocatable normalised ranges without allocating metadata internally.</summary>
50    /// <nova.when>Call once after memory-map normalisation and before the VMM or kernel heap.</nova.when>
51    /// <nova.depends>Caller-owned PhysicalAllocatorWorkspace</nova.depends>
52    /// <returns><see langword="true"/> when the frame-state table and ownership records fit the supplied workspace.</returns>
53    /// <example><code>bool ready = allocator.TryInitialize(map, workspace, 128, 32, out PhysicalMemoryStatus status);</code></example>
54    public bool TryInitialize(NormalisedMemoryMap map, PhysicalAllocatorWorkspace workspace, int allocationCapacity, int reservationCapacity, out PhysicalMemoryStatus status)
55    {
56        status = PhysicalMemoryStatus.InvalidParameter;
57        if (_initialized)
58        {
59            status = PhysicalMemoryStatus.AlreadyInitialized;
60            return false;
61        }
62        if (!PhysicalAllocatorInternals.TryGetManagedFrameBounds(map, out ulong minimumFrame, out ulong maximumFrameExclusive, out ulong managedPages, out _)) return false;
63        if (!TryGetRequiredWorkspaceBytes(map, allocationCapacity, reservationCapacity, out ulong requiredBytes)) return false;
64        ulong span = maximumFrameExclusive - minimumFrame;
65        if (!PhysicalAllocatorInternals.TryAlignWorkspaceBytes(span, out ulong primaryBytes)) return false;
66        if (workspace.ByteLength < requiredBytes || span > long.MaxValue || !PhysicalAllocatorInternals.TryPartition(workspace, primaryBytes, allocationCapacity, reservationCapacity, out byte* primary, out AllocationRecord* allocations, out ReservationRecord* reservations))
67        {
68            status = PhysicalMemoryStatus.WorkspaceTooSmall;
69            return false;
70        }
71
72        _states = (sbyte*)primary;
73        _allocations = allocations;
74        _reservations = reservations;
75        _minimumFrame = minimumFrame;
76        _frameSpan = span;
77        _totalManagedPages = managedPages;
78        _allocationCapacity = allocationCapacity;
79        _reservationCapacity = reservationCapacity;
80        _nextToken = 0;
81        for (ulong index = 0; index < _frameSpan; index++) _states[index] = 0;
82        ClearRecords();
83
84        for (int index = 0; index < map.Count; index++)
85        {
86            if (!map.TryGetDescriptor(index, out MemoryDescriptor descriptor)) return false;
87            if (descriptor.Availability != MemoryAvailability.AvailableAfterExitBootServices) continue;
88            SeedRange(descriptor.PhysicalStart.Value / PageSize, descriptor.PageCount);
89        }
90        CoalesceSeededBlocks();
91        _initialized = true;
92        status = PhysicalMemoryStatus.Success;
93        return true;
94    }

96    /// <summary>Allocates one power-of-two buddy block large enough for the requested page count and alignment.</summary>
97    /// <nova.when>Use when natural alignment and deterministic buddy coalescing are useful; inspect RequestedPageCount versus Range.PageCount for rounding.</nova.when>
98    /// <nova.depends>Successful TryInitialize</nova.depends>
99    /// <returns><see langword="true"/> when a suitable free block can be split and recorded.</returns>
100    /// <example><code>bool ok = allocator.TryAllocate(request, out PhysicalAllocation allocation, out PhysicalMemoryStatus status);</code></example>
101    public bool TryAllocate(PhysicalAllocationRequest request, out PhysicalAllocation allocation, out PhysicalMemoryStatus status)
102    {
103        allocation = default;
104        status = PhysicalMemoryStatus.NotInitialized;
105        if (!_initialized) return false;
106        if (!PhysicalAllocatorInternals.TryGetRequestFrameWindow(request, out ulong minimumFrame, out ulong maximumFrameExclusive))
107        {
108            status = PhysicalMemoryStatus.InvalidParameter;
109            return false;
110        }
111        if (!PhysicalAllocatorInternals.TryFindFreeAllocationRecord(_allocations, _allocationCapacity, out int recordIndex))
112        {
113            status = PhysicalMemoryStatus.RecordCapacityExhausted;
114            return false;
115        }
116
117        int requestedOrder = PhysicalAllocatorInternals.CeilLog2(request.PageCount);
118        int alignmentOrder = PhysicalAllocatorInternals.CeilLog2(request.AlignmentPages);
119        int targetOrder = requestedOrder > alignmentOrder ? requestedOrder : alignmentOrder;
120        if (targetOrder < 0 || targetOrder > 62)
121        {
122            status = PhysicalMemoryStatus.InvalidParameter;
123            return false;
124        }
125        ulong targetPages = PhysicalAllocatorInternals.BlockPages(targetOrder);
126
127        for (ulong offset = 0; offset < _frameSpan; offset++)
128        {
129            sbyte state = _states[offset];
130            if (state <= 0) continue;
131            int blockOrder = state - 1;
132            if (blockOrder < targetOrder) continue;
133            ulong blockStart = _minimumFrame + offset;
134            ulong blockPages = PhysicalAllocatorInternals.BlockPages(blockOrder);
135            ulong blockEnd = blockStart + blockPages;
136            ulong lower = blockStart > minimumFrame ? blockStart : minimumFrame;
137            ulong upper = blockEnd < maximumFrameExclusive ? blockEnd : maximumFrameExclusive;
138            if (lower >= upper || !PhysicalAllocatorInternals.TryAlignFrame(lower, targetPages, out ulong targetStart)) continue;
139            if (!PhysicalAllocatorInternals.FitsWindow(targetStart, targetPages, lower, upper)) continue;
140            if (!SplitToTarget(blockStart, blockOrder, targetStart, targetOrder)) continue;
141            ulong targetOffset = targetStart - _minimumFrame;
142            _states[targetOffset] = (sbyte)-(targetOrder + 1);
143
144            ref AllocationRecord record = ref _allocations[recordIndex];
145            record.Token = PhysicalAllocatorInternals.NextToken(ref _nextToken);
146            record.StartFrame = targetStart;
147            record.RequestedPages = request.PageCount;
148            record.ActualPages = targetPages;
149            record.Purpose = request.Purpose;
150            record.Active = 1;
151            allocation = PhysicalAllocatorInternals.CreateAllocation(ref record);
152            status = PhysicalMemoryStatus.Success;
153            return true;
154        }
155
156        status = HasFreeBlockAtLeast(targetOrder) ? PhysicalMemoryStatus.AddressConstraintUnsatisfied : PhysicalMemoryStatus.OutOfMemory;
157        return false;
158    }

160    /// <summary>Releases a live buddy allocation and recursively coalesces matching free buddies.</summary>
161    /// <nova.when>Call when the allocation owner no longer uses the whole returned buddy block.</nova.when>
162    /// <nova.depends>Allocation token returned by this manager</nova.depends>
163    /// <returns><see langword="true"/> when the token is live and the block is released exactly once.</returns>
164    /// <example><code>bool ok = allocator.TryRelease(allocation, out PhysicalMemoryStatus status);</code></example>
165    public bool TryRelease(PhysicalAllocation allocation, out PhysicalMemoryStatus status)
166    {
167        status = PhysicalMemoryStatus.NotInitialized;
168        if (!_initialized) return false;
169        if (!PhysicalAllocatorInternals.TryFindAllocationRecord(_allocations, _allocationCapacity, allocation, out int recordIndex))
170        {
171            status = PhysicalMemoryStatus.AllocationNotFound;
172            return false;
173        }
174        AllocationRecord record = _allocations[recordIndex];
175        int order = PhysicalAllocatorInternals.CeilLog2(record.ActualPages);
176        ulong offset = record.StartFrame - _minimumFrame;
177        _states[offset] = (sbyte)(order + 1);
178        Coalesce(record.StartFrame, order);
179        _allocations[recordIndex].Active = 0;
180        status = PhysicalMemoryStatus.Success;
181        return true;
182    }

184    /// <summary>Reserves an exact arbitrary free range by splitting containing buddy blocks down to individual pages.</summary>
185    /// <nova.when>Use for late fixed-range exclusions without requiring the reserved range itself to be a power of two.</nova.when>
186    /// <nova.depends>PhysicalFrameRange</nova.depends>
187    /// <returns><see langword="true"/> when every requested frame was free and is reserved exactly.</returns>
188    /// <example><code>bool ok = allocator.TryReserve(range, PhysicalMemoryPurpose.Metadata, out PhysicalReservation reservation, out PhysicalMemoryStatus status);</code></example>
189    public bool TryReserve(PhysicalFrameRange range, PhysicalMemoryPurpose purpose, out PhysicalReservation reservation, out PhysicalMemoryStatus status)
190    {
191        reservation = default;
192        status = PhysicalMemoryStatus.NotInitialized;
193        if (!_initialized) return false;
194        if (range.PageCount == 0 || (range.Start.Value & 0xFFFUL) != 0)
195        {
196            status = PhysicalMemoryStatus.InvalidParameter;
197            return false;
198        }
199        if (!PhysicalAllocatorInternals.TryFindFreeReservationRecord(_reservations, _reservationCapacity, out int recordIndex))
200        {
201            status = PhysicalMemoryStatus.RecordCapacityExhausted;
202            return false;
203        }
204        ulong startFrame = range.Start.Value / PageSize;
205        if (startFrame < _minimumFrame || startFrame - _minimumFrame > _frameSpan || range.PageCount > _frameSpan - (startFrame - _minimumFrame))
206        {
207            status = PhysicalMemoryStatus.RangeNotFree;
208            return false;
209        }
210
211        ulong reservedCount = 0;
212        for (; reservedCount < range.PageCount; reservedCount++)
213        {
214            ulong frame = startFrame + reservedCount;
215            if (!ReserveSinglePage(frame))
216            {
217                for (ulong rollback = 0; rollback < reservedCount; rollback++) ReleaseReservedPage(startFrame + rollback);
218                status = PhysicalMemoryStatus.RangeNotFree;
219                return false;
220            }
221        }
222
223        ref ReservationRecord record = ref _reservations[recordIndex];
224        record.Token = PhysicalAllocatorInternals.NextToken(ref _nextToken);
225        record.StartFrame = startFrame;
226        record.PageCount = range.PageCount;
227        record.Purpose = purpose;
228        record.Active = 1;
229        reservation = PhysicalAllocatorInternals.CreateReservation(ref record);
230        status = PhysicalMemoryStatus.Success;
231        return true;
232    }

234    /// <summary>Releases every page of a live exact reservation and coalesces newly matching buddies.</summary>
235    /// <nova.when>Call only when the reserving subsystem has made the complete range reusable.</nova.when>
236    /// <nova.depends>Reservation token returned by this manager</nova.depends>
237    /// <returns><see langword="true"/> when the reservation token is live and released exactly once.</returns>
238    /// <example><code>bool ok = allocator.TryReleaseReservation(reservation, out PhysicalMemoryStatus status);</code></example>
239    public bool TryReleaseReservation(PhysicalReservation reservation, out PhysicalMemoryStatus status)
240    {
241        status = PhysicalMemoryStatus.NotInitialized;
242        if (!_initialized) return false;
243        if (!PhysicalAllocatorInternals.TryFindReservationRecord(_reservations, _reservationCapacity, reservation, out int recordIndex))
244        {
245            status = PhysicalMemoryStatus.ReservationNotFound;
246            return false;
247        }
248        ReservationRecord record = _reservations[recordIndex];
249        for (ulong index = 0; index < record.PageCount; index++) ReleaseReservedPage(record.StartFrame + index);
250        _reservations[recordIndex].Active = 0;
251        status = PhysicalMemoryStatus.Success;
252        return true;

255    /// <summary>Returns current buddy block, allocation, reservation, and internal-fragmentation accounting.</summary>
256    /// <nova.when>Use for diagnostics and allocator comparison.</nova.when>
257    /// <nova.depends>Current buddy state and record tables</nova.depends>
258    /// <returns>A current physical-memory statistics snapshot.</returns>
259    /// <example><code>PhysicalMemoryStatistics statistics = allocator.GetStatistics();</code></example>
260    public readonly PhysicalMemoryStatistics GetStatistics()
261    {
262        if (!_initialized) return default;
263        ulong freePages = 0;
264        ulong largest = 0;
265        for (ulong offset = 0; offset < _frameSpan; offset++)
266        {
267            sbyte state = _states[offset];
268            if (state <= 0) continue;
269            ulong blockPages = PhysicalAllocatorInternals.BlockPages(state - 1);
270            freePages += blockPages;
271            if (blockPages > largest) largest = blockPages;
272        }
273        CountRecords(out ulong allocatedPages, out ulong reservedPages, out int allocations, out int reservations);
274        return PhysicalMemoryStatistics.Create(_totalManagedPages, freePages, allocatedPages, reservedPages, largest, allocations, reservations);
275    }

277    private void SeedRange(ulong startFrame, ulong pageCount)
278    {
279        ulong current = startFrame;
280        ulong remaining = pageCount;
281        while (remaining > 0)
282        {
283            int order = PhysicalAllocatorInternals.FloorLog2(remaining);
284            while (order > 0)
285            {
286                ulong pages = PhysicalAllocatorInternals.BlockPages(order);
287                if ((current & (pages - 1UL)) == 0) break;
288                order--;
289            }
290            ulong blockPages = PhysicalAllocatorInternals.BlockPages(order);
291            _states[current - _minimumFrame] = (sbyte)(order + 1);
292            current += blockPages;
293            remaining -= blockPages;
294        }
295    }

297    private void CoalesceSeededBlocks()
298    {
299        for (int order = 0; order < 62; order++)
300        {
301            bool merged;
302            do
303            {
304                merged = false;
305                ulong pages = PhysicalAllocatorInternals.BlockPages(order);
306                for (ulong offset = 0; offset < _frameSpan; offset++)
307                {
308                    if (_states[offset] != order + 1) continue;
309                    ulong frame = _minimumFrame + offset;
310                    ulong buddyFrame = frame ^ pages;
311                    if (buddyFrame < _minimumFrame || buddyFrame - _minimumFrame >= _frameSpan) continue;
312                    ulong buddyOffset = buddyFrame - _minimumFrame;
313                    if (_states[buddyOffset] != order + 1) continue;
314                    ulong lowerFrame = frame < buddyFrame ? frame : buddyFrame;
315                    ulong lowerOffset = lowerFrame - _minimumFrame;
316                    ulong upperOffset = lowerOffset + pages;
317                    _states[lowerOffset] = (sbyte)(order + 2);
318                    _states[upperOffset] = 0;
319                    merged = true;
320                }
321            } while (merged);
322        }
323    }

325    private bool SplitToTarget(ulong blockStart, int blockOrder, ulong targetStart, int targetOrder)
326    {
327        ulong currentStart = blockStart;
328        int currentOrder = blockOrder;
329        while (currentOrder > targetOrder)
330        {
331            ulong currentOffset = currentStart - _minimumFrame;
332            if (_states[currentOffset] != currentOrder + 1) return false;
333            _states[currentOffset] = 0;
334            currentOrder--;
335            ulong childPages = PhysicalAllocatorInternals.BlockPages(currentOrder);
336            ulong rightStart = currentStart + childPages;
337            ulong leftOffset = currentStart - _minimumFrame;
338            ulong rightOffset = rightStart - _minimumFrame;
339            _states[leftOffset] = (sbyte)(currentOrder + 1);
340            _states[rightOffset] = (sbyte)(currentOrder + 1);
341            if (targetStart >= rightStart) currentStart = rightStart;
342        }
343        return currentStart == targetStart && _states[targetStart - _minimumFrame] == targetOrder + 1;
344    }

346    private readonly bool HasFreeBlockAtLeast(int targetOrder)
347    {
348        for (ulong offset = 0; offset < _frameSpan; offset++)
349        {
350            if (_states[offset] > 0 && _states[offset] - 1 >= targetOrder) return true;
351        }
352        return false;

355    private bool ReserveSinglePage(ulong frame)
356    {
357        if (frame < _minimumFrame || frame - _minimumFrame >= _frameSpan) return false;
358        for (int order = 0; order <= 62; order++)
359        {
360            ulong pages = PhysicalAllocatorInternals.BlockPages(order);
361            ulong candidate = frame & ~(pages - 1UL);
362            if (candidate < _minimumFrame || candidate - _minimumFrame >= _frameSpan) continue;
363            ulong offset = candidate - _minimumFrame;
364            if (_states[offset] != order + 1) continue;
365            if (!SplitToTarget(candidate, order, frame, 0)) return false;
366            _states[frame - _minimumFrame] = ReservedPageState;
367            return true;
368        }
369        return false;
370    }

372    private void ReleaseReservedPage(ulong frame)
373    {
374        ulong offset = frame - _minimumFrame;
375        if (_states[offset] != ReservedPageState) return;
376        _states[offset] = 1;
377        Coalesce(frame, 0);
378    }

380    private void Coalesce(ulong frame, int order)
381    {
382        ulong currentFrame = frame;
383        int currentOrder = order;
384        while (currentOrder < 62)
385        {
386            ulong pages = PhysicalAllocatorInternals.BlockPages(currentOrder);
387            ulong buddyFrame = currentFrame ^ pages;
388            if (buddyFrame < _minimumFrame || buddyFrame - _minimumFrame >= _frameSpan) break;
389            ulong currentOffset = currentFrame - _minimumFrame;
390            ulong buddyOffset = buddyFrame - _minimumFrame;
391            if (_states[buddyOffset] != currentOrder + 1) break;
392            _states[currentOffset] = 0;
393            _states[buddyOffset] = 0;
394            currentFrame = currentFrame < buddyFrame ? currentFrame : buddyFrame;
395            currentOrder++;
396            _states[currentFrame - _minimumFrame] = (sbyte)(currentOrder + 1);
397        }
398    }

400    private void ClearRecords()
401    {
402        for (int index = 0; index < _allocationCapacity; index++) _allocations[index].Active = 0;
403        for (int index = 0; index < _reservationCapacity; index++) _reservations[index].Active = 0;
404    }

406    private readonly void CountRecords(out ulong allocatedPages, out ulong reservedPages, out int allocations, out int reservations)
407    {
408        allocatedPages = 0;
409        reservedPages = 0;
410        allocations = 0;
411        reservations = 0;
412        for (int index = 0; index < _allocationCapacity; index++)
413        {
414            if (_allocations[index].Active == 0) continue;
415            allocatedPages += _allocations[index].ActualPages;
416            allocations++;
417        }
418        for (int index = 0; index < _reservationCapacity; index++)
419        {
420            if (_reservations[index].Active == 0) continue;
421            reservedPages += _reservations[index].PageCount;
422            reservations++;
423        }
424    }
