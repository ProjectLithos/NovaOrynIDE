34    /// <summary>Calculates bitmap and record-table metadata required for a normalised map.</summary>
35    /// <nova.when>Call before reserving caller-owned bitmap metadata memory.</nova.when>
36    /// <nova.depends>Highest and lowest immediately allocatable physical frame</nova.depends>
37    /// <returns><see langword="true"/> when the sparse frame span and record tables can be represented.</returns>
38    /// <example><code>bool sized = BitmapPhysicalMemoryManager.TryGetRequiredWorkspaceBytes(map, 128, 32, out ulong bytes);</code></example>
39    public static bool TryGetRequiredWorkspaceBytes(NormalisedMemoryMap map, int allocationCapacity, int reservationCapacity, out ulong bytes)
40    {
41        bytes = 0;
42        if (!PhysicalAllocatorInternals.TryGetManagedFrameBounds(map, out ulong minimumFrame, out ulong maximumFrameExclusive, out _, out _)) return false;
43        ulong span = maximumFrameExclusive - minimumFrame;
44        if (span > ulong.MaxValue - 7UL) return false;
45        ulong bitmapBytes = (span + 7UL) / 8UL;
46        if (!PhysicalAllocatorInternals.TryAlignWorkspaceBytes(bitmapBytes, out ulong primaryBytes)) return false;
47        return PhysicalAllocatorInternals.TryWorkspaceBytes(primaryBytes, allocationCapacity, reservationCapacity, out bytes);
48    }

50    /// <summary>Initialises the bitmap to unavailable, then marks only normalised immediately-allocatable frames free.</summary>
51    /// <nova.when>Call once after memory-map normalisation and before virtual-memory construction.</nova.when>
52    /// <nova.depends>Caller-owned PhysicalAllocatorWorkspace</nova.depends>
53    /// <returns><see langword="true"/> when the bitmap and bounded record tables fit the workspace.</returns>
54    /// <example><code>bool ready = allocator.TryInitialize(map, workspace, 128, 32, out PhysicalMemoryStatus status);</code></example>
55    public bool TryInitialize(NormalisedMemoryMap map, PhysicalAllocatorWorkspace workspace, int allocationCapacity, int reservationCapacity, out PhysicalMemoryStatus status)
56    {
57        status = PhysicalMemoryStatus.InvalidParameter;
58        if (_initialized)
59        {
60            status = PhysicalMemoryStatus.AlreadyInitialized;
61            return false;
62        }
63        if (!PhysicalAllocatorInternals.TryGetManagedFrameBounds(map, out ulong minimumFrame, out ulong maximumFrameExclusive, out ulong managedPages, out _)) return false;
64        if (!TryGetRequiredWorkspaceBytes(map, allocationCapacity, reservationCapacity, out ulong requiredBytes)) return false;
65        ulong span = maximumFrameExclusive - minimumFrame;
66        ulong bitmapBytes = (span + 7UL) / 8UL;
67        if (!PhysicalAllocatorInternals.TryAlignWorkspaceBytes(bitmapBytes, out ulong primaryBytes)) return false;
68        if (workspace.ByteLength < requiredBytes || bitmapBytes > long.MaxValue || !PhysicalAllocatorInternals.TryPartition(workspace, primaryBytes, allocationCapacity, reservationCapacity, out byte* primary, out AllocationRecord* allocations, out ReservationRecord* reservations))
69        {
70            status = PhysicalMemoryStatus.WorkspaceTooSmall;
71            return false;
72        }
73
74        _bitmap = primary;
75        _allocations = allocations;
76        _reservations = reservations;
77        _minimumFrame = minimumFrame;
78        _frameSpan = span;
79        _totalManagedPages = managedPages;
80        _allocationCapacity = allocationCapacity;
81        _reservationCapacity = reservationCapacity;
82        _nextToken = 0;
83        for (ulong index = 0; index < bitmapBytes; index++) _bitmap[index] = 0xFF;
84        ClearRecords();
85
86        for (int index = 0; index < map.Count; index++)
87        {
88            if (!map.TryGetDescriptor(index, out MemoryDescriptor descriptor)) return false;
89            if (descriptor.Availability != MemoryAvailability.AvailableAfterExitBootServices) continue;
90            ulong startFrame = descriptor.PhysicalStart.Value / PageSize;
91            SetRange(startFrame, descriptor.PageCount, false);
92        }
93        _initialized = true;
94        status = PhysicalMemoryStatus.Success;
95        return true;
96    }

98    /// <summary>Allocates an exact contiguous run of free bitmap frames.</summary>
99    /// <nova.when>Use for arbitrary page counts with page or power-of-two alignment constraints.</nova.when>
100    /// <nova.depends>Successful TryInitialize</nova.depends>
101    /// <returns><see langword="true"/> when a matching free run is marked owned and recorded.</returns>
102    /// <example><code>bool ok = allocator.TryAllocate(request, out PhysicalAllocation allocation, out PhysicalMemoryStatus status);</code></example>
103    public bool TryAllocate(PhysicalAllocationRequest request, out PhysicalAllocation allocation, out PhysicalMemoryStatus status)
104    {
105        allocation = default;
106        status = PhysicalMemoryStatus.NotInitialized;
107        if (!_initialized) return false;
108        if (!PhysicalAllocatorInternals.TryGetRequestFrameWindow(request, out ulong minimumFrame, out ulong maximumFrameExclusive))
109        {
110            status = PhysicalMemoryStatus.InvalidParameter;
111            return false;
112        }
113        if (!PhysicalAllocatorInternals.TryFindFreeAllocationRecord(_allocations, _allocationCapacity, out int recordIndex))
114        {
115            status = PhysicalMemoryStatus.RecordCapacityExhausted;
116            return false;
117        }
118
119        ulong spanEnd = _minimumFrame + _frameSpan;
120        ulong scanStart = minimumFrame > _minimumFrame ? minimumFrame : _minimumFrame;
121        ulong scanEnd = maximumFrameExclusive < spanEnd ? maximumFrameExclusive : spanEnd;
122        if (scanStart < scanEnd && PhysicalAllocatorInternals.TryAlignFrame(scanStart, request.AlignmentPages, out ulong candidate))
123        {
124            while (candidate < scanEnd)
125            {
126                if (PhysicalAllocatorInternals.FitsWindow(candidate, request.PageCount, scanStart, scanEnd) && IsRangeFree(candidate, request.PageCount))
127                {
128                    SetRange(candidate, request.PageCount, true);
129                    ref AllocationRecord record = ref _allocations[recordIndex];
130                    record.Token = PhysicalAllocatorInternals.NextToken(ref _nextToken);
131                    record.StartFrame = candidate;
132                    record.RequestedPages = request.PageCount;
133                    record.ActualPages = request.PageCount;
134                    record.Purpose = request.Purpose;
135                    record.Active = 1;
136                    allocation = PhysicalAllocatorInternals.CreateAllocation(ref record);
137                    status = PhysicalMemoryStatus.Success;
138                    return true;
139                }
140                if (candidate > ulong.MaxValue - request.AlignmentPages) break;
141                candidate += request.AlignmentPages;
142            }
143        }
144        status = HasAnyFreeRun(request.PageCount) ? PhysicalMemoryStatus.AddressConstraintUnsatisfied : PhysicalMemoryStatus.OutOfMemory;
145        return false;
146    }

148    /// <summary>Releases a live bitmap allocation after validating its opaque token and exact range.</summary>
149    /// <nova.when>Call when the allocation owner has stopped using the pages.</nova.when>
150    /// <nova.depends>Allocation token returned by this manager</nova.depends>
151    /// <returns><see langword="true"/> when the live allocation is cleared exactly once.</returns>
152    /// <example><code>bool ok = allocator.TryRelease(allocation, out PhysicalMemoryStatus status);</code></example>
153    public bool TryRelease(PhysicalAllocation allocation, out PhysicalMemoryStatus status)
154    {
155        status = PhysicalMemoryStatus.NotInitialized;
156        if (!_initialized) return false;
157        if (!PhysicalAllocatorInternals.TryFindAllocationRecord(_allocations, _allocationCapacity, allocation, out int recordIndex))
158        {
159            status = PhysicalMemoryStatus.AllocationNotFound;
160            return false;
161        }
162        AllocationRecord record = _allocations[recordIndex];
163        SetRange(record.StartFrame, record.ActualPages, false);
164        _allocations[recordIndex].Active = 0;
165        status = PhysicalMemoryStatus.Success;
166        return true;
167    }

169    /// <summary>Marks an exact currently-free bitmap range reserved.</summary>
170    /// <nova.when>Use for late fixed-range exclusions such as hardware or metadata discoveries.</nova.when>
171    /// <nova.depends>PhysicalFrameRange</nova.depends>
172    /// <returns><see langword="true"/> when every frame is managed and currently free.</returns>
173    /// <example><code>bool ok = allocator.TryReserve(range, PhysicalMemoryPurpose.Metadata, out PhysicalReservation reservation, out PhysicalMemoryStatus status);</code></example>
174    public bool TryReserve(PhysicalFrameRange range, PhysicalMemoryPurpose purpose, out PhysicalReservation reservation, out PhysicalMemoryStatus status)
175    {
176        reservation = default;
177        status = PhysicalMemoryStatus.NotInitialized;
178        if (!_initialized) return false;
179        if (range.PageCount == 0 || (range.Start.Value & 0xFFFUL) != 0)
180        {
181            status = PhysicalMemoryStatus.InvalidParameter;
182            return false;
183        }
184        if (!PhysicalAllocatorInternals.TryFindFreeReservationRecord(_reservations, _reservationCapacity, out int recordIndex))
185        {
186            status = PhysicalMemoryStatus.RecordCapacityExhausted;
187            return false;
188        }
189        ulong startFrame = range.Start.Value / PageSize;
190        if (!IsRangeFree(startFrame, range.PageCount))
191        {
192            status = PhysicalMemoryStatus.RangeNotFree;
193            return false;
194        }
195        SetRange(startFrame, range.PageCount, true);
196        ref ReservationRecord record = ref _reservations[recordIndex];
197        record.Token = PhysicalAllocatorInternals.NextToken(ref _nextToken);
198        record.StartFrame = startFrame;
199        record.PageCount = range.PageCount;
200        record.Purpose = purpose;
201        record.Active = 1;
202        reservation = PhysicalAllocatorInternals.CreateReservation(ref record);
203        status = PhysicalMemoryStatus.Success;
204        return true;
205    }

228    /// <summary>Returns current bitmap allocator page accounting and largest free run.</summary>
229    /// <nova.when>Use for diagnostics and physical-memory pressure reporting.</nova.when>
230    /// <nova.depends>Current bitmap and record tables</nova.depends>
231    /// <returns>A current physical-memory statistics snapshot.</returns>
232    /// <example><code>PhysicalMemoryStatistics statistics = allocator.GetStatistics();</code></example>
233    public readonly PhysicalMemoryStatistics GetStatistics()
234    {
235        if (!_initialized) return default;
236        ulong freePages = 0;
237        ulong largest = 0;
238        ulong run = 0;
239        for (ulong offset = 0; offset < _frameSpan; offset++)
240        {
241            if (!IsUsedOffset(offset))
242            {
243                freePages++;
244                run++;
245                if (run > largest) largest = run;
246            }
247            else run = 0;
248        }
249        CountRecords(out ulong allocatedPages, out ulong reservedPages, out int allocations, out int reservations);
250        return PhysicalMemoryStatistics.Create(_totalManagedPages, freePages, allocatedPages, reservedPages, largest, allocations, reservations);
251    }

253    private readonly bool IsRangeFree(ulong startFrame, ulong pageCount)
254    {
255        if (pageCount == 0 || startFrame < _minimumFrame) return false;
256        ulong offset = startFrame - _minimumFrame;
257        if (offset > _frameSpan || pageCount > _frameSpan - offset) return false;
258        for (ulong index = 0; index < pageCount; index++)
259        {
260            if (IsUsedOffset(offset + index)) return false;
261        }
262        return true;
263    }
264
265    private readonly bool HasAnyFreeRun(ulong pageCount)
266    {
267        ulong run = 0;
268        for (ulong offset = 0; offset < _frameSpan; offset++)
269        {
270            if (!IsUsedOffset(offset))
271            {
272                run++;
273                if (run >= pageCount) return true;
274            }
275            else run = 0;
276        }
277        return false;
278    }

280    private readonly bool IsUsedOffset(ulong offset)
281    {
282        ulong byteIndex = offset >> 3;
283        int bit = (int)(offset & 7UL);
284        return (_bitmap[byteIndex] & (1 << bit)) != 0;
285    }
286
287    private void SetRange(ulong startFrame, ulong pageCount, bool used)
288    {
289        ulong offset = startFrame - _minimumFrame;
290        for (ulong index = 0; index < pageCount; index++)
291        {
292            ulong current = offset + index;
293            ulong byteIndex = current >> 3;
294            byte mask = (byte)(1 << (int)(current & 7UL));
295            if (used) _bitmap[byteIndex] |= mask;
296            else _bitmap[byteIndex] &= (byte)~mask;
297        }
298    }

300    private void ClearRecords()
301    {
302        for (int index = 0; index < _allocationCapacity; index++) _allocations[index].Active = 0;
303        for (int index = 0; index < _reservationCapacity; index++) _reservations[index].Active = 0;
304    }

306    private readonly void CountRecords(out ulong allocatedPages, out ulong reservedPages, out int allocations, out int reservations)
307    {
308        allocatedPages = 0;
309        reservedPages = 0;
310        allocations = 0;
311        reservations = 0;
312        for (int index = 0; index < _allocationCapacity; index++)
313        {
314            if (_allocations[index].Active == 0) continue;
315            allocatedPages += _allocations[index].ActualPages;
316            allocations++;
317        }
318        for (int index = 0; index < _reservationCapacity; index++)
319        {
320            if (_reservations[index].Active == 0) continue;
321            reservedPages += _reservations[index].PageCount;
322            reservations++;
323        }
324    }
