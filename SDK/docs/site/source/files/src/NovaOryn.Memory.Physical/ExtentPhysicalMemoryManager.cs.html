34    /// <summary>Calculates the metadata bytes required for the supplied map and record capacities.</summary>
35    /// <nova.when>Call before reserving the caller-owned metadata workspace.</nova.when>
36    /// <nova.depends>NormalisedMemoryMap descriptor count</nova.depends>
37    /// <returns><see langword="true"/> when the required byte count can be represented.</returns>
38    /// <example><code>bool sized = ExtentPhysicalMemoryManager.TryGetRequiredWorkspaceBytes(map, 128, 32, out ulong bytes);</code></example>
39    public static bool TryGetRequiredWorkspaceBytes(NormalisedMemoryMap map, int allocationCapacity, int reservationCapacity, out ulong bytes)
40    {
41        bytes = 0;
42        if (!PhysicalAllocatorInternals.TryGetManagedFrameBounds(map, out _, out _, out _, out int usableDescriptors)) return false;
43        if (allocationCapacity < 1 || reservationCapacity < 0) return false;
44        ulong extentCapacity = (ulong)usableDescriptors + (ulong)allocationCapacity + (ulong)reservationCapacity + 2UL;
45        if (extentCapacity > int.MaxValue) return false;
46        ulong primaryBytes = extentCapacity * (ulong)sizeof(ExtentRecord);
47        return PhysicalAllocatorInternals.TryWorkspaceBytes(primaryBytes, allocationCapacity, reservationCapacity, out bytes);
48    }

50    /// <summary>Initialises the extent manager from immediately allocatable normalised ranges.</summary>
51    /// <nova.when>Call once after memory-map normalisation and before the VMM or kernel heap exists.</nova.when>
52    /// <nova.depends>Caller-owned PhysicalAllocatorWorkspace</nova.depends>
53    /// <returns><see langword="true"/> when all free extents and record tables fit the supplied workspace.</returns>
54    /// <example><code>bool ready = allocator.TryInitialize(map, workspace, 128, 32, out PhysicalMemoryStatus status);</code></example>
55    public bool TryInitialize(NormalisedMemoryMap map, PhysicalAllocatorWorkspace workspace, int allocationCapacity, int reservationCapacity, out PhysicalMemoryStatus status)
56    {
57        status = PhysicalMemoryStatus.InvalidParameter;
58        if (_initialized)
59        {
60            status = PhysicalMemoryStatus.AlreadyInitialized;
61            return false;
62        }
63        if (!PhysicalAllocatorInternals.TryGetManagedFrameBounds(map, out _, out _, out ulong managedPages, out int usableDescriptors)) return false;
64        if (!TryGetRequiredWorkspaceBytes(map, allocationCapacity, reservationCapacity, out ulong requiredBytes)) return false;
65        ulong extentCapacityValue = (ulong)usableDescriptors + (ulong)allocationCapacity + (ulong)reservationCapacity + 2UL;
66        ulong primaryBytes = extentCapacityValue * (ulong)sizeof(ExtentRecord);
67        if (workspace.ByteLength < requiredBytes || !PhysicalAllocatorInternals.TryPartition(workspace, primaryBytes, allocationCapacity, reservationCapacity, out byte* primary, out AllocationRecord* allocations, out ReservationRecord* reservations))
68        {
69            status = PhysicalMemoryStatus.WorkspaceTooSmall;
70            return false;
71        }
72
73        _extents = (ExtentRecord*)primary;
74        _allocations = allocations;
75        _reservations = reservations;
76        _extentCount = 0;
77        _extentCapacity = (int)extentCapacityValue;
78        _allocationCapacity = allocationCapacity;
79        _reservationCapacity = reservationCapacity;
80        _totalManagedPages = managedPages;
81        _nextToken = 0;
82        ClearRecords();
83
84        for (int index = 0; index < map.Count; index++)
85        {
86            if (!map.TryGetDescriptor(index, out MemoryDescriptor descriptor)) return false;
87            if (descriptor.Availability != MemoryAvailability.AvailableAfterExitBootServices) continue;
88            if (!InsertFreeRange(descriptor.PhysicalStart.Value / PageSize, descriptor.PageCount))
89            {
90                status = PhysicalMemoryStatus.WorkspaceTooSmall;
91                return false;
92            }
93        }
94        _initialized = true;
95        status = PhysicalMemoryStatus.Success;
96        return true;
97    }

99    /// <summary>Allocates an exact-size contiguous range satisfying the supplied alignment and address window.</summary>
100    /// <nova.when>Use when an exact physical page count is preferred.</nova.when>
101    /// <nova.depends>Successful TryInitialize</nova.depends>
102    /// <returns><see langword="true"/> when a matching extent was split and recorded.</returns>
103    /// <example><code>bool ok = allocator.TryAllocate(request, out PhysicalAllocation allocation, out PhysicalMemoryStatus status);</code></example>
104    public bool TryAllocate(PhysicalAllocationRequest request, out PhysicalAllocation allocation, out PhysicalMemoryStatus status)
105    {
106        allocation = default;
107        status = PhysicalMemoryStatus.NotInitialized;
108        if (!_initialized) return false;
109        if (!PhysicalAllocatorInternals.TryGetRequestFrameWindow(request, out ulong minimumFrame, out ulong maximumFrameExclusive))
110        {
111            status = PhysicalMemoryStatus.InvalidParameter;
112            return false;
113        }
114        if (!PhysicalAllocatorInternals.TryFindFreeAllocationRecord(_allocations, _allocationCapacity, out int recordIndex))
115        {
116            status = PhysicalMemoryStatus.RecordCapacityExhausted;
117            return false;
118        }
119
120        bool sizeCandidateExists = false;
121        for (int index = 0; index < _extentCount; index++)
122        {
123            ExtentRecord extent = _extents[index];
124            if (extent.PageCount < request.PageCount) continue;
125            sizeCandidateExists = true;
126            ulong firstFrame = extent.StartFrame > minimumFrame ? extent.StartFrame : minimumFrame;
127            if (!PhysicalAllocatorInternals.TryAlignFrame(firstFrame, request.AlignmentPages, out ulong candidateFrame)) continue;
128            if (!PhysicalAllocatorInternals.FitsWindow(candidateFrame, request.PageCount, minimumFrame, maximumFrameExclusive)) continue;
129            if (candidateFrame < extent.StartFrame || candidateFrame - extent.StartFrame > extent.PageCount) continue;
130            ulong offset = candidateFrame - extent.StartFrame;
131            if (request.PageCount > extent.PageCount - offset) continue;
132            if (!TakeRange(index, candidateFrame, request.PageCount))
133            {
134                status = PhysicalMemoryStatus.WorkspaceTooSmall;
135                return false;
136            }
137
138            ref AllocationRecord record = ref _allocations[recordIndex];
139            record.Token = PhysicalAllocatorInternals.NextToken(ref _nextToken);
140            record.StartFrame = candidateFrame;
141            record.RequestedPages = request.PageCount;
142            record.ActualPages = request.PageCount;
143            record.Purpose = request.Purpose;
144            record.Active = 1;
145            allocation = PhysicalAllocatorInternals.CreateAllocation(ref record);
146            status = PhysicalMemoryStatus.Success;
147            return true;
148        }
149
150        status = sizeCandidateExists ? PhysicalMemoryStatus.AddressConstraintUnsatisfied : PhysicalMemoryStatus.OutOfMemory;
151        return false;
152    }

154    /// <summary>Releases an exact live extent allocation and merges adjacent free ranges.</summary>
155    /// <nova.when>Call when the owning subsystem no longer uses the physical pages.</nova.when>
156    /// <nova.depends>Allocation token returned by this manager</nova.depends>
157    /// <returns><see langword="true"/> when the token was live and the pages were returned once.</returns>
158    /// <example><code>bool ok = allocator.TryRelease(allocation, out PhysicalMemoryStatus status);</code></example>
159    public bool TryRelease(PhysicalAllocation allocation, out PhysicalMemoryStatus status)
160    {
161        status = PhysicalMemoryStatus.NotInitialized;
162        if (!_initialized) return false;
163        if (!PhysicalAllocatorInternals.TryFindAllocationRecord(_allocations, _allocationCapacity, allocation, out int recordIndex))
164        {
165            status = PhysicalMemoryStatus.AllocationNotFound;
166            return false;
167        }
168        AllocationRecord record = _allocations[recordIndex];
169        if (!InsertFreeRange(record.StartFrame, record.ActualPages))
170        {
171            status = PhysicalMemoryStatus.WorkspaceTooSmall;
172            return false;
173        }
174        _allocations[recordIndex].Active = 0;
175        status = PhysicalMemoryStatus.Success;
176        return true;
177    }

179    /// <summary>Reserves an exact currently-free physical range and splits the containing extent.</summary>
180    /// <nova.when>Use for late-discovered hardware or allocator metadata reservations.</nova.when>
181    /// <nova.depends>PhysicalFrameRange</nova.depends>
182    /// <returns><see langword="true"/> when the whole range was free and a reservation token was recorded.</returns>
183    /// <example><code>bool ok = allocator.TryReserve(range, PhysicalMemoryPurpose.Metadata, out PhysicalReservation reservation, out PhysicalMemoryStatus status);</code></example>
184    public bool TryReserve(PhysicalFrameRange range, PhysicalMemoryPurpose purpose, out PhysicalReservation reservation, out PhysicalMemoryStatus status)
185    {
186        reservation = default;
187        status = PhysicalMemoryStatus.NotInitialized;
188        if (!_initialized) return false;
189        if (range.PageCount == 0 || (range.Start.Value & 0xFFFUL) != 0)
190        {
191            status = PhysicalMemoryStatus.InvalidParameter;
192            return false;
193        }
194        if (!PhysicalAllocatorInternals.TryFindFreeReservationRecord(_reservations, _reservationCapacity, out int recordIndex))
195        {
196            status = PhysicalMemoryStatus.RecordCapacityExhausted;
197            return false;
198        }
199        ulong startFrame = range.Start.Value / PageSize;
200        for (int index = 0; index < _extentCount; index++)
201        {
202            ExtentRecord extent = _extents[index];
203            if (startFrame < extent.StartFrame) break;
204            ulong offset = startFrame - extent.StartFrame;
205            if (offset > extent.PageCount || range.PageCount > extent.PageCount - offset) continue;
206            if (!TakeRange(index, startFrame, range.PageCount))
207            {
208                status = PhysicalMemoryStatus.WorkspaceTooSmall;
209                return false;
210            }
211            ref ReservationRecord record = ref _reservations[recordIndex];
212            record.Token = PhysicalAllocatorInternals.NextToken(ref _nextToken);
213            record.StartFrame = startFrame;
214            record.PageCount = range.PageCount;
215            record.Purpose = purpose;
216            record.Active = 1;
217            reservation = PhysicalAllocatorInternals.CreateReservation(ref record);
218            status = PhysicalMemoryStatus.Success;
219            return true;
220        }
221        status = PhysicalMemoryStatus.RangeNotFree;
222        return false;
223    }

251    /// <nova.when>Use for diagnostics and allocator comparison.</nova.when>
252    /// <nova.depends>Current extent and record tables</nova.depends>

255    public readonly PhysicalMemoryStatistics GetStatistics()
256    {
257        if (!_initialized) return default;
258        ulong freePages = 0;
259        ulong largest = 0;
260        for (int index = 0; index < _extentCount; index++)
261        {
262            freePages += _extents[index].PageCount;
263            if (_extents[index].PageCount > largest) largest = _extents[index].PageCount;
264        }

266        return PhysicalMemoryStatistics.Create(_totalManagedPages, freePages, allocatedPages, reservedPages, largest, allocations, reservations);

269    private bool TakeRange(int index, ulong startFrame, ulong pageCount)
270    {
271        ExtentRecord extent = _extents[index];
272        ulong before = startFrame - extent.StartFrame;
273        ulong afterStart = startFrame + pageCount;
274        ulong extentEnd = extent.StartFrame + extent.PageCount;
275        ulong after = extentEnd - afterStart;
276        if (before > 0 && after > 0)
277        {
278            if (_extentCount >= _extentCapacity) return false;
279            for (int move = _extentCount; move > index + 1; move--) _extents[move] = _extents[move - 1];
280            _extents[index] = new ExtentRecord(extent.StartFrame, before);
281            _extents[index + 1] = new ExtentRecord(afterStart, after);
282            _extentCount++;
283        }
284        else if (before > 0)
285        {
286            _extents[index] = new ExtentRecord(extent.StartFrame, before);
287        }
288        else if (after > 0)
289        {
290            _extents[index] = new ExtentRecord(afterStart, after);
291        }
292        else
293        {
294            for (int move = index; move < _extentCount - 1; move++) _extents[move] = _extents[move + 1];
295            _extentCount--;
296        }
297        return true;
298    }

300    private bool InsertFreeRange(ulong startFrame, ulong pageCount)
301    {
302        if (pageCount == 0 || _extentCount >= _extentCapacity) return false;
303        int insert = 0;
304        while (insert < _extentCount && _extents[insert].StartFrame < startFrame) insert++;
305        for (int move = _extentCount; move > insert; move--) _extents[move] = _extents[move - 1];
306        _extents[insert] = new ExtentRecord(startFrame, pageCount);
307        _extentCount++;
308        MergeAround(insert);
309        return true;
310    }
311
312    private void MergeAround(int index)
313    {
314        if (index > 0 && _extents[index - 1].StartFrame + _extents[index - 1].PageCount == _extents[index].StartFrame)
315        {
316            _extents[index - 1].PageCount += _extents[index].PageCount;
317            for (int move = index; move < _extentCount - 1; move++) _extents[move] = _extents[move + 1];
318            _extentCount--;
319            index--;
320        }
321        if (index + 1 < _extentCount && _extents[index].StartFrame + _extents[index].PageCount == _extents[index + 1].StartFrame)
322        {
323            _extents[index].PageCount += _extents[index + 1].PageCount;
324            for (int move = index + 1; move < _extentCount - 1; move++) _extents[move] = _extents[move + 1];
325            _extentCount--;
326        }
327    }

329    private void ClearRecords()
330    {
331        for (int index = 0; index < _allocationCapacity; index++) _allocations[index].Active = 0;
332        for (int index = 0; index < _reservationCapacity; index++) _reservations[index].Active = 0;
333    }

335    private readonly void CountRecords(out ulong allocatedPages, out ulong reservedPages, out int allocations, out int reservations)
336    {
337        allocatedPages = 0;
338        reservedPages = 0;
339        allocations = 0;
340        reservations = 0;
341        for (int index = 0; index < _allocationCapacity; index++)
342        {
343            if (_allocations[index].Active == 0) continue;
344            allocatedPages += _allocations[index].ActualPages;
345            allocations++;
346        }
347        for (int index = 0; index < _reservationCapacity; index++)
348        {
349            if (_reservations[index].Active == 0) continue;
350            reservedPages += _reservations[index].PageCount;
351            reservations++;
352        }
353    }
