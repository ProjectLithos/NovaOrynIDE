10    internal static bool TryGetManagedFrameBounds(NormalisedMemoryMap map, out ulong minimumFrame, out ulong maximumFrameExclusive, out ulong managedPages, out int usableDescriptorCount)
11    {
12        minimumFrame = ulong.MaxValue;
13        maximumFrameExclusive = 0;
14        managedPages = 0;
15        usableDescriptorCount = 0;
16        if (map is null || map.Count < 1) return false;
17
18        for (int index = 0; index < map.Count; index++)
19        {
20            if (!map.TryGetDescriptor(index, out MemoryDescriptor descriptor)) return false;
21            if (descriptor.Availability != MemoryAvailability.AvailableAfterExitBootServices) continue;
22            ulong startFrame = descriptor.PhysicalStart.Value / PageSize;
23            if (descriptor.PageCount > ulong.MaxValue - startFrame) return false;
24            ulong endFrame = startFrame + descriptor.PageCount;
25            if (startFrame < minimumFrame) minimumFrame = startFrame;
26            if (endFrame > maximumFrameExclusive) maximumFrameExclusive = endFrame;
27            if (managedPages > ulong.MaxValue - descriptor.PageCount) return false;
28            managedPages += descriptor.PageCount;
29            usableDescriptorCount++;
30        }
31
32        return usableDescriptorCount > 0 && minimumFrame < maximumFrameExclusive;
33    }

35    internal static bool TryGetRequestFrameWindow(PhysicalAllocationRequest request, out ulong minimumFrame, out ulong maximumFrameExclusive)
36    {
37        minimumFrame = request.MinimumAddress.Value / PageSize;
38        maximumFrameExclusive = request.MaximumAddressExclusive.Value == 0
39            ? (ulong.MaxValue / PageSize) + 1UL
40            : request.MaximumAddressExclusive.Value / PageSize;
41        if (request.PageCount == 0 || request.AlignmentPages == 0) return false;
42        if ((request.AlignmentPages & (request.AlignmentPages - 1)) != 0) return false;
43        return maximumFrameExclusive > minimumFrame;
44    }

46    internal static bool TryAlignFrame(ulong frame, ulong alignmentPages, out ulong alignedFrame)
47    {
48        alignedFrame = 0;
49        if (alignmentPages == 0 || (alignmentPages & (alignmentPages - 1)) != 0) return false;
50        ulong mask = alignmentPages - 1;
51        if (frame > ulong.MaxValue - mask) return false;
52        alignedFrame = (frame + mask) & ~mask;
53        return true;

57    {

78    internal static int CeilLog2(ulong value)
79    {
80        if (value <= 1) return 0;
81        int order = 0;
82        ulong current = 1;
83        while (current < value && order < 63)
84        {
85            current <<= 1;

113    internal static bool TryWorkspaceBytes(ulong primaryBytes, int allocationCapacity, int reservationCapacity, out ulong requiredBytes)
114    {
115        requiredBytes = 0;
116        if (allocationCapacity < 1 || reservationCapacity < 0) return false;
117        ulong allocationBytes = (ulong)allocationCapacity * (ulong)sizeof(AllocationRecord);
118        ulong reservationBytes = (ulong)reservationCapacity * (ulong)sizeof(ReservationRecord);
119        if (primaryBytes > ulong.MaxValue - allocationBytes) return false;
120        ulong total = primaryBytes + allocationBytes;
121        if (total > ulong.MaxValue - reservationBytes) return false;
122        requiredBytes = total + reservationBytes;
123        return true;
124    }

126    internal static bool TryPartition(PhysicalAllocatorWorkspace workspace, ulong primaryBytes, int allocationCapacity, int reservationCapacity, out byte* primary, out AllocationRecord* allocations, out ReservationRecord* reservations)
127    {
128        primary = null;
129        allocations = null;
130        reservations = null;
131        if (!TryWorkspaceBytes(primaryBytes, allocationCapacity, reservationCapacity, out ulong requiredBytes)) return false;
132        if (workspace.Address == 0 || workspace.ByteLength < requiredBytes || requiredBytes > long.MaxValue) return false;
133        byte* baseAddress = (byte*)workspace.Address;
134        primary = baseAddress;
135        allocations = (AllocationRecord*)(baseAddress + (long)primaryBytes);
136        ulong allocationBytes = (ulong)allocationCapacity * (ulong)sizeof(AllocationRecord);
137        reservations = (ReservationRecord*)(baseAddress + (long)(primaryBytes + allocationBytes));
138        return true;
139    }
