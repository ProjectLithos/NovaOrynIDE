48    internal static unsafe class SpanHelpers
49    {
50        internal static void ClearWithoutReferences(ref Byte destination, UIntPtr byteCount)
51        {
52            fixed (Byte* pointer = &destination)
53            {
54                UInt64 length = (UInt64)(nuint)byteCount;
55                for (UInt64 index = 0; index < length; index++) pointer[index] = 0;
56            }
57        }
58
59        internal static void Memmove(ref Byte destination, ref Byte source, UIntPtr byteCount)
60        {
61            fixed (Byte* destinationPointer = &destination)
62            fixed (Byte* sourcePointer = &source)
63            {
64                UInt64 length = (UInt64)(nuint)byteCount;
65                if (length == 0UL || destinationPointer == sourcePointer) return;
66
67                UInt64 destinationAddress = (UInt64)(nuint)destinationPointer;
68                UInt64 sourceAddress = (UInt64)(nuint)sourcePointer;
69                if (destinationAddress < sourceAddress || destinationAddress - sourceAddress >= length)
70                {
71                    for (UInt64 index = 0; index < length; index++) destinationPointer[index] = sourcePointer[index];
72                    return;
73                }
74
75                for (UInt64 index = length; index != 0UL; index--) destinationPointer[index - 1UL] = sourcePointer[index - 1UL];
76            }
77        }
