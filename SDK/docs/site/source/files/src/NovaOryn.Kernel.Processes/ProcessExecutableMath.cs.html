14    /// <summary>Inspects one executable image and returns its validated entry metadata.</summary>
15    public static Boolean TryInspect(Byte* image, UInt64 length, out ProcessExecutableInfo info)
16    {
17        info=default;
18        if (image==null || length<64UL) return false;
19        if (image[0]==0x7F && image[1]==(Byte)'E' && image[2]==(Byte)'L' && image[3]==(Byte)'F') return TryInspectElf(image,length,out info);
20        if (image[0]==(Byte)'M' && image[1]==(Byte)'Z') return TryInspectPe(image,length,out info);
21        return false;
22    }

26    {
27        segment=default;

40    private static Boolean TryInspectElf(Byte* p, UInt64 n, out ProcessExecutableInfo info)
41    {
42        info=default;
43        if (n<64UL || p[4]!=2 || p[5]!=1 || Read16(p,18)!=ElfMachineX64) return false;
44        UInt16 type=Read16(p,16); if(type!=2 && type!=3) return false;
45        UInt64 entry=Read64(p,24), phoff=Read64(p,32); UInt16 phsize=Read16(p,54), phcount=Read16(p,56);
46        if (type!=2 || phsize<56 || phcount==0 || !Range(phoff,(UInt64)phsize*phcount,n) || !KernelProtectionMath.IsValidUserEntry(entry)) return false;
47        UInt32 loads=0U;
48        for(UInt32 i=0;i<phcount;i++) if(Read32(p,phoff+(UInt64)i*phsize)==ElfLoad) loads++;
49        if(loads==0U || loads>MaximumSegments) return false;
50        Boolean entryExecutable=false; for(UInt32 i=0;i<loads;i++){if(!TryGetElfLoad(p,n,i,out ProcessImageSegment segment)) return false; if((segment.Protection&ProcessSegmentProtection.Execute)!=0 && entry>=segment.VirtualAddress && entry-segment.VirtualAddress<segment.MemorySize)entryExecutable=true;}
51        if(!entryExecutable)return false;
52        info=new ProcessExecutableInfo(ProcessExecutableFormat.Elf64,entry,0UL,loads,false); return true;
53    }

55    private static Boolean TryGetElfLoad(Byte* p, UInt64 n, UInt32 wanted, out ProcessImageSegment segment)
56    {
57        segment=default; UInt64 phoff=Read64(p,32); UInt16 phsize=Read16(p,54), phcount=Read16(p,56); UInt32 found=0U;
58        for(UInt32 i=0;i<phcount;i++)
59        {
60            UInt64 o=phoff+(UInt64)i*phsize; if(Read32(p,o)!=ElfLoad) continue;
61            if(found++!=wanted) continue;
62            UInt32 flags=Read32(p,o+4); UInt64 fileOffset=Read64(p,o+8), vaddr=Read64(p,o+16), fileSize=Read64(p,o+32), memSize=Read64(p,o+40);
63            if(memSize==0UL || fileSize>memSize || !Range(fileOffset,fileSize,n) || !KernelProtectionMath.IsUserRange(vaddr,memSize)) return false;
64            ProcessSegmentProtection protection=ProcessSegmentProtection.Read;
65            if((flags&2U)!=0U) protection|=ProcessSegmentProtection.Write; if((flags&1U)!=0U) protection|=ProcessSegmentProtection.Execute;
66            segment=new ProcessImageSegment(vaddr,memSize,fileSize,fileOffset,protection); return true;
67        }
68        return false;

72    {
73        info=default; if(n<0x40UL) return false; UInt32 pe=Read32(p,0x3CUL);
74        if(!Range(pe,24UL,n) || Read32(p,pe)!=0x00004550U || Read16(p,pe+4)!=PeMachineX64) return false;
75        UInt16 sections=Read16(p,pe+6), optionalSize=Read16(p,pe+20); UInt64 opt=(UInt64)pe+24UL;
76        if(sections==0 || sections>MaximumSegments || optionalSize<112 || !Range(opt,optionalSize,n) || Read16(p,opt)!=0x020B) return false;
77        UInt32 entryRva=Read32(p,opt+16), sizeOfImage=Read32(p,opt+56); UInt64 imageBase=Read64(p,opt+24), entry=imageBase+entryRva;
78        if(entry<imageBase || sizeOfImage==0U || !KernelProtectionMath.IsValidUserEntry(entry)) return false;
79        UInt64 table=opt+optionalSize; if(!Range(table,(UInt64)sections*40UL,n)) return false;
80        ProcessExecutableInfo temp=new(ProcessExecutableFormat.PortableExecutable64,entry,imageBase,sections,false);
81        Boolean entryExecutable=false; for(UInt32 i=0;i<sections;i++){if(!TryGetPeSection(p,n,i,out ProcessImageSegment segment)) return false; if((segment.Protection&ProcessSegmentProtection.Execute)!=0 && entry>=segment.VirtualAddress && entry-segment.VirtualAddress<segment.MemorySize)entryExecutable=true;}
82        if(!entryExecutable)return false; info=temp; return true;
83    }

86    {
87        segment=default; UInt32 pe=Read32(p,0x3CUL); UInt16 sections=Read16(p,pe+6), optionalSize=Read16(p,pe+20); if(index>=sections)return false;

89        if(!Range(o,40UL,n)) return false;

91        UInt64 memorySize=virtualSize>rawSize?virtualSize:rawSize; if(memorySize==0UL) return false;
92        UInt64 vaddr=imageBase+rva; if(vaddr<imageBase || !Range(rawOffset,rawSize,n) || !KernelProtectionMath.IsUserRange(vaddr,memorySize)) return false;
93        ProcessSegmentProtection protection=ProcessSegmentProtection.Read;
94        if((c&0x80000000U)!=0U) protection|=ProcessSegmentProtection.Write; if((c&0x20000000U)!=0U) protection|=ProcessSegmentProtection.Execute;
95        segment=new ProcessImageSegment(vaddr,memorySize,rawSize,rawOffset,protection); return true;
96    }
