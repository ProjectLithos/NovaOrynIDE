9    /// <summary>Initializes the x64 COM1 serial device used by the managed console.</summary>
10    public static Boolean InitializeSerial()
11    {
12        if (!WritePort8(0x3F9, 0x00)) return false;
13        if (!WritePort8(0x3FB, 0x80)) return false;
14        if (!WritePort8(0x3F8, 0x01)) return false;
15        if (!WritePort8(0x3F9, 0x00)) return false;
16        if (!WritePort8(0x3FB, 0x03)) return false;
17        if (!WritePort8(0x3FA, 0xC7)) return false;
18        return WritePort8(0x3FC, 0x0B);
19    }
