181    /// <summary>Initializes virtual memory by attaching to and protecting the active x64 page-table hierarchy.</summary>
182    /// <returns><see langword="true"/> when the active root and all reachable page-table pages were accepted.</returns>
183    public static Boolean Initialize()
184    {
185        if (_initialized) return SetFailure(KernelVirtualMemoryStatus.AlreadyInitialized);
186        if (!KernelPhysicalMemory.IsInitialized()) return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
187        KernelPhysicalMemoryStatistics physicalStatistics = KernelPhysicalMemory.GetStatistics();
188        if (physicalStatistics.LiveAllocationCount != 0) return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
189        UInt64 root = Native.ReadPageTableRoot() & AddressMask4KiB;
190        if (root == 0UL || (root & 0xFFFUL) != 0UL) return SetFailure(KernelVirtualMemoryStatus.ArchitectureOperationFailed);
191        ResetState();
192        _rootPhysicalAddress = root;
193        if (!CaptureDirectMapPlan()) return false;
194        _executeDisableEnabled = Native.EnableExecuteDisable();
195        _page1GiBSupported = Native.Supports1GiBPages();
196        if (!ProtectTableHierarchy(root, 4)) return false;
197        _initialized = true;
198        _lastStatus = KernelVirtualMemoryStatus.Success;
199        return true;
200    }

202    /// <summary>Installs one physical-to-virtual leaf mapping into the active x64 address space.</summary>
203    /// <returns><see langword="true"/> when the leaf was installed and its translation invalidated.</returns>
204    public static Boolean TryMap(UInt64 virtualAddress, UInt64 physicalAddress, KernelVirtualPageSize pageSize, KernelVirtualMemoryProtection protection)
205    {
206        if (!_initialized) return SetFailure(KernelVirtualMemoryStatus.NotInitialized);
207        UInt64 size = (UInt64)pageSize;
208        if (!IsSupportedPageSize(size)) return SetFailure(KernelVirtualMemoryStatus.UnsupportedPageSize);
209        if (pageSize == KernelVirtualPageSize.Page1GiB && !_page1GiBSupported) return SetFailure(KernelVirtualMemoryStatus.UnsupportedPageSize);
210        if (!HasProtection(protection, KernelVirtualMemoryProtection.Execute) && !_executeDisableEnabled) return SetFailure(KernelVirtualMemoryStatus.UnsupportedProtection);
211        if (!IsCanonical(virtualAddress)) return SetFailure(KernelVirtualMemoryStatus.NonCanonicalAddress);
212        if ((virtualAddress & (size - 1UL)) != 0UL || (physicalAddress & (size - 1UL)) != 0UL)
213            return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
214        if ((physicalAddress & ~LeafAddressMask(pageSize)) != 0UL) return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
215        if (!HasProtection(protection, KernelVirtualMemoryProtection.Read)) return SetFailure(KernelVirtualMemoryStatus.UnsupportedProtection);
216
217        if (!IsTableWritableBeforeDirectMap(_rootPhysicalAddress)) return SetFailure(KernelVirtualMemoryStatus.ArchitectureOperationFailed);
218        UInt64* pml4 = TablePointer(_rootPhysicalAddress);
219        Int32 pml4Index = (Int32)((virtualAddress >> 39) & 0x1FFUL);
220        Int32 pdptIndex = (Int32)((virtualAddress >> 30) & 0x1FFUL);
221        Int32 pdIndex = (Int32)((virtualAddress >> 21) & 0x1FFUL);
222        Int32 ptIndex = (Int32)((virtualAddress >> 12) & 0x1FFUL);
223
224        if (!TryGetOrCreateChild(pml4, pml4Index, protection, out UInt64* pdpt)) return false;
225        if (pageSize == KernelVirtualPageSize.Page1GiB)
226            return TryInstallLeaf(pdpt, pdptIndex, virtualAddress, physicalAddress, pageSize, protection);
227        if (IsLarge(pdpt[pdptIndex])) return SetFailure(KernelVirtualMemoryStatus.AlreadyMapped);
228        if (!TryGetOrCreateChild(pdpt, pdptIndex, protection, out UInt64* pd)) return false;
229        if (pageSize == KernelVirtualPageSize.Page2MiB)
230            return TryInstallLeaf(pd, pdIndex, virtualAddress, physicalAddress, pageSize, protection);
231        if (IsLarge(pd[pdIndex])) return SetFailure(KernelVirtualMemoryStatus.AlreadyMapped);
232        if (!TryGetOrCreateChild(pd, pdIndex, protection, out UInt64* pt)) return false;
233        return TryInstallLeaf(pt, ptIndex, virtualAddress, physicalAddress, pageSize, protection);
234    }

249    /// <summary>Replaces access and cache protection on the present leaf mapping covering one virtual address.</summary>
250    /// <returns><see langword="true"/> when the leaf was rewritten without changing its physical target.</returns>
251    public static Boolean TryProtect(UInt64 virtualAddress, KernelVirtualMemoryProtection protection)
252    {
253        if (!_initialized) return SetFailure(KernelVirtualMemoryStatus.NotInitialized);
254        if (!HasProtection(protection, KernelVirtualMemoryProtection.Read)) return SetFailure(KernelVirtualMemoryStatus.UnsupportedProtection);
255        if (!HasProtection(protection, KernelVirtualMemoryProtection.Execute) && !_executeDisableEnabled) return SetFailure(KernelVirtualMemoryStatus.UnsupportedProtection);
256        if (!TryFindLeaf(virtualAddress, out UInt64* table, out Int32 index, out KernelVirtualPageSize pageSize)) return false;
257        UInt64 current = table[index];
258        UInt64 physicalAddress = current & LeafAddressMask(pageSize);
259        if (!TryEncodeLeaf(physicalAddress, pageSize, protection, out UInt64 replacement))
260            return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
261        table[index] = replacement;
262        if (!Native.InvalidatePage(virtualAddress)) return SetFailure(KernelVirtualMemoryStatus.ArchitectureOperationFailed);
263        _lastStatus = KernelVirtualMemoryStatus.Success;
264        return true;
265    }

268    /// <returns><see langword="true"/> when a present leaf covers the queried address.</returns>
269    public static Boolean TryTranslate(UInt64 virtualAddress, out KernelVirtualTranslation translation)
270    {
271        translation = default;
272        if (!_initialized) return SetFailure(KernelVirtualMemoryStatus.NotInitialized);
273        if (!TryFindLeaf(virtualAddress, out UInt64* table, out Int32 index, out KernelVirtualPageSize pageSize)) return false;
274        UInt64 entry = table[index];
275        UInt64 size = (UInt64)pageSize;
276        UInt64 physicalBase = entry & LeafAddressMask(pageSize);
277        UInt64 physicalAddress = physicalBase + (virtualAddress & (size - 1UL));

279        _lastStatus = KernelVirtualMemoryStatus.Success;

286    private static Boolean TryGetOrCreateChild(UInt64* table, Int32 index, KernelVirtualMemoryProtection protection, out UInt64* child)
287    {
288        child = (UInt64*)0;
289        UInt64 entry = table[index];
290        if (IsPresent(entry))
291        {
292            if (IsLarge(entry)) return SetFailure(KernelVirtualMemoryStatus.AlreadyMapped);
293            if (HasProtection(protection, KernelVirtualMemoryProtection.Write) && (entry & Writable) == 0UL)
294                return SetFailure(KernelVirtualMemoryStatus.UnsupportedProtection);
295            if (HasProtection(protection, KernelVirtualMemoryProtection.User) && (entry & User) == 0UL)
296                return SetFailure(KernelVirtualMemoryStatus.UnsupportedProtection);
297            if (HasProtection(protection, KernelVirtualMemoryProtection.Execute) && (entry & NoExecute) != 0UL)
298                return SetFailure(KernelVirtualMemoryStatus.UnsupportedProtection);
299            UInt64 childPhysicalAddress = entry & AddressMask4KiB;
300            if (!IsTableWritableBeforeDirectMap(childPhysicalAddress))
301                return SetFailure(KernelVirtualMemoryStatus.ArchitectureOperationFailed);
302            child = TablePointer(childPhysicalAddress);
303            return true;
304        }
305
306        if (!TryAllocatePageTable(out UInt64 createdPhysicalAddress))
307            return SetFailure(KernelVirtualMemoryStatus.PhysicalAllocationFailed);
308        UInt64* created = TablePointer(createdPhysicalAddress);
309        for (Int32 i = 0; i < 512; i++) created[i] = 0UL;
310        table[index] = (createdPhysicalAddress & AddressMask4KiB) | Present | Writable | User;
311        _createdPageTables++;
312        child = created;
313        return true;
314    }

316    private static Boolean TryInstallLeaf(UInt64* table, Int32 index, UInt64 virtualAddress, UInt64 physicalAddress, KernelVirtualPageSize pageSize, KernelVirtualMemoryProtection protection)
317    {
318        if (IsPresent(table[index])) return SetFailure(KernelVirtualMemoryStatus.AlreadyMapped);
319        if (!TryEncodeLeaf(physicalAddress, pageSize, protection, out UInt64 entry))
320            return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
321        table[index] = entry;
322        IncrementMapped(pageSize);
323        if (!Native.InvalidatePage(virtualAddress)) return SetFailure(KernelVirtualMemoryStatus.ArchitectureOperationFailed);
324        _lastStatus = KernelVirtualMemoryStatus.Success;
325        return true;
326    }

328    private static Boolean TryFindLeaf(UInt64 virtualAddress, out UInt64* table, out Int32 index, out KernelVirtualPageSize pageSize)
329    {
330        table = (UInt64*)0;
331        index = 0;
332        pageSize = KernelVirtualPageSize.Page4KiB;
333        if (!IsCanonical(virtualAddress)) return SetFailure(KernelVirtualMemoryStatus.NonCanonicalAddress);
334        UInt64* pml4 = TablePointer(_rootPhysicalAddress);
335        UInt64 pml4Entry = pml4[(Int32)((virtualAddress >> 39) & 0x1FFUL)];
336        if (!IsPresent(pml4Entry) || IsLarge(pml4Entry)) return SetFailure(KernelVirtualMemoryStatus.NotMapped);
337        UInt64* pdpt = TablePointer(pml4Entry & AddressMask4KiB);
338        Int32 pdptIndex = (Int32)((virtualAddress >> 30) & 0x1FFUL);
339        UInt64 pdptEntry = pdpt[pdptIndex];
340        if (!IsPresent(pdptEntry)) return SetFailure(KernelVirtualMemoryStatus.NotMapped);
341        if (IsLarge(pdptEntry))
342        {
343            table = pdpt;
344            index = pdptIndex;
345            pageSize = KernelVirtualPageSize.Page1GiB;
346            return true;
347        }
348        UInt64* pd = TablePointer(pdptEntry & AddressMask4KiB);
349        Int32 pdIndex = (Int32)((virtualAddress >> 21) & 0x1FFUL);
350        UInt64 pdEntry = pd[pdIndex];
351        if (!IsPresent(pdEntry)) return SetFailure(KernelVirtualMemoryStatus.NotMapped);
352        if (IsLarge(pdEntry))
353        {
354            table = pd;
355            index = pdIndex;
356            pageSize = KernelVirtualPageSize.Page2MiB;
357            return true;
358        }
359        UInt64* pt = TablePointer(pdEntry & AddressMask4KiB);
360        Int32 ptIndex = (Int32)((virtualAddress >> 12) & 0x1FFUL);
361        if (!IsPresent(pt[ptIndex])) return SetFailure(KernelVirtualMemoryStatus.NotMapped);
362        table = pt;
363        index = ptIndex;
364        pageSize = KernelVirtualPageSize.Page4KiB;
365        return true;
366    }

368    private static Boolean ProtectTableHierarchy(UInt64 physicalAddress, Int32 level)
369    {
370        if (WasProtected(physicalAddress)) return true;
371        if (_protectedTableCount >= MaximumProtectedTables) return SetFailure(KernelVirtualMemoryStatus.PageTableCapacityExhausted);
372        if (!KernelPhysicalMemory.TryExcludePage(physicalAddress)) return SetFailure(KernelVirtualMemoryStatus.PhysicalAllocationFailed);
373        fixed (UInt64* protectedTables = _state.ProtectedTables)
374        {
375            protectedTables[_protectedTableCount] = physicalAddress;
376            _protectedTableCount++;
377        }
378        if (level <= 1) return true;
379        UInt64* table = TablePointer(physicalAddress);
380        for (Int32 index = 0; index < 512; index++)
381        {
382            UInt64 entry = table[index];
383            if (!IsPresent(entry)) continue;
384            if ((level == 3 || level == 2) && IsLarge(entry)) continue;
385            UInt64 child = entry & AddressMask4KiB;
386            if (child == 0UL) return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
387            if (!ProtectTableHierarchy(child, level - 1)) return false;
388        }
389        return true;
390    }

402    private static Boolean TryEncodeLeaf(UInt64 physicalAddress, KernelVirtualPageSize pageSize, KernelVirtualMemoryProtection protection, out UInt64 entry)
403    {
404        entry = 0UL;
405        UInt64 mask = LeafAddressMask(pageSize);
406        UInt64 size = (UInt64)pageSize;
407        if (mask == 0UL || (physicalAddress & (size - 1UL)) != 0UL || (physicalAddress & ~mask) != 0UL) return false;
408        UInt64 flags = Present;
409        if (HasProtection(protection, KernelVirtualMemoryProtection.Write)) flags |= Writable;
410        if (HasProtection(protection, KernelVirtualMemoryProtection.User)) flags |= User;
411        if (HasProtection(protection, KernelVirtualMemoryProtection.Global)) flags |= Global;
412        if (HasProtection(protection, KernelVirtualMemoryProtection.Device)) flags |= CacheDisable;
413        if (HasProtection(protection, KernelVirtualMemoryProtection.WriteThrough)) flags |= WriteThrough;
414        if (!HasProtection(protection, KernelVirtualMemoryProtection.Execute)) flags |= NoExecute;
415        if (pageSize != KernelVirtualPageSize.Page4KiB) flags |= LargePage;
416        entry = (physicalAddress & mask) | flags;
417        return true;
418    }

420    private static KernelVirtualMemoryProtection DecodeProtection(UInt64 entry)
421    {
422        KernelVirtualMemoryProtection protection = KernelVirtualMemoryProtection.Read;
423        if ((entry & Writable) != 0UL) protection = protection | KernelVirtualMemoryProtection.Write;
424        if ((entry & User) != 0UL) protection = protection | KernelVirtualMemoryProtection.User;
425        if ((entry & Global) != 0UL) protection = protection | KernelVirtualMemoryProtection.Global;
426        if ((entry & CacheDisable) != 0UL) protection = protection | KernelVirtualMemoryProtection.Device;
427        if ((entry & WriteThrough) != 0UL) protection = protection | KernelVirtualMemoryProtection.WriteThrough;
428        if ((entry & NoExecute) == 0UL) protection = protection | KernelVirtualMemoryProtection.Execute;
429        return protection;
430    }

458    private static Boolean IncrementMapped(KernelVirtualPageSize pageSize)
459    {
460        if (pageSize == KernelVirtualPageSize.Page4KiB) _mapped4KiB++;
461        else if (pageSize == KernelVirtualPageSize.Page2MiB) _mapped2MiB++;
462        else _mapped1GiB++;

466    private static Boolean DecrementMapped(KernelVirtualPageSize pageSize)
467    {
468        if (pageSize == KernelVirtualPageSize.Page4KiB && _mapped4KiB != 0UL) _mapped4KiB--;
469        else if (pageSize == KernelVirtualPageSize.Page2MiB && _mapped2MiB != 0UL) _mapped2MiB--;
470        else if (pageSize == KernelVirtualPageSize.Page1GiB && _mapped1GiB != 0UL) _mapped1GiB--;
471        return true;
