35    /// <summary>Builds the direct map from the PMM free extents derived from the final normalized UEFI memory map.</summary>
36    /// <param name="virtualBase">Virtual base reserved for the physical direct map.</param>
37    /// <param name="virtualLength">Maximum physical span representable by that reservation.</param>
38    /// <returns><see langword="true"/> when every currently allocatable PMM range is directly mapped.</returns>
39    public static Boolean InitializeDirectMap(UInt64 virtualBase, UInt64 virtualLength)
40    {
41        if (!_initialized) return SetFailure(KernelVirtualMemoryStatus.NotInitialized);
42        if (_directMapReady) return SetFailure(KernelVirtualMemoryStatus.AlreadyInitialized);
43        if (!IsCanonical(virtualBase) || virtualBase < 0xFFFF800000000000UL || virtualLength == 0UL || (virtualBase & 0xFFFUL) != 0UL || (virtualLength & 0xFFFUL) != 0UL)
44            return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
45
46        if (_directMapExtentCount <= 0) return SetFailure(KernelVirtualMemoryStatus.PhysicalAllocationFailed);
47        if (!AdoptPrivateRoot()) return false;
48
49        UInt64 physicalLimit = 0UL;
50        fixed (UInt64* starts = _directMapState.Starts)
51        fixed (UInt64* pages = _directMapState.Pages)
52        {
53            for (Int32 index = 0; index < _directMapExtentCount; index++)
54            {
55                UInt64 start = starts[index];
56                UInt64 count = pages[index];
57                UInt64 bytes = count * 4096UL;
58                UInt64 end = start + bytes;
59                if (end > virtualLength) return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
60                if (end > physicalLimit) physicalLimit = end;
61            }
62
63            _directMapBase = virtualBase;
64            for (Int32 index = 0; index < _directMapExtentCount; index++)
65                if (!MapDirectExtent(virtualBase, starts[index], pages[index])) return false;
66
67            UInt64 workspaceAddress = KernelPhysicalMemory.GetBootstrapPageTableWorkspaceAddress();
68            UInt64 workspacePages = KernelPhysicalMemory.GetBootstrapPageTableWorkspacePages();
69            if (workspaceAddress == 0UL || workspacePages == 0UL || !MapDirectExtent(virtualBase, workspaceAddress, workspacePages))
70                return SetFailure(KernelVirtualMemoryStatus.PhysicalAllocationFailed);
71            UInt64 workspaceEnd = workspaceAddress + workspacePages * 4096UL;
72            if (workspaceEnd > virtualLength) return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
73            if (workspaceEnd > physicalLimit) physicalLimit = workspaceEnd;
74        }
75
76        _directMapPhysicalLimit = physicalLimit;
77        _directMapReady = true;
78        _lastStatus = KernelVirtualMemoryStatus.Success;
79        return true;
80    }

82    private static Boolean CaptureDirectMapPlan()
83    {
84        Int32 extentCount = KernelPhysicalMemory.GetFreeExtentCount();
85        if (extentCount <= 0 || extentCount > MaximumDirectMapExtents)
86            return SetFailure(KernelVirtualMemoryStatus.PhysicalAllocationFailed);
87        fixed (UInt64* starts = _directMapState.Starts)
88        fixed (UInt64* pages = _directMapState.Pages)
89        {
90            for (Int32 index = 0; index < extentCount; index++)
91            {
92                if (!KernelPhysicalMemory.TryGetFreeExtent(index, out UInt64 start, out UInt64 count))
93                    return SetFailure(KernelVirtualMemoryStatus.PhysicalAllocationFailed);
94                if (count == 0UL || count > 0xFFFFFFFFFFFFFFFFUL / 4096UL || start > 0xFFFFFFFFFFFFFFFFUL - count * 4096UL)
95                    return SetFailure(KernelVirtualMemoryStatus.InvalidParameter);
96                starts[index] = start;
97                pages[index] = count;
98            }
99        }
100        _directMapExtentCount = extentCount;
101        return true;
102    }

104    private static Boolean MapDirectExtent(UInt64 virtualBase, UInt64 physicalStart, UInt64 pageCount)
105    {
106        KernelVirtualMemoryProtection protection = KernelVirtualMemoryProtection.Read | KernelVirtualMemoryProtection.Write | KernelVirtualMemoryProtection.Global;
107        if (!_executeDisableEnabled) protection = protection | KernelVirtualMemoryProtection.Execute;
108        UInt64 physical = physicalStart;
109        UInt64 remaining = pageCount * 4096UL;
110        while (remaining != 0UL)
111        {
112            KernelVirtualPageSize pageSize = KernelVirtualPageSize.Page4KiB;
113            UInt64 bytes = 4096UL;
114            UInt64 virtualAddress = virtualBase + physical;
115            if (_page1GiBSupported && remaining >= 1073741824UL && (physical & 0x3FFFFFFFUL) == 0UL && (virtualAddress & 0x3FFFFFFFUL) == 0UL)
116            {
117                pageSize = KernelVirtualPageSize.Page1GiB;
118                bytes = 1073741824UL;
119            }
120            else if (remaining >= 2097152UL && (physical & 0x1FFFFFUL) == 0UL && (virtualAddress & 0x1FFFFFUL) == 0UL)
121            {
122                pageSize = KernelVirtualPageSize.Page2MiB;
123                bytes = 2097152UL;
124            }
125            if (!TryMap(virtualAddress, physical, pageSize, protection)) return false;
126            physical += bytes;
127            remaining -= bytes;
128        }
129        return true;
130    }

132    private static Boolean AdoptPrivateRoot()
133    {
134        UInt64 inheritedRoot = _rootPhysicalAddress;
135        UInt64* inherited = (UInt64*)(nuint)inheritedRoot;
136        if (!TryAllocatePageTable(out UInt64 ownedPhysicalAddress))
137            return SetFailure(KernelVirtualMemoryStatus.PhysicalAllocationFailed);
138        UInt64* owned = (UInt64*)(nuint)ownedPhysicalAddress;
139        for (Int32 index = 0; index < 512; index++) owned[index] = inherited[index];
140        if (!Native.WritePageTableRoot(ownedPhysicalAddress))
141            return SetFailure(KernelVirtualMemoryStatus.ArchitectureOperationFailed);
142        _rootPhysicalAddress = ownedPhysicalAddress;
143        _createdPageTables++;
144        return true;
145    }

147    private static Boolean TryAllocatePageTable(out UInt64 physicalAddress)
148    {
149        physicalAddress = 0UL;
150        if (_directMapReady)
151        {
152            if (!KernelPhysicalMemory.TryAllocate(1UL, 1UL, out KernelPhysicalAllocation allocation)) return false;
153            physicalAddress = allocation.StartAddress;
154            return true;
155        }
156        if (!KernelPhysicalMemory.TryTakeBootstrapPageTable(out physicalAddress)) return false;
157        _bootstrapTablePages++;
158        return true;
159    }

161    private static Boolean IsIdentityWritable(UInt64 physicalAddress)
162    {
163        UInt64* pml4 = (UInt64*)(nuint)_rootPhysicalAddress;
164        UInt64 pml4Entry = pml4[(Int32)((physicalAddress >> 39) & 0x1FFUL)];
165        if (!IsPresent(pml4Entry) || IsLarge(pml4Entry) || (pml4Entry & Writable) == 0UL) return false;
166        UInt64* pdpt = (UInt64*)(nuint)(pml4Entry & AddressMask4KiB);
167        UInt64 pdptEntry = pdpt[(Int32)((physicalAddress >> 30) & 0x1FFUL)];
168        if (!IsPresent(pdptEntry) || (pdptEntry & Writable) == 0UL) return false;
169        if (IsLarge(pdptEntry)) return ((pdptEntry & AddressMask1GiB) + (physicalAddress & 0x3FFFFFFFUL)) == physicalAddress;
170        UInt64* pd = (UInt64*)(nuint)(pdptEntry & AddressMask4KiB);
171        UInt64 pdEntry = pd[(Int32)((physicalAddress >> 21) & 0x1FFUL)];
172        if (!IsPresent(pdEntry) || (pdEntry & Writable) == 0UL) return false;
173        if (IsLarge(pdEntry)) return ((pdEntry & AddressMask2MiB) + (physicalAddress & 0x1FFFFFUL)) == physicalAddress;
174        UInt64* pt = (UInt64*)(nuint)(pdEntry & AddressMask4KiB);
175        UInt64 ptEntry = pt[(Int32)((physicalAddress >> 12) & 0x1FFUL)];
176        return IsPresent(ptEntry) && (ptEntry & Writable) != 0UL &&
177            ((ptEntry & AddressMask4KiB) + (physicalAddress & 0xFFFUL)) == physicalAddress;
178    }
