37    /// <summary>Writes a managed string without appending a line terminator.</summary>
38    public static Boolean Write(String value)
39    {
40        if (!_initialized || value == null) return false;
41        Int32 length = value.Length;
42        Int32 index = 0;
43        while (index < length)
44        {
45            Char character = value[index];
46            if ((UInt32)character > 0x7FU) character = (Char)'?';
47            if (!Write((Byte)character)) return false;
48            index++;
49        }
50        return true;
51    }

82    /// <summary>Writes one unsigned 64-bit value as an ordinary base-10 number.</summary>
83    public static unsafe Boolean WriteUInt64(UInt64 value)
84    {
85        Byte* digits = stackalloc Byte[20];
86        Int32 count = 0;
87        do
88        {
89            digits[count] = (Byte)('0' + (Byte)(value % 10UL));
90            value /= 10UL;
91            count++;
92        }
93        while (value != 0UL);
94
95        while (count > 0)
96        {
97            count--;
98            if (!Write(digits[count])) return false;
99        }
100        return true;
101    }

104    public static Boolean WriteByteSize(UInt64 bytes)
105    {
106        if (bytes >= 1099511627776UL) return WriteScaled(bytes, 1099511627776UL, " TiB");
107        if (bytes >= 1073741824UL) return WriteScaled(bytes, 1073741824UL, " GiB");
108        if (bytes >= 1048576UL) return WriteScaled(bytes, 1048576UL, " MiB");
109        if (bytes >= 1024UL) return WriteScaled(bytes, 1024UL, " KiB");
110        if (!WriteUInt64(bytes)) return false;
111        return Write(" B");
112    }

134    private static Boolean WriteScaled(UInt64 value, UInt64 divisor, String suffix)
135    {
136        UInt64 whole = value / divisor;
137        UInt64 remainder = value % divisor;
138        UInt64 hundredths = (remainder * 100UL) / divisor;
139        if (!WriteUInt64(whole)) return false;
140        if (hundredths != 0UL)
141        {
142            if (!Write(".")) return false;
143            if (hundredths < 10UL && !Write("0")) return false;
144            if (!WriteUInt64(hundredths)) return false;
145        }
146        return Write(suffix);
147    }

149    /// <summary>Writes one unsigned 64-bit value as a fixed-width hexadecimal number prefixed with 0x.</summary>
150    /// <returns><see langword="true"/> when every hexadecimal character was written.</returns>
151    public static Boolean WriteHex(UInt64 value)
152    {
153        if (!Write("0x")) return false;
154        Int32 shift = 60;
155        while (shift >= 0)
156        {
157            UInt32 nibble = (UInt32)((value >> shift) & 0xFUL);
158            Byte character = nibble < 10U ? (Byte)('0' + nibble) : (Byte)('A' + (nibble - 10U));
159            if (!Write(character)) return false;
160            shift -= 4;
161        }
162        return true;
163    }
