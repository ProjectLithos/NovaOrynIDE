345    private Boolean RedrawHistory()
346    {
347        if (!ClearPixels()) return false;
348        UInt32 totalLines = CountVisualLines();
349        UInt32 visibleLines = GetVisibleLineCount();
350        UInt32 maximumOffset = totalLines > visibleLines ? totalLines - visibleLines : 0U;
351        if (_scrollLinesFromBottom > maximumOffset) _scrollLinesFromBottom = maximumOffset;
352        UInt32 firstLine = totalLines > visibleLines + _scrollLinesFromBottom
353            ? totalLines - visibleLines - _scrollLinesFromBottom
354            : 0U;
355        UInt32 index = FindVisualLineStart(firstLine);
356        UInt32 x = _margin;
357        UInt32 y = _margin;
358        while (index < _historyLength)
359        {
360            Byte value = GetHistoryByte(index);
361            if (value == (Byte)'\n')
362            {
363                x = _margin;
364                if (y > _height - _fontSize || _lineHeight > (_height - _fontSize) - y) break;
365                y += _lineHeight;
366                index++;
367                continue;
368            }

370            {
371                x = _margin;
372                if (y > _height - _fontSize || _lineHeight > (_height - _fontSize) - y) break;
373                y += _lineHeight;
374            }
375            if (y > _height - _fontSize) break;
376            if (!DrawGlyph(value, x, y)) return false;

378            index++;

415    {

456    private Boolean DrawGlyph(Byte value, UInt32 originX, UInt32 originY)
457    {
458        UInt32 renderedRow = 0U;
459        while (renderedRow < _fontSize)
460        {
461            UInt32 sourceRow = BitmapFont.GetSourceRow(renderedRow, _fontSize);
462            UInt32 bits = BitmapFont.GetGlyphRow(value, sourceRow);
463            UInt32 renderedColumn = 0U;
464            while (renderedColumn < _glyphWidth)
465            {
466                UInt32 sourceColumn = BitmapFont.GetSourceColumn(renderedColumn, _glyphWidth);
467                UInt32 mask = 1U << (Int32)((BitmapFont.GlyphWidth - 1U) - sourceColumn);
468                if ((bits & mask) != 0U)
469                {
470                    if (!DrawPixel(originX + renderedColumn, originY + renderedRow)) return false;
471                }
472                renderedColumn++;
473            }
474            renderedRow++;
475        }
476        MarkDirtyRectangle(originX, originY, _glyphWidth, _fontSize);
477        return true;
478    }

516    {
517        if (sourceAddress == 0UL || destinationAddress == 0UL) return false;
518        if ((sourceAddress & 3UL) != 0UL || (destinationAddress & 3UL) != 0UL) return false;
519        if (left >= right || top >= bottom || right > _width || bottom > _height) return false;

522        UInt32 row = top;
523        while (row < bottom)
524        {
525            UInt64 rowStart = (UInt64)row * (UInt64)_pitch;
526            UInt64 start = rowStart + left;
527            UInt64 end = rowStart + right;
528            if (end > _size / 4UL) return false;

530            while (index < end)
531            {
532                destination[index] = source[index];

541    {

543        UInt64 right64 = (UInt64)left + width;
544        UInt64 bottom64 = (UInt64)top + height;

589    private UInt32 PackColor(Byte red, Byte green, Byte blue)
590    {
591        if (_pixelFormat == 0U) return (UInt32)red | ((UInt32)green << 8) | ((UInt32)blue << 16);
592        if (_pixelFormat == 1U) return (UInt32)blue | ((UInt32)green << 8) | ((UInt32)red << 16);
593        return EncodeMask(red, _redMask) | EncodeMask(green, _greenMask) | EncodeMask(blue, _blueMask);
594    }

596    private static Boolean IsContiguousMask(UInt32 mask)
597    {
598        while ((mask & 1U) == 0U) mask >>= 1;
599        while ((mask & 1U) != 0U) mask >>= 1;
600        return mask == 0U;
601    }

603    private static UInt32 EncodeMask(Byte component, UInt32 mask)
604    {
605        UInt32 shift = 0U;
606        while (((mask >> (Int32)shift) & 1U) == 0U && shift < 31U) shift++;
607        UInt32 shiftedMask = mask >> (Int32)shift;
608        UInt32 bits = 0U;
609        while ((shiftedMask & 1U) != 0U)
610        {
611            bits++;
612            shiftedMask >>= 1;
613        }
614        UInt64 maximum = bits == 32U ? 0xFFFFFFFFUL : ((1UL << (Int32)bits) - 1UL);
615        UInt32 encoded = (UInt32)(((UInt64)component * maximum) / 255UL);
616        return (encoded << (Int32)shift) & mask;
617    }
