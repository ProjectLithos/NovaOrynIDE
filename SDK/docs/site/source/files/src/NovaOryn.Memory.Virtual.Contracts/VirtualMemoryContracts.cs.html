94/// <summary>Describes one page-aligned virtual-address range.</summary>
95/// <nova.when>Use for unmapping, protection changes, and address-space diagnostics.</nova.when>
96/// <nova.depends>VirtualPageSize</nova.depends>
97[SupportedArchitecture(SupportedArchitecture.All)]
98[BootStage(BootStage.ManagedBootstrap)]
99public readonly struct VirtualAddressRange
100{
101    private readonly ulong _pageSizeBytes;
102
103    private VirtualAddressRange(ulong start, ulong pageCount, VirtualPageSize pageSize)
104    {
105        Start = start;
106        PageCount = pageCount;
107        PageSize = pageSize;
108        _pageSizeBytes = (ulong)pageSize;
109    }
110
111    /// <summary>Gets the first virtual byte in the range.</summary>
112    /// <nova.when>Use as the first mapping or protection-change address.</nova.when>
113    public ulong Start { get; }
114    /// <summary>Gets the number of pages in the range.</summary>
115    /// <nova.when>Use for range iteration and accounting.</nova.when>
116    public ulong PageCount { get; }
117    /// <summary>Gets the mapping granularity of every page in the range.</summary>
118    /// <nova.when>Use when walking or removing leaf entries.</nova.when>
119    public VirtualPageSize PageSize { get; }
120    /// <summary>Gets the byte length of the range.</summary>
121    /// <nova.when>Use for diagnostics and end-address calculations.</nova.when>
122    public ulong Length { get { return PageCount * _pageSizeBytes; } }
123
124    /// <summary>Attempts to create a validated aligned virtual-address range.</summary>
125    /// <nova.when>Use before unmapping or changing protection on a contiguous range.</nova.when>
126    /// <nova.depends>Power-of-two page size and checked address arithmetic</nova.depends>
127    /// <returns><see langword="true"/> when the range is non-empty, aligned, and does not overflow.</returns>
128    /// <example><code>bool valid = VirtualAddressRange.TryCreate(0xFFFF800000000000, 4, VirtualPageSize.Page4KiB, out VirtualAddressRange range);</code></example>
129    public static bool TryCreate(ulong start, ulong pageCount, VirtualPageSize pageSize, out VirtualAddressRange range)
130    {
131        range = default;
132        ulong size = (ulong)pageSize;
133        if (!IsSupportedSize(size) || pageCount == 0 || (start & (size - 1UL)) != 0) return false;
134        if (pageCount > ulong.MaxValue / size) return false;
135        ulong length = pageCount * size;
136        if (start > ulong.MaxValue - length) return false;
137        range = new VirtualAddressRange(start, pageCount, pageSize);
138        return true;
139    }
140
141    private static bool IsSupportedSize(ulong size) => size == 4096UL || size == 2097152UL || size == 1073741824UL;
142}

144/// <summary>Describes one physical-to-virtual mapping request.</summary>
145/// <nova.when>Use to add a leaf translation to an address space.</nova.when>
146/// <nova.depends>VirtualPageSize and VirtualMemoryProtection</nova.depends>
147[SupportedArchitecture(SupportedArchitecture.All)]
148[BootStage(BootStage.ManagedBootstrap)]
149public readonly struct VirtualMappingRequest
150{
151    private VirtualMappingRequest(ulong virtualAddress, PhysicalAddress physicalAddress, VirtualPageSize pageSize, VirtualMemoryProtection protection)
152    {
153        VirtualAddress = virtualAddress;
154        PhysicalAddress = physicalAddress;
155        PageSize = pageSize;
156        Protection = protection;
157    }
158
159    /// <summary>Gets the aligned virtual leaf address.</summary>
160    /// <nova.when>Use as the translation key.</nova.when>
161    public ulong VirtualAddress { get; }
162    /// <summary>Gets the aligned physical leaf address.</summary>
163    /// <nova.when>Use as the translation target.</nova.when>
164    public PhysicalAddress PhysicalAddress { get; }
165    /// <summary>Gets the requested leaf page size.</summary>
166    /// <nova.when>Use to select the architecture leaf level.</nova.when>
167    public VirtualPageSize PageSize { get; }
168    /// <summary>Gets architecture-neutral access and caching intent.</summary>
169    /// <nova.when>Use to encode the leaf entry.</nova.when>
170    public VirtualMemoryProtection Protection { get; }
171
172    /// <summary>Attempts to create a validated single-page mapping request.</summary>
173    /// <nova.when>Use before calling <see cref="IVirtualMemoryManager.TryMap"/>.</nova.when>
174    /// <nova.depends>Aligned physical and virtual addresses</nova.depends>
175    /// <returns><see langword="true"/> when both addresses are aligned and at least read access is requested.</returns>
176    /// <example><code>bool valid = VirtualMappingRequest.TryCreate(0xFFFF800000200000, new PhysicalAddress(0x200000), VirtualPageSize.Page2MiB, VirtualMemoryProtection.Read | VirtualMemoryProtection.Write, out VirtualMappingRequest request);</code></example>
177    public static bool TryCreate(ulong virtualAddress, PhysicalAddress physicalAddress, VirtualPageSize pageSize, VirtualMemoryProtection protection, out VirtualMappingRequest request)
178    {
179        request = default;
180        ulong size = (ulong)pageSize;
181        if (size != 4096UL && size != 2097152UL && size != 1073741824UL) return false;
182        if ((virtualAddress & (size - 1UL)) != 0 || (physicalAddress.Value & (size - 1UL)) != 0) return false;
183        if ((protection & VirtualMemoryProtection.Read) == 0) return false;
184        request = new VirtualMappingRequest(virtualAddress, physicalAddress, pageSize, protection);
185        return true;
186    }
187}
