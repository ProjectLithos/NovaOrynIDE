6/// <summary>Initialises and owns one caller-allocated x64 Task State Segment.</summary>
7public sealed unsafe class X64TaskStateSegment : ITaskStateSegment
8{
9    private const uint ArchitecturalSize = 104;
10    private TaskStateSegmentConfiguration configuration;
11    private bool configured;
12
13    /// <inheritdoc />
14    public ProcessorId GetProcessorId() => configuration.ProcessorId;
15
16    /// <inheritdoc />
17    public bool Configure(TaskStateSegmentConfiguration value)
18    {
19        if (value.TaskStateSegmentAddress.Value == 0 || value.TaskStateSegmentCapacity < ArchitecturalSize ||
20            value.RingZeroStackTop.Value == 0 || value.DoubleFaultStackTop.Value == 0 || value.NmiStackTop.Value == 0)
21        {
22            return false;
23        }
24        if (value.IoPermissionBitmapPolicy == IoPermissionBitmapPolicy.DenyByDefault &&
25            (value.IoPermissionBitmapAddress.Value == 0 || value.IoPermissionBitmapLength == 0))
26        {
27            return false;
28        }
29
30        configuration = value;
31        byte* tss = (byte*)value.TaskStateSegmentAddress.Value;
32        for (uint index = 0; index < ArchitecturalSize; index++) tss[index] = 0;
33        Write64(tss, 4, value.RingZeroStackTop.Value);
34        Write64(tss, 36, value.DoubleFaultStackTop.Value); // IST1
35        Write64(tss, 44, value.NmiStackTop.Value);         // IST2
36        if (value.MachineCheckStackTop.Value != 0) Write64(tss, 52, value.MachineCheckStackTop.Value); // IST3
37
38        ushort bitmapOffset = (ushort)ArchitecturalSize;
39        if (value.IoPermissionBitmapPolicy == IoPermissionBitmapPolicy.DenyByDefault)
40        {
41            ulong bitmapDistance = value.IoPermissionBitmapAddress.Value - value.TaskStateSegmentAddress.Value;
42            if (value.IoPermissionBitmapAddress.Value < value.TaskStateSegmentAddress.Value || bitmapDistance > ushort.MaxValue)
43                return false;
44            bitmapOffset = (ushort)bitmapDistance;
45            byte* bitmap = (byte*)value.IoPermissionBitmapAddress.Value;
46            for (uint index = 0; index < value.IoPermissionBitmapLength; index++) bitmap[index] = 0xFF;
47        }
48        Write16(tss, 102, bitmapOffset);
49        configured = true;
50        return true;
51    }
52
53    /// <inheritdoc />
54    public bool Install(SegmentSelector taskStateSelector) => configured && NativeMethods.LoadTaskRegister(taskStateSelector.Value);
55
56    /// <inheritdoc />
57    public bool SetRingZeroStack(Address stackTop)
58    {
59        if (!configured || stackTop.Value == 0) return false;
60        Write64((byte*)configuration.TaskStateSegmentAddress.Value, 4, stackTop.Value);
61        configuration = configuration with { RingZeroStackTop = stackTop };
62        return true;
63    }
64
65    /// <inheritdoc />
66    public bool SetInterruptStack(byte entry, Address stackTop)
67    {
68        if (!configured || entry is < 1 or > 7 || stackTop.Value == 0) return false;
69        Write64((byte*)configuration.TaskStateSegmentAddress.Value, (uint)(28 + entry * 8), stackTop.Value);
70        return true;
71    }
72
73    /// <inheritdoc />
74    public Address GetAddress() => configured ? configuration.TaskStateSegmentAddress : new Address(0);
75
76    /// <inheritdoc />
77    public uint GetLimit()
78    {
79        if (!configured) return 0;
80        if (configuration.IoPermissionBitmapPolicy != IoPermissionBitmapPolicy.DenyByDefault) return ArchitecturalSize - 1;
81        ulong length = configuration.IoPermissionBitmapAddress.Value - configuration.TaskStateSegmentAddress.Value + configuration.IoPermissionBitmapLength;
82        return length > uint.MaxValue ? 0 : (uint)length;
83    }
84
85    private static void Write16(byte* target, uint offset, ushort value) => *(ushort*)(target + offset) = value;
86    private static void Write64(byte* target, uint offset, ulong value) => *(ulong*)(target + offset) = value;
87}
