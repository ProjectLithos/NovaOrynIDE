6/// <summary>Builds and installs one processor-local x64 GDT.</summary>
7public sealed unsafe class X64GlobalDescriptorTable : IGlobalDescriptorTable
8{
9    private const uint RequiredBytes = 7 * 8;
10    private GlobalDescriptorTableConfiguration configuration;
11    private ITaskStateSegment? taskStateSegment;
12    private bool configured;
13
14    /// <inheritdoc />
15    public ProcessorId GetProcessorId() => configuration.ProcessorId;
16
17    /// <inheritdoc />
18    public bool Configure(GlobalDescriptorTableConfiguration value, ITaskStateSegment taskState)
19    {
20        if (value.TableAddress.Value == 0 || value.TableCapacity < RequiredBytes || taskState.GetAddress().Value == 0 || taskState.GetLimit() == 0)
21            return false;
22        if (value.KernelCodeSelector.Index != 1 || value.KernelDataSelector.Index != 2 ||
23            value.UserDataSelector.Index != 3 || value.UserCodeSelector.Index != 4 || value.TaskStateSelector.Index != 5)
24            return false;
25
26        configuration = value;
27        taskStateSegment = taskState;
28        ulong* table = (ulong*)value.TableAddress.Value;
29        table[0] = 0;
30        table[1] = 0x00AF9A000000FFFFUL; // kernel 64-bit code
31        table[2] = 0x00CF92000000FFFFUL; // kernel data
32        table[3] = 0x00CFF2000000FFFFUL; // user data
33        table[4] = 0x00AFFA000000FFFFUL; // user 64-bit code
34        WriteTaskStateDescriptor(table, 5, taskState.GetAddress().Value, taskState.GetLimit());
35        configured = true;
36        return true;
37    }
38
39    /// <inheritdoc />
40    public bool Install()
41    {
42        if (!configured || taskStateSegment is null) return false;
43        bool loaded = NativeMethods.LoadGlobalDescriptorTable(configuration.TableAddress.Value, (ushort)(RequiredBytes - 1),
44            configuration.KernelCodeSelector.Value, configuration.KernelDataSelector.Value);
45        return loaded && taskStateSegment.Install(configuration.TaskStateSelector);
46    }
47
48    /// <inheritdoc />
49    public SegmentSelector GetSelector(ushort descriptorIndex) => descriptorIndex switch
50    {
51        1 => configuration.KernelCodeSelector,
52        2 => configuration.KernelDataSelector,
53        3 => configuration.UserDataSelector,
54        4 => configuration.UserCodeSelector,
55        5 => configuration.TaskStateSelector,
56        _ => new SegmentSelector(0)
57    };
58
59    private static void WriteTaskStateDescriptor(ulong* table, int index, ulong address, uint limit)
60    {
61        ulong low = (limit & 0xFFFFUL) |
62                    ((address & 0xFFFFFFUL) << 16) |
63                    (0x89UL << 40) |
64                    (((ulong)limit & 0xF0000UL) << 32) |
65                    (((address >> 24) & 0xFFUL) << 56);
66        table[index] = low;
67        table[index + 1] = address >> 32;
68    }
69}
