4namespace NovaOryn.Memory.Heap;

6/// <summary>Implements a bounded monotonic early allocator over a caller-selected address range.</summary>
7/// <nova.when>Use when bootstrap metadata needs aligned raw storage and individual releases are unnecessary.</nova.when>
8/// <nova.depends>NovaOryn.Memory.Heap.Contracts</nova.depends>
9[SupportedArchitecture(SupportedArchitecture.All)]
10[BootStage(BootStage.ManagedBootstrap)]
11public sealed class BumpEarlyAllocator : IEarlyAllocator
12{
13    private readonly ulong _baseAddress;
14    private readonly ulong _length;
15    private ulong _offset;
16
17    /// <summary>Creates a bump allocator over one half-open address range.</summary>
18    /// <nova.when>Use after reserving a stable caller-owned range for bootstrap-lifetime allocations.</nova.when>
19    /// <nova.depends>The supplied range must be non-empty and must not overflow the 64-bit address space.</nova.depends>
20    /// <param name="baseAddress">First byte in the caller-owned range.</param>
21    /// <param name="length">Range length in bytes.</param>
22    /// <returns>A bump allocator positioned at the beginning of the supplied range.</returns>
23    /// <example><code>BumpEarlyAllocator allocator = new(0x100000, 65536);</code></example>
24    public BumpEarlyAllocator(ulong baseAddress, ulong length)
25    {
26        if (length == 0 || baseAddress > ulong.MaxValue - length) throw new ArgumentOutOfRangeException(nameof(length));
27        _baseAddress = baseAddress;
28        _length = length;
29    }
30
31    /// <summary>Attempts to allocate aligned bytes monotonically from the caller-owned range.</summary>
32    /// <nova.when>Use for bootstrap-lifetime allocations that need no individual release.</nova.when>
33    /// <nova.depends>A power-of-two alignment and remaining range capacity.</nova.depends>
34    /// <returns><see langword="true"/> when the aligned allocation fits.</returns>
35    /// <example><code>bool ok = allocator.TryAllocate(128, 16, out ulong address);</code></example>
36    public bool TryAllocate(ulong byteCount, ulong alignment, out ulong address)
37    {
38        address = 0;
39        if (byteCount == 0 || alignment == 0 || (alignment & (alignment - 1)) != 0) return false;
40        ulong current = _baseAddress + _offset;
41        if (current < _baseAddress) return false;
42        ulong mask = alignment - 1;
43        if (current > ulong.MaxValue - mask) return false;
44        ulong aligned = (current + mask) & ~mask;
45        if (aligned < _baseAddress) return false;
46        ulong used = aligned - _baseAddress;
47        if (used > _length || byteCount > _length - used) return false;
48        _offset = used + byteCount;
49        address = aligned;
50        return true;
51    }
52
53    /// <summary>Gets the remaining unallocated byte count.</summary>
54    /// <nova.when>Use to check remaining bootstrap capacity before requesting more storage.</nova.when>
55    /// <nova.depends>The allocator's current monotonic offset.</nova.depends>
56    /// <returns>The number of bytes not yet consumed by the bump allocator.</returns>
57    /// <example><code>ulong remaining = allocator.GetRemainingBytes();</code></example>
58    public ulong GetRemainingBytes() => _offset <= _length ? _length - _offset : 0UL;
59}
