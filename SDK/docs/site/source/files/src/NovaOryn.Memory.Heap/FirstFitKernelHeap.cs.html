6/// <summary>Provides a caller-backed first-fit heap over an existing raw address range.</summary>
7/// <nova.when>Use when a kernel wants a replaceable first-fit heap methodology over storage it already owns.</nova.when>
8/// <nova.depends>NovaOryn.Memory.Heap.Contracts and a valid writable caller-owned address range.</nova.depends>
9[SupportedArchitecture(SupportedArchitecture.All)]
10[BootStage(BootStage.ManagedBootstrap)]
11public sealed unsafe class FirstFitKernelHeap : IKernelHeap

23    /// <summary>Creates a first-fit allocator using caller-selected metadata capacity.</summary>
24    /// <nova.when>Use after reserving a writable raw range that this allocator may subdivide and zero-fill.</nova.when>
25    /// <nova.depends>The range must be non-empty, non-overflowing, writable, and remain valid for the allocator lifetime.</nova.depends>
26    /// <param name="baseAddress">First writable byte in the caller-owned range.</param>
27    /// <param name="length">Range length in bytes.</param>
28    /// <param name="maximumBlocks">Maximum number of allocation/free-list metadata records.</param>
29    /// <returns>A first-fit heap over the supplied range.</returns>
30    /// <example><code>FirstFitKernelHeap heap = new(baseAddress, 65536, 128);</code></example>
31    public FirstFitKernelHeap(ulong baseAddress, ulong length, int maximumBlocks)
32    {
33        if (length == 0 || maximumBlocks < 4) throw new ArgumentOutOfRangeException(nameof(maximumBlocks));
34        if (baseAddress > ulong.MaxValue - length) throw new ArgumentOutOfRangeException(nameof(length));
35        _length = length;
36        _starts = new ulong[maximumBlocks];
37        _lengths = new ulong[maximumBlocks];
38        _tokens = new ulong[maximumBlocks];
39        _states = new byte[maximumBlocks];
40        _starts[0] = baseAddress;
41        _lengths[0] = length;
42        _states[0] = 1;
43    }

50    /// <summary>Attempts to allocate one aligned raw byte range.</summary>
51    /// <nova.when>Use to allocate aligned raw storage and optionally zero-fill the returned bytes.</nova.when>
52    /// <nova.depends>A valid request, writable backing range, and available metadata capacity.</nova.depends>
53    /// <returns><see langword="true"/> when a live allocation is created.</returns>
54    /// <example><code>bool ok = heap.TryAllocate(KernelHeapAllocationRequest.Default(256), out KernelHeapAllocation allocation);</code></example>
55    public bool TryAllocate(KernelHeapAllocationRequest request, out KernelHeapAllocation allocation)
56    {
57        allocation = default;
58        if (request.ByteCount == 0 || request.Alignment == 0 || (request.Alignment & (request.Alignment - 1)) != 0) return false;
59        for (int i = 0; i < _states.Length; i++)
60        {
61            if (_states[i] != 1) continue;
62            ulong start = _starts[i];
63            ulong length = _lengths[i];
64            ulong mask = request.Alignment - 1;
65            if (start > ulong.MaxValue - mask) continue;
66            ulong aligned = (start + mask) & ~mask;
67            ulong prefix = aligned - start;
68            if (prefix > length || request.ByteCount > length - prefix) continue;
69            ulong suffix = length - prefix - request.ByteCount;
70            int firstSlot = prefix > 0 ? FindUnused(i, -1) : -1;
71            int secondSlot = suffix > 0 ? FindUnused(i, firstSlot) : -1;
72            if ((prefix > 0 && firstSlot < 0) || (suffix > 0 && secondSlot < 0)) return false;
73
74            ulong token = NextToken();
75            _states[i] = 2;
76            _starts[i] = aligned;
77            _lengths[i] = request.ByteCount;
78            _tokens[i] = token;
79            if (prefix > 0)
80            {
81                _states[firstSlot] = 1;
82                _starts[firstSlot] = start;
83                _lengths[firstSlot] = prefix;
84            }
85            if (suffix > 0)
86            {
87                _states[secondSlot] = 1;
88                _starts[secondSlot] = aligned + request.ByteCount;
89                _lengths[secondSlot] = suffix;
90            }
91            _allocated += request.ByteCount;
92            if (_allocated > _peak) _peak = _allocated;
93            _live++;
94            if (request.ZeroFill)
95            {
96                byte* pointer = (byte*)(nuint)aligned;
97                for (ulong n = 0; n < request.ByteCount; n++) pointer[n] = 0;
98            }
99            allocation = new KernelHeapAllocation(token, aligned, request.ByteCount);
100            return true;
101        }
102        return false;
103    }

105    /// <summary>Attempts to release one exact live allocation.</summary>
106    /// <nova.when>Use when an exact allocation returned by this heap is no longer needed.</nova.when>
107    /// <nova.depends>A live token/address/length tuple created by this heap.</nova.depends>
108    /// <returns><see langword="true"/> when the allocation was live and is now reusable.</returns>
109    /// <example><code>bool released = heap.TryRelease(allocation);</code></example>
110    public bool TryRelease(KernelHeapAllocation allocation)
111    {
112        for (int i = 0; i < _states.Length; i++)
113        {
114            if (_states[i] != 2 || _tokens[i] != allocation.Token || _starts[i] != allocation.Address || _lengths[i] != allocation.ByteCount) continue;
115            _states[i] = 1;
116            _tokens[i] = 0;
117            _allocated -= _lengths[i];
118            _live--;
119            Coalesce();
120            return true;
121        }
122        return false;
123    }

125    /// <summary>Gets current first-fit allocation accounting.</summary>
126    /// <nova.when>Use to inspect allocation pressure and free-list fragmentation.</nova.when>
127    /// <nova.depends>The allocator's current block metadata.</nova.depends>
128    /// <returns>An immutable statistics snapshot.</returns>
129    /// <example><code>KernelHeapStatistics statistics = heap.GetStatistics();</code></example>
130    public KernelHeapStatistics GetStatistics()
131    {
132        ulong free = 0;
133        int blocks = 0;
134        for (int i = 0; i < _states.Length; i++)
135        {
136            if (_states[i] != 1) continue;
137            free += _lengths[i];
138            blocks++;
139        }
140        return new KernelHeapStatistics(_length, _length, _allocated, free, _peak, _live, blocks);
141    }

143    private int FindUnused(int excluded, int excluded2)
144    {
145        for (int i = 0; i < _states.Length; i++) if (i != excluded && i != excluded2 && _states[i] == 0) return i;
146        return -1;
147    }
148
149    private ulong NextToken()
150    {
151        ulong token = _nextToken++;
152        if (token == 0) token = _nextToken++;
153        return token;
154    }
155
156    private void Coalesce()
157    {
158        bool changed = true;
159        while (changed)
160        {
161            changed = false;
162            for (int i = 0; i < _states.Length && !changed; i++)
163            {
164                if (_states[i] != 1) continue;
165                for (int j = 0; j < _states.Length; j++)
166                {
167                    if (i == j || _states[j] != 1) continue;
168                    if (_starts[i] > ulong.MaxValue - _lengths[i] || _starts[i] + _lengths[i] != _starts[j]) continue;
169                    _lengths[i] += _lengths[j];
170                    _states[j] = 0;
171                    _starts[j] = 0;
172                    _lengths[j] = 0;
173                    changed = true;
174                    break;
175                }
176            }
177        }
178    }
