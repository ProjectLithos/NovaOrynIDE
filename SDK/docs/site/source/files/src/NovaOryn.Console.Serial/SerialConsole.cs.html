12public sealed class SerialConsole : IConsole
13{
14    private SerialConfiguration _configuration;
15    private bool _initialized;
16
17    public bool Configure(SerialConfiguration configuration)
18    {
19        if (configuration.Port == 0 || configuration.BaudRate == 0)
20            throw new ArgumentOutOfRangeException(nameof(configuration));
21        _configuration = configuration;
22        return true;
23    }
24
25    public bool Initialize(BootContext boot)
26    {
27        _ = boot;
28        if (_configuration.Port == 0 && !Configure(SerialConfiguration.Com1())) return false;
29        ushort divisor = checked((ushort)(115200 / _configuration.BaudRate));
30        ushort port = _configuration.Port;
31        if (!Port.Write8((ushort)(port + 1), 0x00)) return false;
32        if (!Port.Write8((ushort)(port + 3), 0x80)) return false;
33        if (!Port.Write8(port, (byte)(divisor & 0xFF))) return false;
34        if (!Port.Write8((ushort)(port + 1), (byte)(divisor >> 8))) return false;
35        if (!Port.Write8((ushort)(port + 3), 0x03)) return false;
36        if (!Port.Write8((ushort)(port + 2), 0xC7)) return false;
37        if (!Port.Write8((ushort)(port + 4), 0x0B)) return false;
38        _initialized = true;
39        return true;
40    }
41
42    public bool Write(ReadOnlySpan<char> text)
43    {
44        if (!_initialized) return false;
45        foreach (char character in text)
46        {
47            if (character > 0x7F || !WriteByte((byte)character)) return false;
48        }
49        return true;
50    }
51
52    public bool WriteLine(ReadOnlySpan<char> text) => Write(text) && WriteLine();
53    public bool WriteLine() => Write("\r\n");
54
55    private bool WriteByte(byte value)
56    {
57        for (uint attempt = 0; attempt < 1_000_000; attempt++)
58        {
59            if (!Port.TryRead8((ushort)(_configuration.Port + 5), out byte status)) return false;
60            if ((status & 0x20) != 0) return Port.Write8(_configuration.Port, value);
61        }
62        return false;
63    }
64}
