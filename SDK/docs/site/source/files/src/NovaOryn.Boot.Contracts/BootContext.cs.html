21public readonly struct PixelBitMask
22{
23    public PixelBitMask(uint red, uint green, uint blue, uint reserved)
24    {
25        Red = red;
26        Green = green;
27        Blue = blue;
28        Reserved = reserved;
29    }
30
31    public uint Red { get; }
32    public uint Green { get; }
33    public uint Blue { get; }
34    public uint Reserved { get; }
35
36    public bool IsDirectColor()
37    {
38        if (Red == 0 || Green == 0 || Blue == 0) return false;
39        if ((Red & Green) != 0 || (Red & Blue) != 0 || (Green & Blue) != 0) return false;
40        if (Reserved != 0 && ((Reserved & Red) != 0 || (Reserved & Green) != 0 || (Reserved & Blue) != 0)) return false;
41        return IsContiguous(Red) && IsContiguous(Green) && IsContiguous(Blue);
42    }
43
44    private static bool IsContiguous(uint mask)
45    {
46        while ((mask & 1U) == 0U) mask >>= 1;
47        while ((mask & 1U) != 0U) mask >>= 1;
48        return mask == 0U;
49    }
50}

72    public PhysicalAddress Address { get; }
73    public ulong SizeInBytes { get; }
74    public uint Width { get; }

77    public FramebufferPixelFormat PixelFormat { get; }
78    public PixelBitMask PixelMask { get; }
79
80    public bool IsAvailable()
81    {
82        if (Address.Value == 0 || SizeInBytes == 0 || Width == 0 || Height == 0) return false;
83        if (PixelsPerScanLine < Width || PixelFormat > FramebufferPixelFormat.BitMask) return false;
84        ulong bytesPerScanLine = checked((ulong)PixelsPerScanLine * 4UL);
85        if ((ulong)Height > SizeInBytes / bytesPerScanLine) return false;
86        return PixelFormat != FramebufferPixelFormat.BitMask || PixelMask.IsDirectColor();
87    }
