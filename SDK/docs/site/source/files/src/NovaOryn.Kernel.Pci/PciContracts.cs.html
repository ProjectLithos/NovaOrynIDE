16    /// <summary>PCI segment group.</summary>
17    public UInt16 Segment { get; }
18    /// <summary>PCI bus number.</summary>
19    public Byte Bus { get; }
20    /// <summary>PCI device number.</summary>
21    public Byte Device { get; }
22    /// <summary>PCI function number.</summary>
23    public Byte Function { get; }
24    /// <summary>Packs the complete segment/bus/device/function address into 32 bits.</summary>
25    public UInt32 Encode()=>((UInt32)Segment<<16)|((UInt32)Bus<<8)|((UInt32)Device<<3)|Function;
26    /// <summary>Decodes a packed PCI location.</summary>
27    public static PciLocation Decode(UInt32 value)=>new((UInt16)(value>>16),(Byte)(value>>8),(Byte)((value>>3)&31U),(Byte)(value&7U));
