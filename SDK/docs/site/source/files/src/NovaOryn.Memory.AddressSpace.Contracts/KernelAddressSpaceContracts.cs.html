5/// <summary>Identifies the purpose of a reserved virtual-address region.</summary>
6public enum KernelAddressSpaceRegionKind
7{
8    /// <summary>User-mode virtual address space.</summary>
9    User = 0,
10    /// <summary>Kernel image and static kernel mappings.</summary>
11    KernelImage = 1,
12    /// <summary>Future kernel heap reservation.</summary>
13    KernelHeap = 2,
14    /// <summary>Kernel stacks and their guard-page neighbourhood.</summary>
15    KernelStacks = 3,
16    /// <summary>Direct physical-memory mapping window.</summary>
17    DirectPhysicalMap = 4,
18    /// <summary>Memory-mapped device I/O window.</summary>
19    Mmio = 5,
20    /// <summary>Dedicated page-table access window.</summary>
21    PageTableWindow = 6

24/// <summary>Describes one half-open virtual-address region in a kernel layout.</summary>
25/// <nova.when>Use when defining, validating, or querying a kernel virtual-address policy.</nova.when>
26/// <nova.depends>Canonical architecture address rules and page alignment.</nova.depends>
27public readonly struct KernelAddressSpaceRegion
28{
29    private KernelAddressSpaceRegion(KernelAddressSpaceRegionKind kind, ulong baseAddress, ulong length)
30    {
31        Kind = kind; BaseAddress = baseAddress; Length = length;
32    }
33    /// <summary>Gets the role assigned to this region.</summary>
34    public KernelAddressSpaceRegionKind Kind { get; }
35    /// <summary>Gets the first virtual byte in the region.</summary>
36    public ulong BaseAddress { get; }
37    /// <summary>Gets the region length in bytes.</summary>
38    public ulong Length { get; }
39    /// <summary>Gets the exclusive end address, or zero when arithmetic would overflow.</summary>
40    public ulong EndExclusive => BaseAddress <= ulong.MaxValue - Length ? BaseAddress + Length : 0UL;
41
42    /// <summary>Creates a non-empty, page-aligned region without arithmetic overflow.</summary>
43    /// <nova.when>Use before placing a region into a kernel address-space layout.</nova.when>
44    /// <nova.depends>4 KiB base and length alignment.</nova.depends>
45    /// <returns><see langword="true"/> when the region is valid.</returns>
46    /// <example><code>bool ok = KernelAddressSpaceRegion.TryCreate(KernelAddressSpaceRegionKind.KernelHeap, 0xFFFF810000000000UL, 0x10000000000UL, out var region);</code></example>
47    public static bool TryCreate(KernelAddressSpaceRegionKind kind, ulong baseAddress, ulong length, out KernelAddressSpaceRegion region)
48    {
49        region = default;
50        if (length == 0 || (baseAddress & 0xFFFUL) != 0 || (length & 0xFFFUL) != 0) return false;
51        if (baseAddress > ulong.MaxValue - length) return false;
52        region = new KernelAddressSpaceRegion(kind, baseAddress, length);
53        return true;
54    }
55
56    /// <summary>Determines whether one virtual byte lies inside this half-open region.</summary>
57    /// <nova.when>Use when routing an address to the region responsible for it.</nova.when>
58    /// <nova.depends>A validated region.</nova.depends>
59    /// <returns><see langword="true"/> when the address is greater than or equal to the base and below the exclusive end.</returns>
60    /// <example><code>bool inside = region.Contains(region.BaseAddress);</code></example>
61    public bool Contains(ulong address) => address >= BaseAddress && address < EndExclusive;
62
63    /// <summary>Determines whether two half-open regions overlap.</summary>
64    /// <nova.when>Use while validating a custom kernel layout.</nova.when>
65    /// <nova.depends>Two validated regions.</nova.depends>
66    /// <returns><see langword="true"/> when at least one byte belongs to both regions.</returns>
67    /// <example><code>bool overlap = first.Overlaps(second);</code></example>
68    public bool Overlaps(KernelAddressSpaceRegion other) => BaseAddress < other.EndExclusive && other.BaseAddress < EndExclusive;
69}

71/// <summary>Defines the complete virtual-address policy consumed by later kernel allocators and subsystems.</summary>
72/// <nova.when>Use to select the permanent regions for a NovaOryn kernel or a custom OS built with the SDK.</nova.when>
73/// <nova.depends>Seven validated, non-overlapping address-space regions.</nova.depends>
74public readonly struct KernelAddressSpaceLayout
75{
76    private KernelAddressSpaceLayout(KernelAddressSpaceRegion user, KernelAddressSpaceRegion kernelImage, KernelAddressSpaceRegion heap, KernelAddressSpaceRegion stacks, KernelAddressSpaceRegion directMap, KernelAddressSpaceRegion mmio, KernelAddressSpaceRegion pageTables)
77    { User = user; KernelImage = kernelImage; KernelHeap = heap; KernelStacks = stacks; DirectPhysicalMap = directMap; Mmio = mmio; PageTableWindow = pageTables; }
78    /// <summary>Gets the user-space region.</summary>
79    public KernelAddressSpaceRegion User { get; }
80    /// <summary>Gets the kernel-image region.</summary>
81    public KernelAddressSpaceRegion KernelImage { get; }
82    /// <summary>Gets the future kernel-heap region.</summary>
83    public KernelAddressSpaceRegion KernelHeap { get; }
84    /// <summary>Gets the kernel-stack region.</summary>
85    public KernelAddressSpaceRegion KernelStacks { get; }
86    /// <summary>Gets the direct physical-map region.</summary>
87    public KernelAddressSpaceRegion DirectPhysicalMap { get; }
88    /// <summary>Gets the MMIO region.</summary>
89    public KernelAddressSpaceRegion Mmio { get; }
90    /// <summary>Gets the page-table access region.</summary>
91    public KernelAddressSpaceRegion PageTableWindow { get; }
92
93    /// <summary>Creates a complete layout after checking region roles and pairwise separation.</summary>
94    /// <nova.when>Use for custom layouts before handing them to an architecture-specific validator.</nova.when>
95    /// <nova.depends>Correct region kinds and non-overlap.</nova.depends>
96    /// <returns><see langword="true"/> when every required region is present and disjoint.</returns>
97    /// <example><code>bool ok = KernelAddressSpaceLayout.TryCreate(user, image, heap, stacks, directMap, mmio, pageTables, out var layout);</code></example>
98    public static bool TryCreate(KernelAddressSpaceRegion user, KernelAddressSpaceRegion kernelImage, KernelAddressSpaceRegion heap, KernelAddressSpaceRegion stacks, KernelAddressSpaceRegion directMap, KernelAddressSpaceRegion mmio, KernelAddressSpaceRegion pageTables, out KernelAddressSpaceLayout layout)
99    {
100        layout = default;
101        KernelAddressSpaceRegion[] r = [user, kernelImage, heap, stacks, directMap, mmio, pageTables];
102        KernelAddressSpaceRegionKind[] k = [KernelAddressSpaceRegionKind.User, KernelAddressSpaceRegionKind.KernelImage, KernelAddressSpaceRegionKind.KernelHeap, KernelAddressSpaceRegionKind.KernelStacks, KernelAddressSpaceRegionKind.DirectPhysicalMap, KernelAddressSpaceRegionKind.Mmio, KernelAddressSpaceRegionKind.PageTableWindow];
103        for (int i=0;i<r.Length;i++) { if (r[i].Kind != k[i] || r[i].Length == 0) return false; for(int j=i+1;j<r.Length;j++) if(r[i].Overlaps(r[j])) return false; }
104        layout = new KernelAddressSpaceLayout(user, kernelImage, heap, stacks, directMap, mmio, pageTables); return true;
105    }
106
107    /// <summary>Gets the region assigned to one well-known purpose.</summary>
108    /// <nova.when>Use when a subsystem needs its designated virtual reservation.</nova.when>
109    /// <nova.depends>A validated layout.</nova.depends>
110    /// <returns>The matching region.</returns>
111    /// <example><code>var heap = layout.GetRegion(KernelAddressSpaceRegionKind.KernelHeap);</code></example>
112    public KernelAddressSpaceRegion GetRegion(KernelAddressSpaceRegionKind kind) => kind switch { KernelAddressSpaceRegionKind.User => User, KernelAddressSpaceRegionKind.KernelImage => KernelImage, KernelAddressSpaceRegionKind.KernelHeap => KernelHeap, KernelAddressSpaceRegionKind.KernelStacks => KernelStacks, KernelAddressSpaceRegionKind.DirectPhysicalMap => DirectPhysicalMap, KernelAddressSpaceRegionKind.Mmio => Mmio, _ => PageTableWindow };
113}
