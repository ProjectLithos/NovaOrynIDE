94/// <summary>Controls interrupt delivery while keeping routing technology private from drivers.</summary>
95public interface IInterruptController
96{
97    /// <summary>Gets the capabilities of the active controller stack.</summary>
98    InterruptControllerCapabilities GetCapabilities();
99    /// <summary>Allocates a vector suitable for a device route.</summary>
100    byte AllocateVector();
101    /// <summary>Releases a vector after every route using it has been removed.</summary>
102    bool ReleaseVector(byte vector);
103    /// <summary>Creates or replaces one hardware interrupt route.</summary>
104    InterruptRouteResult Route(InterruptRouteConfiguration configuration);
105    /// <summary>Removes a route.</summary>
106    bool RemoveRoute(InterruptRouteHandle handle);
107    /// <summary>Masks a routed interrupt.</summary>
108    bool Mask(InterruptRouteHandle handle);
109    /// <summary>Unmasks a routed interrupt.</summary>
110    bool Unmask(InterruptRouteHandle handle);
111    /// <summary>Changes the processor affinity of a route.</summary>
112    bool SetAffinity(InterruptRouteHandle handle, InterruptAffinity affinity);
113    /// <summary>Changes the logical priority of a route.</summary>
114    bool SetPriority(InterruptRouteHandle handle, byte priority);
115    /// <summary>Signals end-of-interrupt for a delivered vector.</summary>
116    bool EndOfInterrupt(byte vector);
117    /// <summary>Sends an interprocessor interrupt.</summary>
118    bool SendInterprocessorInterrupt(InterprocessorInterrupt interrupt);
119    /// <summary>Builds an MSI or MSI-X message for a route.</summary>
120    MessageSignalledInterrupt CreateMessage(InterruptRouteHandle handle);
121}
