74/// <summary>Provides the default first-fit, page-backed, non-executable kernel heap inside the standard heap reservation.</summary>

121    /// <summary>Commits the first 64 KiB of the standard kernel-heap reservation.</summary>
122    /// <returns><see langword="true"/> when dependencies and initial mappings are ready.</returns>
123    public static Boolean Initialize()
124    {
125        if (_initialized)
126        {
127            _status = KernelHeapStatus.AlreadyInitialized;
128            return false;
129        }
130        if (!KernelAddressSpace.IsInitialized() || !KernelVirtualMemory.IsInitialized() || !KernelPhysicalMemory.IsInitialized())
131        {
132            _status = KernelHeapStatus.DependencyNotInitialized;
133            return false;
134        }
135        Reset();
136        if (!Grow(GrowthPages)) return false;
137        _initialized = true;
138        _status = KernelHeapStatus.Success;
139        return true;
140    }

142    /// <summary>Allocates raw virtual bytes using address-ordered first fit and optional zero filling.</summary>
143    /// <returns><see langword="true"/> when a live allocation was created.</returns>
144    public static Boolean TryAllocate(UInt64 byteCount, UInt64 alignment, Boolean zeroFill, out KernelHeapAllocation allocation)
145    {
146        allocation = default;
147        if (!_initialized)
148        {
149            _status = KernelHeapStatus.DependencyNotInitialized;
150            return false;
151        }
152        if (byteCount == 0UL || alignment == 0UL || (alignment & (alignment - 1UL)) != 0UL || alignment > PageSize)
153        {
154            _status = KernelHeapStatus.InvalidParameter;
155            return false;
156        }
157        if (byteCount > 0xFFFFFFFFFFFFFFFFUL - (alignment - 1UL))
158        {
159            _status = KernelHeapStatus.InvalidParameter;
160            return false;
161        }
162
163        for (Int32 attempt = 0; attempt < 2; attempt++)
164        {
165            if (TryAllocateExisting(byteCount, alignment, zeroFill, out allocation)) return true;
166            UInt64 required = byteCount + alignment - 1UL;
167            if (required > 0xFFFFFFFFFFFFFFFFUL - (PageSize - 1UL))
168            {
169                _status = KernelHeapStatus.InvalidParameter;
170                return false;
171            }
172            UInt64 pages = (required + PageSize - 1UL) / PageSize;
173            if (pages < GrowthPages) pages = GrowthPages;
174            if (!Grow(pages)) return false;
175        }
176        _status = KernelHeapStatus.OutOfMemory;
177        return false;
178    }

180    /// <summary>Releases one exact live allocation and coalesces adjacent free blocks.</summary>
181    /// <returns><see langword="true"/> when the allocation was live.</returns>
182    public static Boolean TryRelease(KernelHeapAllocation allocation)
183    {
184        if (!_initialized)
185        {
186            _status = KernelHeapStatus.DependencyNotInitialized;
187            return false;
188        }
189        fixed (UInt64* starts = _state.Starts)
190        fixed (UInt64* lengths = _state.Lengths)
191        fixed (UInt64* tokens = _state.Tokens)
192        fixed (Byte* states = _state.States)
193        {
194            for (Int32 i = 0; i < MaximumBlocks; i++)
195            {
196                if (states[i] != 2 || tokens[i] != allocation.Token || starts[i] != allocation.Address || lengths[i] != allocation.ByteCount) continue;
197                states[i] = 1;
198                tokens[i] = 0UL;
199                _allocated -= lengths[i];
200                _live--;
201                Coalesce();
202                _status = KernelHeapStatus.Success;
203                return true;
204            }
205        }
206        _status = KernelHeapStatus.AllocationNotFound;
207        return false;
208    }

218        {

220            {

229    private static Boolean TryAllocateExisting(UInt64 bytes, UInt64 alignment, Boolean zeroFill, out KernelHeapAllocation allocation)
230    {
231        allocation = default;
232        fixed (UInt64* starts = _state.Starts)
233        fixed (UInt64* lengths = _state.Lengths)
234        fixed (UInt64* tokens = _state.Tokens)
235        fixed (Byte* states = _state.States)
236        {
237            for (Int32 i = 0; i < MaximumBlocks; i++)
238            {
239                if (states[i] != 1) continue;
240                UInt64 start = starts[i];
241                UInt64 length = lengths[i];
242                UInt64 mask = alignment - 1UL;
243                if (start > 0xFFFFFFFFFFFFFFFFUL - mask) continue;
244                UInt64 aligned = (start + mask) & ~mask;
245                UInt64 prefix = aligned - start;
246                if (prefix > length || bytes > length - prefix) continue;
247                UInt64 suffix = length - prefix - bytes;
248                Int32 firstSlot = prefix > 0UL ? FindUnused(i, -1) : -1;
249                Int32 secondSlot = suffix > 0UL ? FindUnused(i, firstSlot) : -1;
250                if ((prefix > 0UL && firstSlot < 0) || (suffix > 0UL && secondSlot < 0))
251                {
252                    _status = KernelHeapStatus.MetadataCapacityExhausted;
253                    return false;
254                }
255
256                UInt64 token = NextToken();
257                states[i] = 2;
258                starts[i] = aligned;
259                lengths[i] = bytes;
260                tokens[i] = token;
261                if (prefix > 0UL)
262                {
263                    states[firstSlot] = 1;
264                    starts[firstSlot] = start;
265                    lengths[firstSlot] = prefix;
266                }
267                if (suffix > 0UL)
268                {
269                    states[secondSlot] = 1;
270                    starts[secondSlot] = aligned + bytes;
271                    lengths[secondSlot] = suffix;
272                }
273                _allocated += bytes;
274                if (_allocated > _peak) _peak = _allocated;
275                _live++;
276                if (zeroFill)
277                {
278                    Byte* pointer = (Byte*)(nuint)aligned;
279                    for (UInt64 n = 0UL; n < bytes; n++) pointer[n] = 0;
280                }
281                allocation = new KernelHeapAllocation(token, aligned, bytes);
282                _status = KernelHeapStatus.Success;
283                return true;
284            }
285        }
286        return false;
287    }

289    private static Boolean Grow(UInt64 pages)
290    {
291        if (pages == 0UL || pages > 0xFFFFFFFFFFFFFFFFUL / PageSize)
292        {
293            _status = KernelHeapStatus.InvalidParameter;
294            return false;
295        }
296        UInt64 bytes = pages * PageSize;
297        if (_committed > KernelAddressSpace.KernelHeapLength || bytes > KernelAddressSpace.KernelHeapLength - _committed)
298        {
299            _status = KernelHeapStatus.OutOfMemory;
300            return false;
301        }
302        Int32 slot = FindUnused(-1, -1);
303        if (slot < 0)
304        {
305            _status = KernelHeapStatus.MetadataCapacityExhausted;
306            return false;
307        }
308        if (!KernelPhysicalMemory.TryAllocate(pages, 1UL, out KernelPhysicalAllocation physical))
309        {
310            _status = KernelHeapStatus.OutOfMemory;
311            return false;
312        }
313
314        UInt64 mapped = 0UL;
315        KernelVirtualMemoryProtection protection = KernelVirtualMemoryProtection.Read | KernelVirtualMemoryProtection.Write | KernelVirtualMemoryProtection.Global;
316        for (UInt64 page = 0UL; page < pages; page++)
317        {
318            UInt64 virtualAddress = KernelAddressSpace.KernelHeapBase + _committed + page * PageSize;
319            UInt64 physicalAddress = physical.StartAddress + page * PageSize;
320            if (!KernelVirtualMemory.TryMap(virtualAddress, physicalAddress, KernelVirtualPageSize.Page4KiB, protection))
321            {
322                for (UInt64 undo = 0UL; undo < mapped; undo++)
323                    KernelVirtualMemory.TryUnmap(KernelAddressSpace.KernelHeapBase + _committed + undo * PageSize);
324                KernelPhysicalMemory.TryRelease(physical);
325                _status = KernelHeapStatus.MappingFailed;
326                return false;
327            }
328            mapped++;
329        }
330
331        fixed (UInt64* starts = _state.Starts)
332        fixed (UInt64* lengths = _state.Lengths)
333        fixed (Byte* states = _state.States)
334        {
335            states[slot] = 1;
336            starts[slot] = KernelAddressSpace.KernelHeapBase + _committed;
337            lengths[slot] = bytes;
338        }
339        _committed += bytes;
340        Coalesce();
341        return true;
342    }

344    private static Int32 FindUnused(Int32 excluded, Int32 excluded2)
345    {
346        fixed (Byte* states = _state.States)
347        {
348            for (Int32 i = 0; i < MaximumBlocks; i++)
349                if (i != excluded && i != excluded2 && states[i] == 0) return i;
350        }
351        return -1;
352    }

354    private static void Coalesce()
355    {
356        Boolean changed = true;
357        fixed (UInt64* starts = _state.Starts)
358        fixed (UInt64* lengths = _state.Lengths)
359        fixed (Byte* states = _state.States)
360        {
361            while (changed)
362            {
363                changed = false;
364                for (Int32 i = 0; i < MaximumBlocks && !changed; i++)
365                {
366                    if (states[i] != 1) continue;
367                    for (Int32 j = 0; j < MaximumBlocks; j++)
368                    {
369                        if (i == j || states[j] != 1) continue;
370                        if (starts[i] > 0xFFFFFFFFFFFFFFFFUL - lengths[i]) continue;
371                        if (starts[i] + lengths[i] != starts[j]) continue;
372                        lengths[i] += lengths[j];
373                        states[j] = 0;
374                        starts[j] = 0UL;
375                        lengths[j] = 0UL;
376                        changed = true;
377                        break;
378                    }
379                }
380            }
381        }
382    }

385    {
386        UInt64 token = _nextToken++;
