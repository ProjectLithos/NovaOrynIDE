5/// <summary>Reports the state of the bounded no-heap bootstrap allocator.</summary>
6public enum KernelEarlyAllocatorStatus
7{
8    /// <summary>The operation completed successfully.</summary>
9    Success = 0,
10    /// <summary>The early allocator has not been initialized.</summary>
11    NotInitialized = 1,
12    /// <summary>The early allocator was already initialized.</summary>
13    AlreadyInitialized = 2,
14    /// <summary>The request has an invalid size or alignment.</summary>
15    InvalidParameter = 3,
16    /// <summary>The fixed early arena cannot satisfy the request.</summary>
17    OutOfMemory = 4

20/// <summary>Provides a fixed 64 KiB monotonic arena for metadata required before the page-backed heap is online.</summary>
21public static unsafe class KernelEarlyAllocator
22{
23    private const UInt64 Capacity = 65536UL;
24
25    private unsafe struct State
26    {
27        internal fixed Byte Buffer[65536];
28    }
29
30#pragma warning disable CS0169 // Fixed-buffer member access is not counted as use of the containing freestanding state field.
31    private static State _state;
32#pragma warning restore CS0169
33    private static UInt64 _offset;
34    private static Boolean _initialized;
35    private static KernelEarlyAllocatorStatus _status;
36
37    /// <summary>Gets whether the early arena is ready.</summary>
38    public static Boolean IsInitialized() => _initialized;
39
40    /// <summary>Gets the most recent early-allocation status.</summary>
41    public static KernelEarlyAllocatorStatus GetLastStatus() => _status;
42
43    /// <summary>Initializes the fixed early arena.</summary>
44    /// <returns><see langword="true"/> on first initialization.</returns>
45    public static Boolean Initialize()
46    {
47        if (_initialized)
48        {
49            _status = KernelEarlyAllocatorStatus.AlreadyInitialized;
50            return false;
51        }
52        _offset = 0UL;
53        _initialized = true;
54        _status = KernelEarlyAllocatorStatus.Success;
55        return true;
56    }
57
58    /// <summary>Allocates aligned bytes monotonically from fixed kernel storage.</summary>
59    /// <returns><see langword="true"/> when capacity remains.</returns>
60    public static Boolean TryAllocate(UInt64 byteCount, UInt64 alignment, out UInt64 address)
61    {
62        address = 0UL;
63        if (!_initialized)
64        {
65            _status = KernelEarlyAllocatorStatus.NotInitialized;
66            return false;
67        }
68        if (byteCount == 0UL || alignment == 0UL || (alignment & (alignment - 1UL)) != 0UL)
69        {
70            _status = KernelEarlyAllocatorStatus.InvalidParameter;
71            return false;
72        }
73
74        fixed (Byte* buffer = _state.Buffer)
75        {
76            UInt64 baseAddress = (UInt64)(nuint)buffer;
77            UInt64 current = baseAddress + _offset;
78            UInt64 mask = alignment - 1UL;
79            if (current > 0xFFFFFFFFFFFFFFFFUL - mask)
80            {
81                _status = KernelEarlyAllocatorStatus.InvalidParameter;
82                return false;
83            }
84            UInt64 aligned = (current + mask) & ~mask;
85            UInt64 used = aligned - baseAddress;
86            if (used > Capacity || byteCount > Capacity - used)
87            {
88                _status = KernelEarlyAllocatorStatus.OutOfMemory;
89                return false;
90            }
91            _offset = used + byteCount;
92            address = aligned;
93        }
94        _status = KernelEarlyAllocatorStatus.Success;
95        return true;
96    }
97
98    /// <summary>Gets unused early-arena capacity.</summary>
99    /// <returns>Remaining bytes.</returns>
100    public static UInt64 GetRemainingBytes() => _initialized ? Capacity - _offset : 0UL;
101}
